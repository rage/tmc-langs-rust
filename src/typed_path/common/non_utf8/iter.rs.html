7/// An iterator over the [`Component`]s of a [`Path`], as [`[u8]`] slices.
8///
9/// This `struct` is created by the [`iter`] method on [`Path`].
10/// See its documentation for more.
11///
12/// [`iter`]: Path::iter
13#[derive(Clone)]
14pub struct Iter<'a, T>
15where
16    T: Encoding,
17{
18    _encoding: PhantomData<T>,
19    inner: <T as Encoding>::Components<'a>,
20}
21
22impl<'a, T> Iter<'a, T>
23where
24    T: Encoding,
25{
26    pub(crate) fn new(inner: <T as Encoding>::Components<'a>) -> Self {
27        Self {
28            _encoding: PhantomData,
29            inner,
30        }
31    }
32
33    /// Extracts a slice corresponding to the portion of the path remaining for iteration.
34    ///
35    /// # Examples
36    ///
37    /// ```
38    /// use typed_path::{Path, UnixEncoding};
39    ///
40    /// // NOTE: A path cannot be created on its own without a defined encoding
41    /// let mut iter = Path::<UnixEncoding>::new("/tmp/foo/bar.txt").iter();
42    /// iter.next();
43    /// iter.next();
44    ///
45    /// assert_eq!(Path::<UnixEncoding>::new("foo/bar.txt"), iter.as_path());
46    /// ```
47    pub fn as_path(&self) -> &Path<T> {
48        Path::new(self.inner.as_bytes())
49    }
50}

52impl<'a, T> fmt::Debug for Iter<'a, T>
53where
54    T: for<'enc> Encoding + 'a,
55{
56    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
57        struct DebugHelper<'a, T>(&'a Path<T>)
58        where
59            T: for<'enc> Encoding;
60
61        impl<T> fmt::Debug for DebugHelper<'_, T>
62        where
63            T: for<'enc> Encoding,
64        {
65            fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
66                f.debug_list().entries(self.0.iter()).finish()
67            }
68        }
69
70        f.debug_tuple(stringify!(Iter))
71            .field(&DebugHelper(self.as_path()))
72            .finish()
73    }
74}

126/// An iterator over [`Path`] and its ancestors.
127///
128/// This `struct` is created by the [`ancestors`] method on [`Path`].
129/// See its documentation for more.
130///
131/// # Examples
132///
133/// ```
134/// use typed_path::{Path, UnixEncoding};
135///
136/// // NOTE: A path cannot be created on its own without a defined encoding
137/// let path = Path::<UnixEncoding>::new("/foo/bar");
138///
139/// for ancestor in path.ancestors() {
140///     println!("{}", ancestor.display());
141/// }
142/// ```
143///
144/// [`ancestors`]: Path::ancestors
145#[derive(Copy, Clone, Debug)]
146pub struct Ancestors<'a, T>
147where
148    T: for<'enc> Encoding,
149{
150    pub(crate) next: Option<&'a Path<T>>,
151}
152
153impl<'a, T> Iterator for Ancestors<'a, T>
154where
155    T: for<'enc> Encoding,
156{
157    type Item = &'a Path<T>;
158
159    #[inline]
160    fn next(&mut self) -> Option<Self::Item> {
161        let next = self.next;
162        self.next = next.and_then(Path::parent);
163        next
164    }
165}
