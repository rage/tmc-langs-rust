1//! Wire types for the exercise-services client API (`/api/v0/exercise-services/client`), the
2//! HTTP surface a native client authenticates against and exchanges for course, exercise,
3//! and submission data. Consumed externally by `tmc-langs-rust`, so it stays free of
4//! server-internal dependencies; the `openapi` feature adds `utoipa::ToSchema` derives for the
5//! server's own OpenAPI generation and is off by default for other consumers.

7use serde::{Deserialize, Serialize};
8use std::fmt::Debug;
9use uuid::Uuid;

197/// Why a client stores an answer snapshot. Neither purpose is graded, uses a try or awards points,
198/// and both are accepted after the deadline and when out of tries.
199#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
200#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
201#[serde(rename_all = "snake_case")]
202pub enum SnapshotPurpose {
203    /// To share through `submissions/{id}/share`. Never deleted to make room for others, so
204    /// rate limited per user and exercise.
205    Paste,
206    /// To keep the answer before discarding local changes. Only the newest few per user and
207    /// exercise are kept, a number the host sets; storing another deletes the oldest.
208    Backup,
209}

365/// The current user's progress on a single exercise.
366///
367/// `standing` is the client's passed/failed signal. `completed` is the activity stage, which
368/// an exercise without peer or self review reaches on any graded submission, even one
369/// worth 0 points.
370#[derive(Debug, Serialize, Deserialize)]
371#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
372pub struct ExerciseProgress {
373    pub exercise_id: Uuid,
374    /// Points the user has been awarded, `0.0` when the user has no state for the exercise.
375    pub score_given: f32,
376    /// The maximum points obtainable from the exercise.
377    pub score_maximum: i32,
378    /// `true` once the exercise has reached the `Completed` activity stage.
379    pub completed: bool,
380    /// `true` once the user has started or submitted the exercise (any activity stage past
381    /// the initial one), regardless of whether it is completed.
382    pub attempted: bool,
383    /// Absent only from a host that predates the field.
384    #[serde(default)]
385    pub standing: Option<ExerciseStanding>,
386    /// How many graded submissions the exercise accepts per slide. Absent when tries are
387    /// unlimited, and from a host that predates the field.
388    #[serde(default)]
389    pub max_tries: Option<i32>,
390    /// How many of `max_tries` are left on the slide the user answers. Absent when tries are
391    /// unlimited, and from a host that predates the field.
392    #[serde(default)]
393    pub tries_left: Option<i32>,
394    /// What the user's points wait for while a teacher review holds them back; `score_given` is
395    /// `0.0` until then. Absent when no points are held back, including when the answer got its
396    /// points before the course turned chapter locking on, and from a host that predates the field.
397    #[serde(default)]
398    pub pending_teacher_review: Option<PendingTeacherReview>,
399    /// `true` once the exercise takes no more submissions from the user, whatever the chapter's
400    /// `locking_status` says: their answer was locked for review, graded by a teacher, or locked
401    /// unanswered with its chapter. Only a teacher resetting the exercise lifts it. Submits are
402    /// refused with 403 `chapter_completed_and_locked` while the chapter is locked for the user,
403    /// and with `exercise_locked` otherwise. Absent from a host that predates the field.
404    #[serde(default)]
405    pub exercise_locked: bool,
406}
