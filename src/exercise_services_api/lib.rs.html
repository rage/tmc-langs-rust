1//! Wire types for the exercise-services client API (`/api/v0/exercise-services/client`), the
2//! HTTP surface a native client authenticates against and exchanges for course, exercise,
3//! and submission data. Consumed externally by `tmc-langs-rust`, so it stays free of
4//! server-internal dependencies; the `openapi` feature adds `utoipa::ToSchema` derives for the
5//! server's own OpenAPI generation and is off by default for other consumers.
6use chrono::{DateTime, Utc};
7use serde::{Deserialize, Serialize};
8use std::fmt::Debug;
9use uuid::Uuid;
10
11/// The token exchanged at the OAuth2 token endpoint; its `access_token` is the bearer token sent
12/// with every other request in this API.
13pub type Token =
14    oauth2::StandardTokenResponse<oauth2::EmptyExtraTokenFields, oauth2::basic::BasicTokenType>;

16/// A course the current user can browse or is enrolled in.
17#[derive(Debug, Serialize, Deserialize)]
18#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
19pub struct Course {
20    pub id: Uuid,
21    pub slug: String,
22    pub name: String,
23    pub description: Option<String>,
24    /// Denormalized so a client needn't resolve the owning organization separately.
25    pub organization_name: String,
26    /// The teacher's AI-use policy for the course, uninterpreted: what each value permits is the
27    /// client's decision. Absent from a host that predates the field, which reads as `not_set`.
28    #[serde(default)]
29    pub ai_policy: CourseAiPolicy,
30}

32/// A course's AI-use policy as chosen by its teacher.
33#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize)]
34#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
35#[serde(rename_all = "snake_case")]
36pub enum CourseAiPolicy {
37    /// The teacher has not chosen a policy. AI use is not allowed by default.
38    #[default]
39    NotSet,
40    /// AI may not be used at any point.
41    NoAi,
42    /// AI may be used for planning, such as brainstorming or outlining, but not in the final work.
43    PlanningOnly,
44    /// AI may be used for specific tasks only, with disclosure.
45    Limited,
46    /// AI may be used freely, with disclosure.
47    FullUse,
48    /// AI use is expected.
49    Required,
50    /// A value newer than this crate. Deserialize-only: serializing it is an error, and the
51    /// OpenAPI schema omits it.
52    #[serde(other, skip_serializing)]
53    Unrecognized,
54}

106/// Which of the two representations an answer is: the JSON in `data_json`, or the files in
107/// `data_files`. A request that omits it means `Json`.
108#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
109#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
110#[serde(rename_all = "snake_case")]
111pub enum AnswerKind {
112    Json,
113    File,
114}

116/// A file the host stored on a client's behalf. The host assigns `id`; a client never invents
117/// one. Returned by the upload endpoint and echoed back by submission download.
118#[derive(Debug, Serialize, Deserialize)]
119#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
120pub struct AnswerFile {
121    /// The host's file id. Names this file in a submit request.
122    pub id: Uuid,
123    /// The original file name the client sent.
124    pub name: String,
125    pub mime: String,
126    /// `None` for a file stored before the size was recorded; never a substitute zero, so a client
127    /// can tell an unknown size from an empty file.
128    pub size_bytes: Option<i64>,
129    /// The file's position in the answer it belongs to. `None` for a file that is not part of an
130    /// answer yet, which is every file the upload endpoint returns.
131    pub order_number: Option<i32>,
132    /// Direct download URL; needs no bearer token.
133    pub url: String,
134}

143/// Body of `POST exercises/{id}/submit`. Plain JSON — no file parts, no archive: the previously

173/// The grading status of a task submission, as polled after `submit`.
174#[derive(Debug, Serialize, Deserialize)]
175#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
176pub enum ExerciseTaskSubmissionStatus {
177    NoGradingYet,
178    Grading {
179        grading_progress: GradingProgress,
180        /// Absent until grading has produced a value; a partial value while `grading_progress`
181        /// is still pending.
182        score_given: Option<f32>,
183        grading_started_at: Option<DateTime<Utc>>,
184        grading_completed_at: Option<DateTime<Utc>>,
185        /// Structured grading feedback, opaque like `ExerciseTask`'s spec fields: only the
186        /// exercise service that produced it interprets its shape.
187        feedback_json: Option<serde_json::Value>,
188        /// Human-readable feedback, for a client to display as-is.
189        feedback_text: Option<String>,
190        /// The user's progress on the whole exercise as of this poll, so a client can record
191        /// completion without a separate `courses/{id}/progress` round-trip. Absent for an exam
192        /// exercise, and from a host that predates the field.
193        #[serde(default)]
194        exercise_progress: Option<ExerciseProgress>,
195    },
196}

198/// How far along a task submission's grading is.
199#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
200#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
201pub enum GradingProgress {
202    /// The grading could not complete.
203    Failed,
204    /// No grading has occurred yet, e.g. the student has made no submission.
205    NotReady,
206    /// Final grade pending, needs human intervention. `score_given`, if present, is partial and may still change.
207    PendingManual,
208    /// Final grade pending, no human intervention needed. `score_given`, if present, is partial and may still change.
209    Pending,
210    /// Grading is complete. `score_given`, if present, is the final grade.
211    FullyGraded,
212}

226/// Response of `GET submissions/{id}/download`: the files the submission was made from, recovered
227/// from the host's own file records rather than from the service's answer.
228///
229/// The same shape regardless of where the submission was made. A native client's uploads are
230/// recorded as it names them; an answer made in the service's IFrame carries its files inside the

240/// The current user's progress across every exercise they can see in a course, returned
241/// by `courses/{id}/progress` in a single round-trip. Course-level totals are not sent
242/// separately; the client derives them by summing over `exercises` (e.g. total awarded =
243/// `sum(score_given)`, total available = `sum(score_maximum)`).
244#[derive(Debug, Serialize, Deserialize)]
245#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
246pub struct CourseProgress {
247    /// The course these progress entries belong to; echoes the path id.
248    pub course_id: Uuid,
249    pub exercises: Vec<ExerciseProgress>,
250}

252/// The current user's progress on a single exercise.
253///
254/// `standing` is the client's passed/failed signal. `completed` is the activity stage, which
255/// an exercise without peer or self review reaches on any graded submission, even one
256/// worth 0 points.
257#[derive(Debug, Serialize, Deserialize)]
258#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
259pub struct ExerciseProgress {
260    pub exercise_id: Uuid,
261    /// Points the user has been awarded, `0.0` when the user has no state for the exercise.
262    pub score_given: f32,
263    /// The maximum points obtainable from the exercise.
264    pub score_maximum: i32,
265    /// `true` once the exercise has reached the `Completed` activity stage.
266    pub completed: bool,
267    /// `true` once the user has started or submitted the exercise (any activity stage past
268    /// the initial one), regardless of whether it is completed.
269    pub attempted: bool,
270    /// Absent only from a host that predates the field.
271    #[serde(default)]
272    pub standing: Option<ExerciseStanding>,
273}

275/// Where the user stands on an exercise, decided by the host so every client agrees.
276#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
277#[cfg_attr(feature = "openapi", derive(utoipa::ToSchema))]
278pub enum ExerciseStanding {
279    /// No graded submission and no try used.
280    NotAttempted,
281    /// Below full points with tries left, or still being graded.
282    Attempted,
283    /// A grading awarded full points. An exercise worth 0 points passes on its first grading.
284    Passed,
285    /// Below full points (0 included) and the try limit is used up on every slide, so the
286    /// score is final.
287    OutOfTries,
288}
