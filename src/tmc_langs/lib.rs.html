111/// How courses.mooc.fi refused one course. The values are the CLI error kinds the same response
112/// fails a single-course command with.
113#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
114#[serde(rename_all = "kebab-case")]
115#[cfg_attr(feature = "ts-rs", derive(ts_rs::TS))]
116pub enum CourseRefusalKind {
117    /// A 403 `insufficient_scope`: the session lacks the `exercise-services` scope.
118    InsufficientScope,
119    /// Any other 403: the user may not view the course, e.g. a draft.
120    Forbidden,
121    /// A 404: the course does not exist, or not for this user.
122    NotFound,
123}
124
125impl CourseRefusalKind {
126    /// The refusal `err` is, if it concerns only the course asked for. A permanent auth failure,
127    /// an obsolete client and every other status concern the whole session.
128    fn of(err: &MoocAuthFailure) -> Option<Self> {
129        let MoocAuthFailure::Other(err) = err else {
130            return None;
131        };
132        let mooc::MoocClientError::HttpError {
133            status,
134            obsolete_client: false,
135            message_key,
136            ..
137        } = err.as_ref()
138        else {
139            return None;
140        };
141        match status.as_u16() {
142            403 if message_key.as_deref() == Some(mooc::INSUFFICIENT_SCOPE_MESSAGE_KEY) => {
143                Some(Self::InsufficientScope)
144            }
145            403 => Some(Self::Forbidden),
146            404 => Some(Self::NotFound),
147            _ => None,
148        }
149    }
150}

152/// Signs the given serializable value with the given secret using JWT.
153///
154/// # Example
155/// ```
156/// use jwt_simple::prelude::*;
157///
158/// #[derive(serde::Serialize, serde::Deserialize)]
159/// struct TestResult {
160///     passed: bool,
161/// }
162///
163/// // the secret must be at least 96 bits (12 bytes) long
164/// let secret = "example secret key".as_bytes();
165/// let token = tmc_langs::sign_with_jwt(TestResult { passed: true }, secret).unwrap();
166///
167/// // token has time-based claims (iat/exp), so verify by decoding it instead
168/// // of comparing to a fixed string
169/// let key = HS256Key::from_bytes(secret);
170/// let claims = key.verify_token::<TestResult>(&token, None).unwrap();
171/// assert!(claims.custom.passed);
172/// ```
173///
174/// # Errors
175/// Should never fail, but returns an error to be safe against changes in external libraries.
176pub fn sign_with_jwt<T: Serialize>(value: T, secret: &[u8]) -> Result<String, LangsError> {
177    let key = HS256Key::from_bytes(secret);
178    let claims = Claims::with_custom_claims(value, Duration::from_mins(15));
179    let token = key.authenticate(claims)?;
180    Ok(token)
181}

196/// Checks the server for updates to the TMC exercises in the given projects directory.
197pub fn check_tmc_exercise_updates(
198    client: &tmc::TestMyCodeClient,
199    projects_dir: &Path,
200) -> Result<ExerciseUpdates, LangsError> {
201    log::debug!("checking exercise updates in {}", projects_dir.display());
202
203    let mut updates = ExerciseUpdates::default();
204
205    let config = ProjectsConfig::load(projects_dir)?;
206    let local_exercises = config.get_all_tmc_exercises().collect::<Vec<_>>();
207
208    // request would fail with empty id list
209    if !local_exercises.is_empty() {
210        let exercise_ids = local_exercises.iter().map(|e| e.id).collect::<Vec<_>>();
211        let server_exercises = client
212            .get_exercises_details(&exercise_ids)?
213            .into_iter()
214            .map(|e| (e.id, e))
215            .collect::<HashMap<_, _>>();
216        for local_exercise in local_exercises {
217            let Some(server_exercise) = server_exercises.get(&local_exercise.id) else {
218                updates.removed.push(local_exercise.id);
219                continue;
220            };
221            if server_exercise.checksum != local_exercise.checksum {
222                updates.updated.push(local_exercise.id);
223            }
224        }
225    }
226    Ok(updates)
227}

229/// Checks the server for updates to the mooc exercises in the given projects directory.
230///
231/// A course courses.mooc.fi refuses lands in `unchecked_courses` and the others are still
232/// checked; any other failure fails the whole check.
233pub fn check_mooc_exercise_updates(
234    client: &mooc::MoocClient,
235    auth: &MoocAuth,
236    projects_dir: &Path,
237) -> Result<MoocExerciseUpdates, LangsError> {
238    log::debug!("checking exercise updates in {}", projects_dir.display());
239
240    let mut updates = MoocExerciseUpdates::default();
241
242    let config = ProjectsConfig::load(projects_dir)?;
243
244    for course_config in config.mooc_courses.values() {
245        if course_config.exercises.is_empty() {
246            continue;
247        }
248        match check_mooc_course_updates(client, auth, course_config) {
249            Ok(course_updates) => {
250                updates.updated.extend(course_updates.updated);
251                updates.removed.extend(course_updates.removed);
252            }
253            Err(err) => {
254                let Some(kind) = CourseRefusalKind::of(&err) else {
255                    return Err(err.into());
256                };
257                log::warn!(
258                    "skipping the update check of course {}: {err}",
259                    course_config.course_id
260                );
261                updates.unchecked_courses.push(UncheckedMoocCourse {
262                    course_id: course_config.course_id,
263                    kind,
264                });
265            }
266        }
267    }
268    Ok(updates)
269}

271/// [`check_mooc_exercise_updates`] for one course; `unchecked_courses` stays empty.
272fn check_mooc_course_updates(
273    client: &mooc::MoocClient,
274    auth: &MoocAuth,
275    course_config: &MoocCourseConfig,
276) -> Result<MoocExerciseUpdates, MoocAuthFailure> {
277    let mut updates = MoocExerciseUpdates::default();
278    // Keyed by exercise id: the stored `task_id` is the editor task's id, not an exercise key.
279    let server_exercises: HashMap<Uuid, mooc::TmcExerciseSlide> =
280        stored_course_exercises(client, auth, course_config)?
281            .into_iter()
282            .map(|slide| (slide.exercise_id, slide))
283            .collect();
284    let mut missing = Vec::new();
285    for (exercise_id, local_exercise) in &course_config.exercises {
286        let Some(server_exercise) = server_exercises.get(exercise_id) else {
287            missing.push(*exercise_id);
288            continue;
289        };
290        if server_exercise
291            .editor_checksum()
292            .map(|cs| cs != local_exercise.checksum)
293            .unwrap_or_default()
294        {
295            updates.updated.push(*exercise_id);
296        }
297    }
298    if !missing.is_empty() && !course_config.is_exam {
299        let closed = closed_chapter_exercise_ids(client, auth, course_config.course_id)?;
300        updates
301            .removed
302            .extend(missing.into_iter().filter(|id| !closed.contains(id)));
303    }
304    Ok(updates)
305}

335/// Downloads the user's old submission from the server and restores it over a fresh template at
336/// the path, replacing the directory's contents.
337/// If a submission_url is given, the current state of the exercise is submitted to that URL before the download.
338pub fn download_old_submission(
339    client: &tmc::TestMyCodeClient,
340    exercise_id: u32,
341    output_path: &Path,
342    submission_id: u32,
343    save_old_state: bool,
344) -> Result<(), LangsError> {
345    log::debug!("downloading old submission {submission_id} for {exercise_id}");
346
347    if save_old_state {
348        // submit old exercise
349        let tmc_project_yml = TmcProjectYml::load_or_default(output_path)?;
350        client.submit(
351            exercise_id,
352            output_path,
353            tmc_project_yml.get_submission_size_limit_mb(),
354            None,
355        )?;
356        log::debug!("finished submission");
357    }
358
359    // Fetch both archives before touching the exercise: a failed download must not wipe the
360    // student's work.
361    let mut submission = vec![];
362    client.download_old_submission(submission_id, &mut submission)?;
363    let mut template = vec![];
364    client.download_exercise(exercise_id, &mut template)?;
365    log::debug!("downloaded old submission");
366
367    let staging = staging_dir_beside(output_path)?;
368    extract_project(
369        Cursor::new(template),
370        staging.path(),
371        Compression::Zip,
372        false,
373        false,
374    )?;
375    extract_student_files(Cursor::new(submission), Compression::Zip, staging.path())?;
376    replace_dir_contents(output_path, staging.path())?;
377    log::debug!("restored old submission");
378    Ok(())
379}

437/// Downloads the given exercises, by either downloading the exercise template, updating the exercise or downloading an old submission.
438/// Requires authentication.
439/// If the exercise doesn't exist on disk yet...
440///   if there are previous submissions and download_template is not set, the latest submission is downloaded.
441///   otherwise, the exercise template is downloaded.
442/// If the exercise exists on disk, it is updated using the course template.
443pub fn download_or_update_course_exercises(
444    client: &tmc::TestMyCodeClient,
445    projects_dir: &Path,
446    exercises: &[u32],
447    download_template: bool,
448) -> Result<TmcDownloadResult, LangsError> {
449    log::debug!(
450        "downloading or updating course exercises in {}",
451        projects_dir.display()
452    );
453
454    client.require_authentication().map_err(Box::new)?;
455
456    let exercises_details = client.get_exercises_details(exercises)?;
457    let mut config_update_lock = ProjectsConfig::lock_course_config_updates(projects_dir)?;
458    let _config_update_guard = config_update_lock.lock()?;
459    let projects_config = ProjectsConfig::load(projects_dir)?;
460
461    // separate exercises into downloads and skipped
462    let mut to_be_downloaded = vec![];
463    let mut to_be_skipped = vec![];
464
465    log::debug!("checking the checksum of each exercise on the server");
466    for exercise_detail in exercises_details {
467        let target = ProjectsConfig::resolve_tmc_exercise_path(
468            projects_dir,
469            &exercise_detail.course_name,
470            &exercise_detail.exercise_name,
471        );
472
473        // check if the exercise is already on disk
474        if let Some(exercise) = projects_config
475            .get_tmc_exercise(&exercise_detail.course_name, &exercise_detail.exercise_name)
476        {
477            // exercise is on disk, check if the checksum is identical
478            if exercise_detail.checksum == exercise.checksum {
479                // skip this exercise
480                log::info!(
481                    "Skipping exercise {} ({} in {}) due to identical checksum",
482                    exercise_detail.id,
483                    exercise_detail.course_name,
484                    exercise_detail.exercise_name
485                );
486                to_be_skipped.push(TmcExerciseDownload {
487                    id: exercise_detail.id,
488                    course_slug: exercise_detail.course_name,
489                    exercise_slug: exercise_detail.exercise_name,
490                    path: target,
491                });
492                continue;
493            }
494        } else {
495            // not on disk, if flag isn't set check if there are any previous submissions and take the latest one if so
496            if !download_template {
497                if let Some(latest_submission) = client
498                    .get_exercise_submissions_for_current_user(exercise_detail.id)?
499                    .into_iter()
500                    .max_by_key(|s| s.created_at)
501                {
502                    // previous submission found, check if exercise submission results hidden (part of exam)
503                    if !exercise_detail.hide_submission_results {
504                        to_be_downloaded.push(DownloadTarget {
505                            target: TmcExerciseDownload {
506                                id: exercise_detail.id,
507                                course_slug: exercise_detail.course_name,
508                                exercise_slug: exercise_detail.exercise_name,
509                                path: target,
510                            },
511                            checksum: exercise_detail.checksum,
512                            kind: DownloadTargetKind::Submission {
513                                submission_id: latest_submission.id,
514                            },
515                        });
516                        continue;
517                    }
518                }
519            }
520        }
521
522        // not skipped, either not on disk or no previous submissions or submission result hidden, downloading template
523        to_be_downloaded.push(DownloadTarget {
524            target: TmcExerciseDownload {
525                id: exercise_detail.id,
526                course_slug: exercise_detail.course_name.clone(),
527                exercise_slug: exercise_detail.exercise_name.clone(),
528                path: target,
529            },
530            checksum: exercise_detail.checksum,
531            kind: DownloadTargetKind::Template,
532        });
533    }
534
535    let course_ids = Arc::new(lookup_unrecorded_tmc_course_ids(
536        client,
537        &projects_config,
538        &to_be_downloaded,
539    ));
540
541    let exercises_len = to_be_downloaded.len();
542    progress_reporter::start_stage::<()>(
543        u32::try_from(exercises_len).expect("should never happen") * 2 + 1, // each download progresses at 2 points, plus the final finishing step
544        format!("Downloading {exercises_len} exercises"),
545        None,
546    );
547
548    log::debug!("downloading exercises");
549    // download and divide the results into successful and failed downloads
550    let thread_count = to_be_downloaded.len().min(4); // max 4 threads
551    let mut handles = vec![];
552    let exercises = Arc::new(Mutex::new(to_be_downloaded));
553    let projects_config = Arc::new(Mutex::new(projects_config));
554    for _thread_id in 0..thread_count {
555        let client = client.clone();
556        let exercises = Arc::clone(&exercises);
557        let projects_config = Arc::clone(&projects_config);
558        let course_ids = Arc::clone(&course_ids);
559        let projects_dir = projects_dir.to_path_buf();
560
561        // each thread returns either a list of successful downloads, or a tuple of successful downloads and errors
562        type ThreadErr = (Vec<DownloadTarget>, Vec<(DownloadTarget, LangsError)>);
563        let handle = std::thread::spawn(move || -> Result<Vec<DownloadTarget>, ThreadErr> {
564            let mut downloaded = vec![];
565            let mut failed = vec![];
566
567            // repeat until out of exercises
568            loop {
569                let mut exercises = exercises.lock().expect("the threads should never panic");
570                let download_target = if let Some(download_target) = exercises.pop() {
571                    download_target
572                } else {
573                    // no exercises left, break loop and exit thread
574                    break;
575                };
576                drop(exercises);
577                // dropped mutex
578
579                let exercise_download_result = || -> Result<(), LangsError> {
580                    progress_reporter::progress_stage::<tmc::ClientUpdateData>(
581                        format!(
582                            "Downloading exercise {} to '{}'",
583                            download_target.target.id,
584                            download_target.target.path.display(),
585                        ),
586                        Some(tmc::ClientUpdateData::ExerciseDownload {
587                            id: download_target.target.id,
588                            path: download_target.target.path.clone(),
589                        }),
590                    );
591
592                    // execute download based on type
593                    match &download_target.kind {
594                        DownloadTargetKind::Template => {
595                            let mut buf = vec![];
596                            client.download_exercise(download_target.target.id, &mut buf)?;
597                            extract_update(buf, &download_target.target.path, Compression::Zip)?;
598                        }
599                        DownloadTargetKind::Submission { submission_id } => {
600                            let mut buf = vec![];
601                            client.download_exercise(download_target.target.id, &mut buf)?;
602                            extract_project(
603                                Cursor::new(buf),
604                                &download_target.target.path,
605                                Compression::Zip,
606                                false,
607                                false,
608                            )?;
609
610                            let plugin = PluginType::from_exercise(&download_target.target.path)?;
611                            let config = plugin.get_exercise_packaging_configuration(
612                                &download_target.target.path,
613                            )?;
614                            for student_file in config.student_file_paths {
615                                let student_file = download_target.target.path.join(student_file);
616                                if student_file.is_file() {
617                                    file_util::remove_file(&student_file)?;
618                                } else {
619                                    file_util::remove_dir_all(&student_file)?;
620                                }
621                            }
622
623                            let mut buf = vec![];
624                            client.download_old_submission(*submission_id, &mut buf)?;
625                            if let Err(err) = plugin.extract_student_files(
626                                Cursor::new(buf),
627                                Compression::Zip,
628                                &download_target.target.path,
629                            ) {
630                                log::error!(
631                                    "Something went wrong when downloading old submission: {err}"
632                                );
633                            }
634                        }
635                    }
636                    // download successful, save to course config
637                    let mut projects_config =
638                        projects_config.lock().map_err(|_| LangsError::MutexError)?; // lock mutex
639                    let course_config = projects_config
640                        .get_or_init_tmc_course_config(download_target.target.course_slug.clone());
641                    if let Some(course_id) = course_ids.get(&download_target.target.course_slug) {
642                        course_config.course_id = Some(*course_id);
643                    }
644                    course_config.add_exercise(
645                        download_target.target.exercise_slug.clone(),
646                        download_target.target.id,
647                        download_target.checksum.clone(),
648                    );
649                    course_config.save_to_projects_dir(&projects_dir)?;
650                    drop(projects_config); // drop mutex
651
652                    progress_reporter::progress_stage::<tmc::ClientUpdateData>(
653                        format!(
654                            "Downloaded exercise {} to '{}'",
655                            download_target.target.id,
656                            download_target.target.path.display(),
657                        ),
658                        Some(tmc::ClientUpdateData::ExerciseDownload {
659                            id: download_target.target.id,
660                            path: download_target.target.path.clone(),
661                        }),
662                    );
663
664                    Ok(())
665                }();
666
667                match exercise_download_result {
668                    Ok(_) => {
669                        downloaded.push(download_target);
670                    }
671                    Err(err) => {
672                        failed.push((download_target, err));
673                    }
674                }
675            }
676            if failed.is_empty() {
677                Ok(downloaded)
678            } else {
679                Err((downloaded, failed))
680            }
681        });
682        handles.push(handle);
683    }
684
685    // gather results from each thread
686    let mut successful = vec![];
687    let mut failed = vec![];
688    for handle in handles {
689        match handle.join().expect("the threads should never panic") {
690            Ok(s) => successful.extend(s),
691            Err((s, f)) => {
692                successful.extend(s);
693                failed.extend(f);
694            }
695        }
696    }
697
698    // report
699    let finish_message = if failed.is_empty() {
700        if successful.is_empty() && exercises_len == 0 {
701            "Exercises are already up-to-date!".to_string()
702        } else {
703            format!(
704                "Successfully downloaded {} out of {} exercises.",
705                successful.len(),
706                exercises_len
707            )
708        }
709    } else {
710        format!(
711            "Downloaded {} out of {} exercises ({} failed)",
712            successful.len(),
713            exercises_len,
714            failed.len(),
715        )
716    };
717    progress_reporter::finish_stage::<tmc::ClientUpdateData>(finish_message, None);
718
719    // return information about the downloads
720    let downloaded = successful.into_iter().map(|t| t.target).collect();
721    if !failed.is_empty() {
722        let failed = failed
723            .into_iter()
724            .map(|(target, err)| {
725                let failure = TmcDownloadFailure::of(&err);
726                (target.target, error_chain(&err), failure)
727            })
728            .collect();
729        return Ok(TmcDownloadResult::Failure {
730            downloaded,
731            skipped: to_be_skipped,
732            failed,
733        });
734    }
735
736    Ok(TmcDownloadResult::Success {
737        downloaded,
738        skipped: to_be_skipped,
739    })
740}

759/// tmc-server hosts that may be handed a courses.mooc.fi access token.
760///
761/// The tmc root URL comes from the environment and so does not prove the host is
762/// tmc-server; this list stops a redirected URL from harvesting the token.
763const TMC_HOSTS_TRUSTED_WITH_MOOC_TOKEN: &[&str] = &["tmc.mooc.fi"];
764
765/// See [`TMC_HOSTS_TRUSTED_WITH_MOOC_TOKEN`]. Loopback needs the same opt-in as
766/// the mooc client, so production never trusts a local host implicitly.
767fn tmc_host_may_receive_mooc_token(root_url: &Url) -> bool {
768    let Some(host) = root_url.host_str() else {
769        return false;
770    };
771    if TMC_HOSTS_TRUSTED_WITH_MOOC_TOKEN.contains(&host) {
772        // A bearer for a second backend must not cross the network in plaintext.
773        return root_url.scheme() == "https";
774    }
775    mooc::trust_localhost() && matches!(host, "localhost" | "127.0.0.1" | "[::1]")
776}

778/// Which credential authenticates a [`tmc::TestMyCodeClient`].
779///
780/// New tmc-server logins no longer exist, so there are only two sources: a token
781/// stored by an older version, and the courses.mooc.fi access token that
782/// tmc-server accepts by introspecting it.
783#[derive(Debug)]
784pub enum TestMyCodeAuth {
785    /// A tmc-server token from the legacy `credentials.json`.
786    StoredTmc(Credentials),
787    /// The courses.mooc.fi access token.
788    Mooc(tmc::Token),
789    /// No usable credential; only unauthenticated endpoints will work.
790    Unauthenticated,
791}
792
793impl TestMyCodeAuth {
794    /// The token the client was given, for reporting login status.
795    pub fn token(&self) -> Option<tmc::Token> {
796        match self {
797            Self::StoredTmc(credentials) => Some(credentials.token()),
798            Self::Mooc(token) => Some(token.clone()),
799            Self::Unauthenticated => None,
800        }
801    }
802
803    /// Takes the stored tmc credentials out, leaving the value unauthenticated.
804    ///
805    /// For the 401 handling, which may delete a rejected *tmc* token but must never
806    /// touch the mooc credentials: tmc-server rejecting a mooc token says nothing
807    /// about its validity at courses.mooc.fi.
808    pub fn take_stored_tmc(&mut self) -> Option<Credentials> {
809        match std::mem::replace(self, Self::Unauthenticated) {
810            Self::StoredTmc(credentials) => Some(credentials),
811            other => {
812                *self = other;
813                None
814            }
815        }
816    }
817}

819/// Initializes a TestMyCodeClient with whichever credential authenticates it.
820///
821/// A stored tmc `credentials.json` wins while it exists, so password-era logins
822/// keep working until tmc-server rejects the token; the 401 handling then deletes
823/// the file and the next invocation falls through to the mooc token. The
824/// credential never changes under a session that still works.
825///
826/// `mooc_root_url` and `mooc_client_id` are used to refresh the mooc token, as in
827/// [`init_mooc_client_with_credentials`].
828pub fn init_testmycode_client_with_credentials(
829    root_url: Url,
830    client_name: &str,
831    client_version: &str,
832    mooc_root_url: &Url,
833    mooc_client_id: &str,
834) -> Result<(tmc::TestMyCodeClient, TestMyCodeAuth), LangsError> {
835    let mut client = tmc::TestMyCodeClient::new(
836        root_url.clone(),
837        client_name.to_string(),
838        client_version.to_string(),
839    )?;
840
841    if let Some(credentials) = Credentials::load(client_name)? {
842        client.set_token(credentials.token(), tmc::TokenSource::Tmc);
843        return Ok((client, TestMyCodeAuth::StoredTmc(credentials)));
844    }
845
846    if !tmc_host_may_receive_mooc_token(&root_url) {
847        log::warn!(
848            "not authenticating with {root_url} using the courses.mooc.fi access token: \
849             the host is not trusted with it"
850        );
851        return Ok((client, TestMyCodeAuth::Unauthenticated));
852    }
853
854    let Some(mooc_credentials) =
855        MoocCredentials::load_valid(client_name, mooc_root_url, mooc_client_id)?
856    else {
857        return Ok((client, TestMyCodeAuth::Unauthenticated));
858    };
859    let token = mooc_credentials.token();
860    client.set_token(token.clone(), tmc::TokenSource::Mooc);
861    Ok((client, TestMyCodeAuth::Mooc(token)))
862}

865///
866/// The stored token is refreshed first if expired (see
867/// [`MoocCredentials::load_valid`]). `client_id` is the OAuth2 client id the
868/// refresh grant is made with.
869pub fn init_mooc_client_with_credentials(
870    root_url: Url,
871    client_name: &str,
872    client_id: &str,
873) -> Result<(mooc::MoocClient, Option<MoocCredentials>), LangsError> {

880    }
881
882    Ok((client, credentials))
883}

888    client: &tmc::TestMyCodeClient,
889    projects_config: &ProjectsConfig,
890    targets: &[DownloadTarget],
891) -> HashMap<String, u32> {

900            continue;

907            }
908        };

910    }
911    looked_up
912        .into_iter()
913        .filter_map(|(course_slug, course_id)| Some((course_slug, course_id?)))
914        .collect()

1032/// Downloads and extracts each planned exercise, a few at a time; results are in plan order.
1033fn download_mooc_archives_in_parallel(
1034    client: &MoocClient,
1035    auth: &MoocAuth,
1036    planned: &[PlannedMoocDownload],
1037) -> Vec<Result<(), DownloadArchiveError>> {
1038    let next = AtomicUsize::new(0);
1039    let mut results: Vec<(usize, Result<(), DownloadArchiveError>)> = std::thread::scope(|scope| {
1040        let workers: Vec<_> = (0..MOOC_DOWNLOAD_THREADS.min(planned.len()))
1041            .map(|_| {
1042                scope.spawn(|| {
1043                    let mut results = vec![];
1044                    loop {
1045                        let index = next.fetch_add(1, AtomicOrdering::Relaxed);
1046                        let Some(plan) = planned.get(index) else {
1047                            break;
1048                        };
1049                        progress_reporter::progress_stage::<mooc::MoocClientUpdateData>(
1050                            format!(
1051                                "Downloading exercise {} to '{}'",
1052                                plan.exercise_id,
1053                                plan.target.display()
1054                            ),
1055                            Some(mooc::MoocClientUpdateData::ExerciseDownload {
1056                                id: plan.exercise_id,
1057                                path: plan.target.clone(),
1058                            }),
1059                        );
1060                        let result = download_and_extract_mooc_archive(
1061                            client,
1062                            auth,
1063                            &plan.download_url,
1064                            &plan.target,
1065                        );
1066                        results.push((index, result));
1067                    }
1068                    results
1069                })
1070            })
1071            .collect();
1072        workers
1073            .into_iter()
1074            .flat_map(|worker| worker.join().expect("download workers do not panic"))
1075            .collect()
1076    });
1077    results.sort_by_key(|(index, _)| *index);
1078    results.into_iter().map(|(_, result)| result).collect()
1079}

1109/// Updates the mooc exercises in the local projects directory. The exercises of a course
1110/// courses.mooc.fi refuses, and those whose archive fails to download or extract, land in
1111/// `failed` and the rest are still updated.
1112pub fn update_mooc_exercises(
1113    client: &MoocClient,
1114    auth: &MoocAuth,
1115    projects_dir: &Path,
1116) -> Result<DownloadOrUpdateMoocCourseExercisesResult, LangsError> {
1117    let mut config_update_lock = ProjectsConfig::lock_course_config_updates(projects_dir)?;
1118    let _config_update_guard = config_update_lock.lock()?;
1119    let mut projects_config = ProjectsConfig::load(projects_dir)?;
1120    // Owned snapshot keyed by exercise id (not the editor task id), so
1121    // `projects_config` can be mutated below.
1122    struct LocalMoocExerciseInfo {
1123        course_id: Uuid,
1124        course_directory: String,
1125        exercise_directory: String,
1126        exercise_name: String,
1127        checksum: String,
1128    }
1129    let locals: HashMap<Uuid, LocalMoocExerciseInfo> = projects_config
1130        .mooc_courses
1131        .iter()
1132        .flat_map(|(course_id, cc)| {
1133            cc.exercises.iter().map(move |(id, e)| {
1134                (
1135                    *id,
1136                    LocalMoocExerciseInfo {
1137                        course_id: *course_id,
1138                        course_directory: cc.directory.clone(),
1139                        exercise_directory: e.directory.clone(),
1140                        exercise_name: e.name.clone(),
1141                        checksum: e.checksum.clone(),
1142                    },
1143                )
1144            })
1145        })
1146        .collect();
1147
1148    let mut downloaded = Vec::new();
1149    let mut failed = Vec::new();
1150    if !locals.is_empty() {
1151        // One exercise list request per course, not one per exercise.
1152        let mut server_exercises: Vec<mooc::TmcExerciseSlide> = Vec::new();
1153        for course_config in projects_config.mooc_courses.values() {
1154            if course_config.exercises.is_empty() {
1155                continue;
1156            }
1157            match stored_course_exercises(client, auth, course_config) {
1158                Ok(slides) => server_exercises.extend(slides),
1159                // An exam stops being served once it ends; that must not block the other courses.
1160                Err(err) => {
1161                    let Some(kind) = CourseRefusalKind::of(&err) else {
1162                        return Err(err.into());
1163                    };
1164                    log::warn!(
1165                        "skipping the update of course {}: {err}",
1166                        course_config.course_id
1167                    );
1168                    let failure = match kind {
1169                        CourseRefusalKind::NotFound => MoocDownloadFailure::NotFound,
1170                        CourseRefusalKind::Forbidden | CourseRefusalKind::InsufficientScope => {
1171                            MoocDownloadFailure::Other
1172                        }
1173                    };
1174                    let reason = error_chain(&LangsError::from(err));
1175                    for (exercise_id, exercise) in &course_config.exercises {
1176                        failed.push((
1177                            MoocExerciseDownload {
1178                                exercise_id: *exercise_id,
1179                                path: ProjectsConfig::get_mooc_exercise_download_target(
1180                                    projects_dir,
1181                                    &course_config.directory,
1182                                    &exercise.directory,
1183                                ),
1184                            },
1185                            reason.clone(),
1186                            failure,
1187                        ));
1188                    }
1189                }
1190            }
1191        }
1192        for slide in server_exercises {
1193            let Some(local) = locals.get(&slide.exercise_id) else {
1194                // Not tracked locally (e.g. deleted).
1195                continue;
1196            };
1197            // Browser exercises have no editor checksum; skip them.
1198            let Some(new_checksum) = slide.editor_checksum() else {
1199                continue;
1200            };
1201            if new_checksum == local.checksum {
1202                continue;
1203            }
1204            let target = ProjectsConfig::get_mooc_exercise_download_target(
1205                projects_dir,
1206                &local.course_directory,
1207                &local.exercise_directory,
1208            );
1209            // The slide already carries the stub URL; no extra fetch.
1210            let Some(download_url) = slide.editor_stub_download_url() else {
1211                log::warn!(
1212                    "Skipping exercise {}: no downloadable editor task in public spec",
1213                    slide.exercise_id
1214                );
1215                continue;
1216            };
1217            match download_and_extract_mooc_archive(client, auth, download_url, &target) {
1218                Ok(()) => {}
1219                Err(DownloadArchiveError::Auth(auth_err)) if auth_err.is_permanent() => {
1220                    return Err(auth_err.into());
1221                }
1222                Err(err) => {
1223                    let err: LangsError = err.into();
1224                    let failure = mooc_download_failure(&err);
1225                    failed.push((
1226                        MoocExerciseDownload {
1227                            exercise_id: slide.exercise_id,
1228                            path: target,
1229                        },
1230                        error_chain(&err),
1231                        failure,
1232                    ));
1233                    continue;
1234                }
1235            }
1236
1237            // Persist the new checksum, or every later check re-reports the update.
1238            // Prefer the editor task id (the checksum's owner) over the stored one.
1239            let task_id = slide
1240                .editor_task_id()
1241                .or_else(|| {
1242                    projects_config
1243                        .mooc_courses
1244                        .get(&local.course_id)
1245                        .and_then(|cc| cc.exercises.get(&slide.exercise_id))
1246                        .map(|e| e.task_id)
1247                })
1248                .unwrap_or_else(Uuid::nil);
1249            if let Some(course_config) = projects_config.mooc_courses.get_mut(&local.course_id) {
1250                course_config.add_exercise(
1251                    slide.exercise_id,
1252                    local.exercise_name.clone(),
1253                    task_id,
1254                    new_checksum.to_string(),
1255                    local.exercise_directory.clone(),
1256                );
1257                course_config.save_to_projects_dir(projects_dir)?;
1258            }
1259
1260            downloaded.push(MoocExerciseDownload {
1261                exercise_id: slide.exercise_id,
1262                path: target,
1263            });
1264        }
1265    }
1266
1267    Ok(DownloadOrUpdateMoocCourseExercisesResult {
1268        downloaded,
1269        skipped: vec![],
1270        failed: (!failed.is_empty()).then_some(failed),
1271        closed_chapters: Vec::new(),
1272    })
1273}

1275impl TmcDownloadFailure {
1276    fn of(err: &LangsError) -> Self {
1277        match err {
1278            LangsError::TestMyCodeClient(err) => match err.as_ref() {
1279                tmc::TestMyCodeClientError::HttpError { status, .. }
1280                    if matches!(status.as_u16(), 403 | 404) =>
1281                {
1282                    Self::NotFound
1283                }
1284                tmc::TestMyCodeClientError::HttpError { status, .. }
1285                    if status.is_server_error() =>
1286                {
1287                    Self::ServerError
1288                }
1289                tmc::TestMyCodeClientError::ConnectionError(..) => Self::ConnectionError,
1290                _ => Self::Other,
1291            },
1292            err if is_extract_error(err) => Self::ExtractFailed,
1293            _ => Self::Other,
1294        }
1295    }

1298/// The [`MoocDownloadFailure`] of an error downloading or extracting a mooc exercise archive.
1299fn mooc_download_failure(err: &LangsError) -> MoocDownloadFailure {
1300    match err {
1301        LangsError::MoocClient(err) => match err.as_ref() {
1302            mooc::MoocClientError::FileNotFound { .. } => MoocDownloadFailure::FileMissing,
1303            mooc::MoocClientError::NoDownloadableExerciseTask { .. } => {
1304                MoocDownloadFailure::BrowserOnly
1305            }
1306            mooc::MoocClientError::SeveralEditorTasks { .. } => {
1307                MoocDownloadFailure::SeveralEditorTasks
1308            }
1309            mooc::MoocClientError::ConnectionError(..) => MoocDownloadFailure::ConnectionError,
1310            mooc::MoocClientError::HttpError { status, .. } if status.is_server_error() => {
1311                MoocDownloadFailure::ServerError
1312            }
1313            _ => MoocDownloadFailure::Other,
1314        },
1315        err if is_extract_error(err) => MoocDownloadFailure::ExtractFailed,
1316        _ => MoocDownloadFailure::Other,
1317    }
1318}

1328/// Flattens an error's `source()` chain into strings, outermost first, for the
1329/// CLI's `failed` list entries.
1330fn error_chain(err: &dyn std::error::Error) -> Vec<String> {
1331    let mut chain = vec![err.to_string()];
1332    let mut error = err;
1333    while let Some(source) = error.source() {
1334        chain.push(source.to_string());
1335        error = source;
1336    }
1337    chain
1338}

1377/// Restores a past mooc submission at `output_path`: the submission's student
1378/// files overlaid on a fresh exercise stub, fully replacing the directory's contents.
1379///
1380/// Mirrors [`download_old_submission`] minus the server reset (mooc has none). If
1381/// `save_old_state` is set, the current state is first stored as an ungraded backup
1382/// snapshot so nothing the student wrote is lost.
1383///
1384/// A submission with no files yields [`MoocOldSubmissionRestore::NothingToDownload`];
1385/// one with several yields [`LangsError::NotATmcAnswer`]. Both are decided before
1386/// the local exercise or the server is touched.
1387pub fn download_mooc_old_submission(
1388    client: &MoocClient,
1389    auth: &MoocAuth,
1390    exercise_id: Uuid,
1391    output_path: &Path,
1392    submission_id: Uuid,
1393    save_old_state: bool,
1394) -> Result<MoocOldSubmissionRestore, LangsError> {
1395    log::debug!(
1396        "downloading old mooc submission {submission_id} for exercise {exercise_id} to {}",
1397        output_path.display()
1398    );
1399
1400    // A tmc answer is exactly one `TarZstd` archive; guessing among several files
1401    // would silently restore the wrong project.
1402    let mut files = auth.call(client, |c| c.download_submission_files(submission_id))?;
1403    let archive_url = match files.len() {
1404        0 => {
1405            log::debug!("submission {submission_id} has no downloadable files");
1406            return Ok(MoocOldSubmissionRestore::NothingToDownload);
1407        }
1408        1 => files.remove(0).url,
1409        file_count => {
1410            return Err(LangsError::NotATmcAnswer {
1411                submission_id,
1412                file_count,
1413            });
1414        }
1415    };
1416
1417    if save_old_state {
1418        let temp = file_util::named_temp_file()?;
1419        compress_project_to(output_path, temp.path(), Compression::TarZstd, false, false)?;
1420        auth.call(client, |c| {
1421            c.store_snapshot(exercise_id, temp.path(), mooc::api::SnapshotPurpose::Backup)
1422        })?;
1423        log::debug!("backed up current state before downloading old submission");
1424    }
1425
1426    let staging = staging_dir_beside(output_path)?;
1427    let base = staging.path();
1428    let stub = auth.call(client, |c| c.download_exercise(exercise_id))?;
1429    extract_project(Cursor::new(stub), base, Compression::TarZstd, false, false)?;
1430    log::debug!("extracted fresh stub to temp base");
1431
1432    let url = Url::parse(&archive_url)
1433        .map_err(|err| Box::new(mooc::MoocClientError::UrlParse(archive_url.clone(), err)))?;
1434    let archive = auth.call(client, |c| c.download(url.clone()))?;
1435    extract_student_files(Cursor::new(archive), Compression::TarZstd, base)?;
1436    log::debug!("overlaid old submission student files");
1437
1438    replace_dir_contents(output_path, base)?;
1439    log::debug!("moved restored exercise into place");
1440    Ok(MoocOldSubmissionRestore::Restored)
1441}

1443/// Downloads or updates the given mooc exercises in the local projects directory.
1444///
1445/// With `course_id` (which the extension always knows) only that course's slides
1446/// are fetched; otherwise the enrolled courses are scanned (O(courses x
1447/// exercises)), since the langs API has no exercise -> course lookup. The course
1448/// name determines the on-disk directory, and the course id doubles as the
1449/// projects config instance key (the API does not expose course instance ids).
1450///
1451/// Per exercise: not found, in a chapter that has not opened yet, or no editor task (e.g.
1452/// browser-only) -> failed;
1453/// stored checksum matches -> skipped; otherwise the stub archive is downloaded,
1454/// extracted and the course config updated. Results are keyed by the requested
1455/// `exercise_id`.
1456pub fn download_or_update_mooc_course_exercises(
1457    client: &MoocClient,
1458    auth: &MoocAuth,
1459    projects_dir: &Path,
1460    exercise_ids: &[Uuid],
1461    source: MoocExerciseSource,
1462) -> Result<DownloadOrUpdateMoocCourseExercisesResult, LangsError> {
1463    log::debug!(
1464        "downloading or updating {} mooc exercises in {}",
1465        exercise_ids.len(),
1466        projects_dir.display()
1467    );
1468
1469    let mut config_update_lock = ProjectsConfig::lock_course_config_updates(projects_dir)?;
1470    let _config_update_guard = config_update_lock.lock()?;
1471    let mut projects_config = ProjectsConfig::load(projects_dir)?;
1472
1473    let requested: HashSet<Uuid> = exercise_ids.iter().copied().collect();
1474    let mut resolved: HashMap<Uuid, (StoredCourse, mooc::TmcExerciseSlide)> = HashMap::new();
1475    // Requested exercises of a chapter that has not opened yet, with why they cannot be downloaded.
1476    let mut closed: HashMap<Uuid, (String, MoocExerciseClosedChapter)> = HashMap::new();
1477    if !requested.is_empty() {
1478        match source {
1479            MoocExerciseSource::Course(course_id) => {
1480                let course = StoredCourse::of_course(&auth.call(client, |c| c.course(course_id))?);
1481                for slide in auth.call(client, |c| c.course_exercises(course_id))? {
1482                    if requested.contains(&slide.exercise_id) {
1483                        resolved.insert(slide.exercise_id, (course.clone(), slide));
1484                    }
1485                }
1486                if resolved.len() < requested.len() {
1487                    let progress = auth.call(client, |c| c.course_progress(course_id))?;
1488                    for chapter in progress.closed_chapters {
1489                        for exercise in &chapter.exercises {
1490                            if requested.contains(&exercise.exercise_id) {
1491                                let reason = format!(
1492                                    "Exercise {} is in chapter {} \"{}\", which opens at {}",
1493                                    exercise.exercise_id,
1494                                    chapter.chapter_number,
1495                                    chapter.name,
1496                                    chapter.opens_at
1497                                );
1498                                let detail = MoocExerciseClosedChapter {
1499                                    exercise_id: exercise.exercise_id,
1500                                    chapter_name: chapter.name.clone(),
1501                                    chapter_number: chapter.chapter_number,
1502                                    opens_at: chapter.opens_at,
1503                                };
1504                                closed.insert(exercise.exercise_id, (reason, detail));
1505                            }
1506                        }
1507                    }
1508                }
1509            }
1510            MoocExerciseSource::Exam(exam_id) => {
1511                let exam = auth
1512                    .call(client, |c| c.exams())?
1513                    .into_iter()
1514                    .find(|exam| exam.id == exam_id)
1515                    .ok_or(LangsError::MoocExamNotAnswerable(exam_id))?;
1516                let exam = StoredCourse {
1517                    id: exam.id,
1518                    name: exam.name,
1519                    is_exam: true,
1520                };
1521                for slide in auth.call(client, |c| c.exam_exercises(exam_id))? {
1522                    if requested.contains(&slide.exercise_id) {
1523                        resolved.insert(slide.exercise_id, (exam.clone(), slide));
1524                    }
1525                }
1526            }
1527            MoocExerciseSource::EnrolledCourses => {
1528                'courses: for course in auth.call(client, |c| c.courses())? {
1529                    let stored_course = StoredCourse::of_course(&course);
1530                    for slide in auth.call(client, |c| c.course_exercises(course.id))? {
1531                        if requested.contains(&slide.exercise_id)
1532                            && !resolved.contains_key(&slide.exercise_id)
1533                        {
1534                            resolved.insert(slide.exercise_id, (stored_course.clone(), slide));
1535                            if resolved.len() == requested.len() {
1536                                break 'courses;
1537                            }
1538                        }
1539                    }
1540                }
1541            }
1542        }
1543    }
1544
1545    let mut downloaded = Vec::new();
1546    let mut skipped = Vec::new();
1547    let mut failed: Vec<(MoocExerciseDownload, Vec<String>, MoocDownloadFailure)> = Vec::new();
1548    let mut closed_chapters = Vec::new();
1549
1550    let total_steps = u32::try_from(exercise_ids.len())
1551        .unwrap_or(u32::MAX)
1552        .saturating_add(1);
1553    progress_reporter::start_stage::<mooc::MoocClientUpdateData>(
1554        total_steps,
1555        format!("Downloading {} mooc exercises", exercise_ids.len()),
1556        None,
1557    );
1558
1559    let mut planned = Vec::new();
1560    // Directories given to exercises earlier in this batch, which are not in the config yet.
1561    let mut reserved_directories: HashMap<Uuid, HashSet<String>> = HashMap::new();
1562    let mut seen_exercise_ids = HashSet::new();
1563    let unplanned = |exercise_id, reason, failure| {
1564        let download = MoocExerciseDownload {
1565            exercise_id,
1566            path: projects_dir.join("mooc"),
1567        };
1568        (download, reason, failure)
1569    };
1570    for &exercise_id in exercise_ids {
1571        // A repeat would be planned into a second directory and extracted in parallel.
1572        if !seen_exercise_ids.insert(exercise_id) {
1573            continue;
1574        }
1575        let Some((course, slide)) = resolved.get(&exercise_id) else {
1576            let (reason, failure) = match closed.remove(&exercise_id) {
1577                Some((reason, detail)) => {
1578                    closed_chapters.push(detail);
1579                    (reason, MoocDownloadFailure::ClosedChapter)
1580                }
1581                None => (
1582                    format!(
1583                        "Exercise {exercise_id} was not found in any of the user's enrolled \
1584                         courses"
1585                    ),
1586                    MoocDownloadFailure::NotFound,
1587                ),
1588            };
1589            failed.push(unplanned(exercise_id, vec![reason], failure));
1590            continue;
1591        };
1592
1593        let (task_id, download_url) = match slide.require_editor_download() {
1594            Ok(editor_download) => editor_download,
1595            Err(err) => {
1596                let err = LangsError::from(err);
1597                failed.push(unplanned(
1598                    exercise_id,
1599                    error_chain(&err),
1600                    mooc_download_failure(&err),
1601                ));
1602                continue;
1603            }
1604        };
1605        let download_url = download_url.to_string();
1606        let checksum = slide.editor_checksum().unwrap_or_default().to_string();
1607        let exercise_name = slide.exercise_name.clone();
1608
1609        let course_config = projects_config.get_or_init_mooc_course_config(
1610            course.id,
1611            course.name.clone(),
1612            course.is_exam,
1613        );
1614
1615        let reserved = reserved_directories.entry(course.id).or_default();
1616        let exercise_directory =
1617            course_config.exercise_directory(exercise_id, &exercise_name, reserved.iter());
1618        let target = ProjectsConfig::get_mooc_exercise_download_target(
1619            projects_dir,
1620            &course_config.directory,
1621            &exercise_directory,
1622        );
1623
1624        if let Some(existing) = course_config.exercises.get(&exercise_id) {
1625            if existing.checksum == checksum && existing.directory == exercise_directory {
1626                log::info!("Skipping exercise {exercise_id} due to identical checksum");
1627                skipped.push(MoocExerciseDownload {
1628                    exercise_id,
1629                    path: target,
1630                });
1631                continue;
1632            }
1633        }
1634
1635        reserved.insert(exercise_directory.clone());
1636        planned.push(PlannedMoocDownload {
1637            exercise_id,
1638            course_id: course.id,
1639            exercise_name,
1640            task_id,
1641            checksum,
1642            exercise_directory,
1643            download_url,
1644            target,
1645        });
1646    }
1647
1648    let results = download_mooc_archives_in_parallel(client, auth, &planned);
1649    let mut permanent_auth_failure = None;
1650    let mut changed_courses = HashSet::new();
1651    for (plan, result) in planned.into_iter().zip(results) {
1652        let exercise_id = plan.exercise_id;
1653        match result {
1654            Ok(()) => {
1655                if let Some(course_config) = projects_config.mooc_courses.get_mut(&plan.course_id) {
1656                    course_config.add_exercise(
1657                        exercise_id,
1658                        plan.exercise_name,
1659                        plan.task_id,
1660                        plan.checksum,
1661                        plan.exercise_directory,
1662                    );
1663                    changed_courses.insert(plan.course_id);
1664                }
1665                downloaded.push(MoocExerciseDownload {
1666                    exercise_id,
1667                    path: plan.target,
1668                });
1669            }
1670            // No session left, so the other downloads failed the same way: surface the auth
1671            // failure instead of a partial result. Finished downloads are still recorded below
1672            // and skipped on retry.
1673            Err(DownloadArchiveError::Auth(auth_err)) if auth_err.is_permanent() => {
1674                permanent_auth_failure.get_or_insert(auth_err);
1675            }
1676            Err(err) => {
1677                let err: LangsError = err.into();
1678                let failure = mooc_download_failure(&err);
1679                failed.push((
1680                    MoocExerciseDownload {
1681                        exercise_id,
1682                        path: plan.target,
1683                    },
1684                    error_chain(&err),
1685                    failure,
1686                ));
1687            }
1688        }
1689    }
1690    for course_id in changed_courses {
1691        if let Some(course_config) = projects_config.mooc_courses.get(&course_id) {
1692            course_config.save_to_projects_dir(projects_dir)?;
1693        }
1694    }
1695    if let Some(auth_err) = permanent_auth_failure {
1696        progress_reporter::finish_stage::<mooc::MoocClientUpdateData>(
1697            "Stopped downloading mooc exercises: not logged in".to_string(),
1698            None,
1699        );
1700        log::error!(
1701            "mooc auth permanently failed while downloading exercises ({} of {} downloaded, {} \
1702             skipped): {auth_err}",
1703            downloaded.len(),
1704            exercise_ids.len(),
1705            skipped.len(),
1706        );
1707        return Err(auth_err.into());
1708    }
1709
1710    progress_reporter::finish_stage::<mooc::MoocClientUpdateData>(
1711        format!("Finished downloading {} mooc exercises", exercise_ids.len()),
1712        None,
1713    );
1714
1715    Ok(DownloadOrUpdateMoocCourseExercisesResult {
1716        downloaded,
1717        skipped,
1718        failed: if failed.is_empty() {
1719            None
1720        } else {
1721            Some(failed)
1722        },
1723        closed_chapters,
1724    })
1725}

1922/// How old a leftover `.tmc-staging-*` directory must be before it is treated as abandoned
1923/// rather than as another running command's.
1924const STALE_STAGING_AGE: std::time::Duration = std::time::Duration::from_secs(24 * 60 * 60);
1925
1926/// Deletes `.tmc-staging-*` directories in `dir` that a killed command left behind.
1927fn sweep_stale_staging_dirs(dir: &Path, min_age: std::time::Duration) {
1928    let Ok(entries) = file_util::read_dir(dir) else {
1929        return;
1930    };
1931    for entry in entries.flatten() {
1932        if !entry
1933            .file_name()
1934            .to_string_lossy()
1935            .starts_with(STAGING_DIR_PREFIX)
1936        {
1937            continue;
1938        }
1939        let is_stale = entry
1940            .metadata()
1941            .and_then(|m| m.modified())
1942            .ok()
1943            .and_then(|modified| modified.elapsed().ok())
1944            .is_some_and(|age| age >= min_age);
1945        if is_stale && entry.file_type().is_ok_and(|t| t.is_dir()) {
1946            log::warn!(
1947                "deleting abandoned staging directory {}",
1948                entry.path().display()
1949            );
1950            if let Err(err) = file_util::remove_dir_all(entry.path()) {
1951                log::warn!("failed to delete it: {err}");
1952            }
1953        }
1954    }
1955}

1970/// A temporary directory beside `path`, on its filesystem, for content to move into `path`
1971/// by rename. Sweeps abandoned ones from earlier runs first.
1972fn staging_dir_beside(path: &Path) -> Result<tempfile::TempDir, LangsError> {
1973    let parent = staging_parent(path)?;

1977        .tempdir_in(parent)

1982/// Replaces the entries of `target` with those of `staged` (a [`staging_dir_beside`]
1983/// `target`), keeping `target` itself: an editor watching the exercise directory loses
1984/// track of it when the directory is replaced.
1985///
1986/// Leaves `target` as it was on failure. If even that fails, the old entries are kept in a
1987/// `.tmc-replaced-*` directory beside `target` and the error names it.
1988fn replace_dir_contents(target: &Path, staged: &Path) -> Result<(), LangsError> {
1989    if !target.exists() {
1990        file_util::rename(staged, target)?;
1991        return Ok(());
1992    }
1993    // Not a guard and never swept: it holds the student's only copy of the old files until
1994    // they are moved back or the swap completes.
1995    let replaced = tempfile::Builder::new()
1996        .prefix(REPLACED_DIR_PREFIX)
1997        .tempdir_in(staging_parent(target)?)
1998        .map_err(FileError::TempFile)?
1999        .keep();
2000
2001    if let Err((moved_out, err)) = move_dir_entries(target, &replaced) {
2002        restore_dir_entries(&replaced, target, &moved_out, &err)?;
2003        return Err(err.into());
2004    }
2005    if let Err((moved_in, err)) = move_dir_entries(staged, target) {
2006        for name in &moved_in {
2007            let _ = file_util::remove_all(target.join(name));
2008        }
2009        let moved_out = dir_entry_names(&replaced)?;
2010        restore_dir_entries(&replaced, target, &moved_out, &err)?;
2011        return Err(err.into());
2012    }
2013    if let Err(err) = file_util::remove_dir_all(&replaced) {
2014        log::warn!(
2015            "replaced {} but could not delete the old files in {}: {err}",
2016            target.display(),
2017            replaced.display()
2018        );
2019    }
2020    Ok(())
2021}

2063/// Moves every entry of `from` into `to` by rename. On failure, returns the names already
2064/// moved with the error.
2065fn move_dir_entries(from: &Path, to: &Path) -> Result<Vec<OsString>, (Vec<OsString>, FileError)> {
2066    let names = dir_entry_names(from).map_err(|err| (vec![], err))?;
2067    let mut moved = Vec::with_capacity(names.len());
2068    for name in names {
2069        if let Err(err) = file_util::rename(from.join(&name), to.join(&name)) {
2070            return Err((moved, err));
2071        }
2072        moved.push(name);
2073    }
2074    Ok(moved)
2075}

2281    target: &Path,

2287/// [`move_dir`], where `try_rename: false` forces the copy fallback used across volumes.
2288fn move_dir_with(
2289    source: &Path,
2290    target: &Path,
2291    try_rename: bool,
2292    record_move: impl FnOnce() -> Result<(), LangsError>,
2293) -> Result<(), LangsError> {
2294    if try_rename {
2295        let is_empty_target = target.is_dir() && dir_entry_names(target)?.is_empty();
2296        if is_empty_target {
2297            file_util::remove_dir_empty(target)?;
2298        }
2299        if let Some(parent) = target.parent() {
2300            file_util::create_dir_all(parent)?;
2301        }
2302        match file_util::rename(source, target) {
2303            Ok(()) => {
2304                if let Err(err) = record_move() {
2305                    if let Err(undo_err) = file_util::rename(target, source) {
2306                        log::error!("failed to move {} back: {undo_err}", target.display());
2307                    }
2308                    return Err(err);
2309                }
2310                return Ok(());
2311            }
2312            Err(err) if file_util::is_cross_volume_error(&err) => {
2313                log::debug!("moving across volumes, copying instead: {err}");
2314            }
2315            Err(err) => {
2316                if is_empty_target {
2317                    file_util::create_dir_all(target)?;
2318                }
2319                return Err(err.into());
2320            }
2321        }
2322    }
2323
2324    file_util::create_dir_all(target)?;
2325    let result = copy_contents_for_move(source, target).and_then(|()| record_move());
2326    if let Err(err) = result {
2327        for name in dir_entry_names(target).unwrap_or_default() {
2328            if let Err(clean_err) = file_util::remove_all(target.join(name)) {
2329                log::warn!("failed to clean up the partial copy: {clean_err}");
2330            }
2331        }
2332        return Err(err);
2333    }
2334
2335    for name in dir_entry_names(source).unwrap_or_default() {
2336        if let Err(err) = file_util::remove_all(source.join(&name)) {
2337            log::warn!(
2338                "failed to delete {} after the move: {err}",
2339                source.display()
2340            );
2341        }
2342    }
2343    if let Err(err) = file_util::remove_dir_empty(source) {
2344        log::warn!(
2345            "failed to delete {} after the move: {err}",
2346            source.display()
2347        );
2348    }
2349    finish_stage("Finished moving project directory");
2350    Ok(())
2351}

2367        file_count_copied += 1;
2368        progress_stage(format!(
2369            "Moved file {file_count_copied} / {file_count_total}"

2373}

2480    /// Builds a `.tar.zst` archive from the given (relative path, contents) pairs.
2481    fn make_tar_zst(files: &[(&str, &[u8])]) -> Vec<u8> {
2482        let mut tar_buf = Vec::new();
2483        {
2484            let mut builder = tar::Builder::new(&mut tar_buf);
2485            for (path, contents) in files {
2486                let mut header = tar::Header::new_gnu();
2487                header.set_size(contents.len() as u64);
2488                header.set_mode(0o644);
2489                header.set_cksum();
2490                builder.append_data(&mut header, path, *contents).unwrap();
2491            }
2492            builder.finish().unwrap();
2493        }
2494        zstd::encode_all(std::io::Cursor::new(tar_buf), 0).unwrap()
2495    }
