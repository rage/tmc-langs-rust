31/// Serializes tests that set `TMC_LANGS_CONFIG_DIR` or the bearer-token trust knob.
32/// Crate-wide because a per-module lock would not serialize process-wide env vars.
33#[cfg(test)]
34pub(crate) fn env_lock() -> std::sync::MutexGuard<'static, ()> {
35    static ENV_LOCK: std::sync::Mutex<()> = std::sync::Mutex::new(());
36    ENV_LOCK.lock().unwrap_or_else(|e| e.into_inner())
37}

39// base directory for a given plugin's settings files
40pub(crate) fn get_tmc_dir(client_name: &str) -> Result<PathBuf, LangsError> {
41    let config_dir = match env::var(TMC_LANGS_CONFIG_DIR_VAR) {
42        Ok(v) => PathBuf::from(v),
43        Err(_) => dirs::config_dir().ok_or(LangsError::NoConfigDir)?,
44    };
45    Ok(config_dir.join(format!("tmc-{client_name}")))
46}

48/// Returns every exercise in the projects directory, from both backends, ordered
49/// by course slug and then exercise slug within each backend.
50///
51/// This is the whole-projects-dir listing; [`list_local_tmc_course_exercises`]
52/// and [`list_local_mooc_course_exercises`] are filtered views of it.
53pub fn list_local_exercises(client_name: &str) -> Result<Vec<LocalExercise>, LangsError> {
54    log::debug!("listing all local exercises for {client_name}");
55
56    let projects_dir = TmcConfig::load(client_name)?.projects_dir;
57    let projects_config = ProjectsConfig::load(&projects_dir)?;
58
59    let mut local_exercises: Vec<LocalExercise> = vec![];
60
61    let mut tmc_courses = projects_config.tmc_courses.iter().collect::<Vec<_>>();
62    tmc_courses.sort_by_key(|(course_slug, _)| *course_slug);
63    for (course_slug, course_config) in tmc_courses {
64        for (exercise_slug, exercise) in &course_config.exercises {
65            local_exercises.push(LocalExercise::Tmc(LocalTmcExercise {
66                course_slug: course_slug.clone(),
67                exercise_slug: exercise_slug.clone(),
68                exercise_id: exercise.id,
69                exercise_path: ProjectsConfig::get_tmc_exercise_download_target(
70                    &projects_dir,
71                    course_slug,
72                    exercise_slug,
73                ),
74            }));
75        }
76    }
77
78    let mut mooc_courses = projects_config.mooc_courses.values().collect::<Vec<_>>();
79    mooc_courses.sort_by(|a, b| a.directory.cmp(&b.directory));
80    for course_config in mooc_courses {
81        for (exercise_id, exercise) in &course_config.exercises {
82            local_exercises.push(LocalExercise::Mooc(LocalMoocExercise {
83                course_slug: course_config.directory.clone(),
84                course_id: course_config.course_id,
85                exercise_slug: exercise.directory.clone(),
86                exercise_id: *exercise_id,
87                exercise_path: ProjectsConfig::get_mooc_exercise_download_target(
88                    &projects_dir,
89                    &course_config.directory,
90                    &exercise.directory,
91                ),
92            }));
93        }
94    }
95
96    Ok(local_exercises)
97}

99/// Returns all of the exercises for the given TMC course, identified by its
100/// on-disk directory name.
101pub fn list_local_tmc_course_exercises(
102    client_name: &str,
103    course_slug: &str,
104) -> Result<Vec<LocalTmcExercise>, LangsError> {
105    log::debug!("listing local course exercises of {course_slug} for {client_name}");
106
107    Ok(list_local_exercises(client_name)?
108        .into_iter()
109        .filter_map(|exercise| match exercise {
110            LocalExercise::Tmc(exercise) if exercise.course_slug == course_slug => Some(exercise),
111            _ => None,
112        })
113        .collect())
114}

116/// Returns the local mooc exercises for the given course, looked up by course id
117/// (mooc courses have no slug, unlike [`list_local_tmc_course_exercises`]).
118pub fn list_local_mooc_course_exercises(
119    client_name: &str,
120    course_id: Uuid,
121) -> Result<Vec<LocalMoocExercise>, LangsError> {
122    log::debug!("listing local course exercises of {course_id} for {client_name}");
123
124    Ok(list_local_exercises(client_name)?
125        .into_iter()
126        .filter_map(|exercise| match exercise {
127            LocalExercise::Mooc(exercise) if exercise.course_id == course_id => Some(exercise),
128            _ => None,
129        })
130        .collect())
131}

133/// Migrates an exercise from a location that's not managed by tmc-langs to the projects directory.
134pub fn migrate_exercise(
135    tmc_config: TmcConfig,
136    course_slug: &str,
137    exercise_slug: &str,
138    exercise_id: u32,
139    exercise_checksum: &str,
140    exercise_path: &Path,
141) -> Result<(), LangsError> {
142    log::debug!(
143        "migrating exercise {} from {}",
144        exercise_id,
145        exercise_path.display()
146    );
147
148    let mut lock = Lock::dir(exercise_path, LockOptions::Write)?;
149    let _guard = lock.lock()?;
150    let mut projects_config = ProjectsConfig::load(&tmc_config.projects_dir)?;
151    let course_config = projects_config
152        .tmc_courses
153        .entry(course_slug.to_string())
154        .or_insert(TmcCourseConfig {
155            course: course_slug.to_string(),
156            exercises: BTreeMap::new(),
157        });
158
159    let target_dir = ProjectsConfig::get_tmc_exercise_download_target(
160        &tmc_config.projects_dir,
161        course_slug,
162        exercise_slug,
163    );
164    if target_dir.exists() {
165        return Err(LangsError::DirectoryExists(target_dir));
166    }
167
168    course_config.exercises.insert(
169        exercise_slug.to_string(),
170        ProjectsDirTmcExercise {
171            id: exercise_id,
172            checksum: exercise_checksum.to_string(),
173        },
174    );
175
176    super::move_dir(exercise_path, &target_dir)?;
177    course_config.save_to_projects_dir(&tmc_config.projects_dir)?;
178    Ok(())
179}

181/// Moves the projects directory from its current location to the target, taking all of the contained exercises with it.
182pub fn move_projects_dir(mut tmc_config: TmcConfig, target: PathBuf) -> Result<(), LangsError> {
183    log::debug!("moving projects dir to {}", target.display());
184
185    if target.is_file() {
186        return Err(FileError::UnexpectedFile(target).into());
187    }
188    if !target.exists() {
189        file_util::create_dir_all(&target)?;
190    }
191
192    let target_canon = target
193        .canonicalize()
194        .map_err(|e| LangsError::Canonicalize(target.clone(), e))?;
195    let prev_dir_canon = tmc_config
196        .projects_dir
197        .canonicalize()
198        .map_err(|e| LangsError::Canonicalize(target.clone(), e))?;
199    if target_canon == prev_dir_canon {
200        return Err(LangsError::MovingProjectsDirToItself);
201    }
202
203    let old_projects_dir = tmc_config.set_projects_dir(target.clone())?;
204
205    let mut lock = Lock::dir(old_projects_dir.clone(), LockOptions::Write)?;
206    let _guard = lock.lock()?;
207    super::move_dir(&old_projects_dir, &target)?;
208    tmc_config.save()?;
209    Ok(())
210}

224    fn file_to(
225        target_dir: impl AsRef<std::path::Path>,
226        target_relative: impl AsRef<std::path::Path>,
227        contents: impl AsRef<[u8]>,
228    ) -> PathBuf {
229        let target = target_dir.as_ref().join(target_relative);
230        if let Some(parent) = target.parent() {
231            std::fs::create_dir_all(parent).unwrap();
232        }
233        std::fs::write(&target, contents.as_ref()).unwrap();
234        target
235    }
