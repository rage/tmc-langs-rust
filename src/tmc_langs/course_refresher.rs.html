49#[derive(Debug, Serialize, Deserialize, JsonSchema)]

51pub struct RepositoryExercise {
52    /// The exercise directory.
53    pub path: PathBuf,
54    /// The directories between the repository root and the exercise directory, joined with `-`.
55    /// Together with `name`, the name `refresh-course` gives the exercise.
56    pub part: String,
57    /// The exercise directory's name.
58    pub name: String,
59    /// The language plugin that recognised the exercise, as `get-exercise-packaging-configuration`

62    /// How long the exercise's tests may run, with the repository's root config merged in.

64    /// The files this command wrote into the exercise directory, relative to it.
65    pub written_files: Vec<PathBuf>,

68/// Finds the exercises in a course repository checkout by the rules [`refresh_course`] uses, and
69/// writes the repository's root `.tmcproject.yml` into each exercise directory, merged under
70/// the exercise's own, so the directory alone carries the exercise's configuration. The
71/// repository root itself is never an exercise.
72pub fn prepare_exercise_repository(
73    repository: &Path,
74) -> Result<Vec<RepositoryExercise>, LangsError> {
75    log::info!("preparing the exercises in {}", repository.display());
76
77    let exercise_dirs: Vec<PathBuf> = super::find_exercise_directories(repository)?
78        .into_iter()
79        .map(|dir| {
80            dir.strip_prefix(repository)
81                .expect("exercise directories are inside the repository")
82                .to_path_buf()
83        })
84        .filter(|relative| !relative.as_os_str().is_empty())
85        .collect();
86    // Without a root config there is nothing to merge in, so the exercises' own stay untouched.
87    let exercises = match TmcProjectYml::load(repository)? {
88        Some(root) => get_and_merge_tmcproject_configs(Some(root), repository, exercise_dirs)?,
89        None => exercise_dirs.into_iter().map(|dir| (dir, None)).collect(),
90    };
91    exercises
92        .into_iter()
93        .map(|(relative, merged_tmcproject_yml)| {
94            let path = repository.join(&relative);
95            let mut written_files = Vec::new();
96            if let Some(merged) = merged_tmcproject_yml {
97                merged.save_to_dir(&path)?;
98                written_files.push(PathBuf::from(TmcProjectYml::FILE_NAME));
99            }
100            let packaging = super::get_exercise_packaging_configuration(&path)?;
101            let part = match relative.parent() {
102                Some(parent) if !parent.as_os_str().is_empty() => exercise_name(parent)?,
103                _ => String::new(),
104            };
105            let name = relative
106                .file_name()
107                .expect("exercise directories have names")
108                .to_string_lossy()
109                .into_owned();
110            Ok(RepositoryExercise {
111                path,
112                part,
113                name,
114                plugin: packaging
115                    .plugin
116                    .expect("the packaging configuration names its plugin"),
117                tests_timeout_ms: packaging
118                    .tests_timeout_ms
119                    .expect("the packaging configuration has a tests timeout"),
120                written_files,
121            })
122        })
123        .collect()
124}

126/// The name `refresh-course` gives the exercise at `relative` in the repository: its path with
127/// the separators replaced by `-`.
128fn exercise_name(relative: &Path) -> Result<String, LangsError> {
129    Ok(entry_name(relative)?.replace('/', "-"))

132/// Used by tmc-server. Refreshes the course.
133pub fn refresh_course(
134    course_name: String,
135    course_cache_path: PathBuf,
136    source_url: String,
137    git_branch: String,
138    cache_root: PathBuf,
139) -> Result<RefreshData, LangsError> {
140    log::info!("refreshing course {course_name}");
141    start_stage(10, "Refreshing course");
142
143    // create new cache path
144    let old_version = course_cache_path
145        .to_str()
146        .and_then(|s| s.split('-').next_back())
147        .and_then(|s| s.parse::<u32>().ok())
148        .ok_or_else(|| LangsError::InvalidCachePath(course_cache_path.clone()))?;
149    let new_cache_path = cache_root.join(format!("{}-{}", course_name, old_version + 1));
150    log::info!("next cache path: {}", new_cache_path.display());
151
152    if new_cache_path.exists() {
153        log::info!("clearing new cache path at {}", new_cache_path.display());
154        file_util::remove_dir_all(&new_cache_path)?;
155    }
156    file_util::create_dir_all(&new_cache_path)?;
157    progress_stage("Created new cache dir");
158
159    // initialize new clone path and verify directory names
160    let new_clone_path = new_cache_path.join("clone");
161    let old_clone_path = course_cache_path.join("clone");
162    initialize_new_cache_clone(
163        &new_cache_path,
164        &new_clone_path,
165        &old_clone_path,
166        &source_url,
167        &git_branch,
168    )?;
169    check_directory_names(&new_clone_path)?;
170    progress_stage("Updated repository");
171
172    let course_options = get_course_options(&new_clone_path, &course_name)?;
173    progress_stage("Fetched course options");
174
175    let new_solution_path = new_cache_path.join("solution");
176    let new_stub_path = new_cache_path.join("stub");
177
178    let exercise_dirs = super::find_exercise_directories(&new_clone_path)?
179        .into_iter()
180        .map(|ed| {
181            ed.strip_prefix(&new_clone_path)
182                .expect("exercise directories are inside new_clone_path")
183                .to_path_buf()
184        })
185        .collect::<Vec<_>>();
186
187    // collect .tmcproject.ymls and merge the root config with each exercise's, if any
188    let root_tmcproject_yml = TmcProjectYml::load(&new_clone_path)?;
189    let exercise_dirs_and_tmcprojects =
190        get_and_merge_tmcproject_configs(root_tmcproject_yml, &new_clone_path, exercise_dirs)?;
191    progress_stage("Merged .tmcproject.yml files in exercise directories to the root file, if any");
192
193    // make_solutions
194    log::info!("preparing solutions to {}", new_solution_path.display());
195    for (exercise, merged_tmcproject) in &exercise_dirs_and_tmcprojects {
196        // save merged config to solution
197        let dest_root = new_solution_path.join(exercise);
198        super::prepare_solution(&new_clone_path.join(exercise), &dest_root)?;
199        if let Some(merged_tmcproject) = merged_tmcproject {
200            merged_tmcproject.save_to_dir(&dest_root)?;
201        }
202    }
203    progress_stage("Prepared solutions");
204
205    // make_stubs
206    log::info!("preparing stubs to {}", new_stub_path.display());
207    for (exercise, merged_tmcproject) in &exercise_dirs_and_tmcprojects {
208        // save merged config to stub
209        let dest_root = new_stub_path.join(exercise);
210        super::prepare_stub(&new_clone_path.join(exercise), &dest_root)?;
211        if let Some(merged_tmcproject) = merged_tmcproject {
212            merged_tmcproject.save_to_dir(&dest_root)?;
213        }
214    }
215    progress_stage("Prepared stubs");
216
217    let exercises = get_exercises(
218        exercise_dirs_and_tmcprojects,
219        &new_clone_path,
220        &new_stub_path,
221    )?;
222    progress_stage("Located exercises");
223
224    // make_zips_of_solutions
225    let new_solution_zip_path = new_cache_path.join("solution_zip");
226    execute_zip(&exercises, &new_solution_path, &new_solution_zip_path)?;
227    progress_stage("Compressed solutions");
228
229    // make_zips_of_stubs
230    let new_stub_zip_path = new_cache_path.join("stub_zip");
231    log::info!(
232        "compressing stubs from {} to {}",
233        new_stub_path.display(),
234        new_stub_zip_path.display()
235    );
236    execute_zip(&exercises, &new_stub_path, &new_stub_zip_path)?;
237    progress_stage("Compressed stubs");
238
239    // make sure the new cache path is readable by anyone
240    set_permissions(&new_cache_path)?;
241
242    finish_stage("Refreshed course");
243    Ok(RefreshData {
244        new_cache_path,
245        course_options,
246        exercises,
247    })
248}

250/// Checks old_cache_path/clone for a git repo.
251/// If found, copies it to course_clone_path fetches origin from course_source_url, checks out origin/course_git_branch, cleans and checks out the repo.
252/// If not found or found but one of the git commands causes an error, deletes course_clone_path and clones course_git_branch from course_source_url there.
253/// NOP during testing.
254fn initialize_new_cache_clone(
255    new_course_root: &Path,
256    new_clone_path: &Path,
257    old_clone_path: &Path,
258    course_source_url: &str,
259    course_git_branch: &str,
260) -> Result<(), LangsError> {
261    log::info!("initializing repository at {}", new_clone_path.display());
262
263    if old_clone_path.join(".git").exists() {
264        log::info!(
265            "trying to copy clone from previous cache at {}",
266            old_clone_path.display()
267        );
268
269        // closure to collect any error that occurs during the process
270        let copy_and_update_repository = || -> Result<(), LangsError> {
271            file_util::copy(old_clone_path, new_course_root)?;
272
273            let run_git = |args: &[&str]| {
274                TmcCommand::piped("git")
275                    .with(|e| e.cwd(new_clone_path).args(args))
276                    .output_with_timeout_checked(Duration::from_secs(60 * 2))
277            };
278
279            run_git(&["remote", "set-url", "origin", course_source_url])?;
280            run_git(&["fetch", "origin"])?;
281            run_git(&["checkout", &format!("origin/{course_git_branch}")])?;
282            run_git(&["clean", "-df"])?;
283            run_git(&["checkout", "."])?;
284            Ok(())
285        };
286        match copy_and_update_repository() {
287            Ok(_) => {
288                log::info!("updated repository");
289                return Ok(());
290            }
291            Err(error) => {
292                log::warn!("failed to update repository: {error}");
293
294                file_util::remove_dir_all(new_clone_path)?;
295            }
296        }
297    };
298
299    log::info!("could not copy from previous cache, cloning");
300
301    // clone_repository
302    TmcCommand::piped("git")
303        .with(|e| {
304            e.args(["clone", "-q", "-b"])
305                .arg(course_git_branch)
306                .arg(course_source_url)
307                .arg(new_clone_path)
308        })
309        .output_with_timeout_checked(Duration::from_secs(60 * 2))?;
310    Ok(())
311}

313/// Makes sure no directory directly under path is an exercise directory containing a dash in the relative path from path to the dir.
314/// A dash is used as a special delimiter.
315fn check_directory_names(path: &Path) -> Result<(), LangsError> {
316    log::info!("checking directory names for dashes");
317
318    // exercise directories in canonicalized form
319    for exercise_dir in super::find_exercise_directories(path)? {
320        let relative = exercise_dir
321            .strip_prefix(path)
322            .expect("the exercise dirs are all inside the path");
323        if relative.to_string_lossy().contains('-') {
324            return Err(LangsError::InvalidDirectory(exercise_dir));
325        }
326    }
327    Ok(())
328}

330fn get_and_merge_tmcproject_configs(
331    root_tmcproject: Option<TmcProjectYml>,
332    clone_path: &Path,
333    exercise_dirs: Vec<PathBuf>,
334) -> Result<Vec<(PathBuf, Option<TmcProjectYml>)>, LangsError> {
335    let mut res = vec![];
336    for exercise_dir in exercise_dirs {
337        let target_dir = clone_path.join(&exercise_dir);
338        let exercise_tmcproject = TmcProjectYml::load(&target_dir)?;
339        match (&root_tmcproject, exercise_tmcproject) {
340            (Some(root), Some(mut exercise)) => {
341                exercise.merge(root.clone());
342                res.push((exercise_dir, Some(exercise)));
343            }
344            (Some(root), None) => {
345                res.push((exercise_dir, Some(root.clone())));
346            }
347            (None, Some(exercise)) => res.push((exercise_dir, Some(exercise))),
348            (None, None) => res.push((exercise_dir, None)),
349        }
350    }
351    Ok(res)
352}

354/// Checks for a course_clone_path/course_options.yml
355/// If found, course-specific options are merged into it and it is returned.
356/// Else, an empty mapping is returned.
357fn get_course_options(course_clone_path: &Path, course_name: &str) -> Result<Mapping, LangsError> {
358    log::info!(
359        "collecting course options for {} in {}",
360        course_name,
361        course_clone_path.display()
362    );
363
364    let options_file = course_clone_path.join("course_options.yml");
365    if options_file.exists() {
366        let file = file_util::open_file(&options_file)?;
367        let course_options: Mapping = deserialize::yaml_from_reader(file)
368            .map_err(|e| LangsError::DeserializeYaml(options_file, e))?;
369        Ok(course_options)
370    } else {
371        Ok(Mapping::new())
372    }
373}

375/// Finds exercise directories, and converts the directories to "exercise names" by swapping the separators for dashes.
376/// Also calculates checksums and fetches points for all
377fn get_exercises(
378    exercise_dirs_and_tmcprojects: Vec<(PathBuf, Option<TmcProjectYml>)>,
379    course_clone_path: &Path,
380    course_stub_path: &Path,
381) -> Result<Vec<RefreshExercise>, LangsError> {
382    log::info!("finding exercise checksums and points");
383
384    let exercises = exercise_dirs_and_tmcprojects
385        .into_iter()
386        .map(|(exercise_dir, tmcproject_yml)| {
387            log::debug!(
388                "processing points and checksum for {}",
389                exercise_dir.display()
390            );
391            let name = exercise_name(&exercise_dir)?;
392            let checksum = calculate_checksum(&course_stub_path.join(&exercise_dir))?;
393            let exercise_path = course_clone_path.join(&exercise_dir);
394            let points = super::get_available_points(&exercise_path)?;
395
396            let sandbox_image = if let Some(image_override) = tmcproject_yml
397                .as_ref()
398                .and_then(|y| y.sandbox_image.as_ref())
399            {
400                image_override.clone()
401            } else {
402                crate::get_default_sandbox_image(&exercise_path)?.to_string()
403            };
404
405            Ok(RefreshExercise {
406                name,
407                points,
408                checksum,
409                path: exercise_dir,
410                sandbox_image,
411                tmcproject_yml,
412            })
413        })
414        .collect::<Result<_, LangsError>>()?;
415    Ok(exercises)
416}

418fn calculate_checksum(exercise_dir: &Path) -> Result<String, LangsError> {
419    let mut digest = Context::new();
420
421    // order filenames for a consistent hash
422    for entry in WalkDir::new(exercise_dir)
423        .min_depth(1) // do not hash the directory itself ('.')
424        .sort_by(|a, b| a.file_name().cmp(b.file_name()))
425    {
426        let entry = entry?;
427        let relative = entry
428            .path()
429            .strip_prefix(exercise_dir)
430            .expect("the entry is inside the exercise dir");
431        digest.consume(entry_name(relative)?);
432        if entry.path().is_file() {
433            let file = file_util::read_file(entry.path())?;
434            digest.consume(file);
435        }
436    }
437
438    // convert the digest into a hex string
439    let digest = digest.finalize();
440    Ok(format!("{digest:x}"))
441}

443fn execute_zip(
444    course_exercises: &[RefreshExercise],
445    root_path: &Path,
446    zip_dir: &Path,
447) -> Result<(), LangsError> {
448    log::info!(
449        "compressing exercises from from {} to {}",
450        root_path.display(),
451        zip_dir.display()
452    );
453
454    file_util::create_dir_all(zip_dir)?;
455    for exercise in course_exercises {
456        let exercise_root = root_path.join(&exercise.path);
457        let zip_file_path = zip_dir.join(format!("{}.zip", exercise.name));
458
459        let mut writer = ArchiveBuilder::new(
460            file_util::create_file(zip_file_path)?,
461            Compression::Zip,
462            None,
463            false,
464            false,
465        );
466        for entry in WalkDir::new(exercise_root) {
467            let entry = entry?;
468            let relative_path = entry
469                .path()
470                .strip_prefix(root_path)
471                .expect("entries are inside root_path");
472
473            if entry.path().is_file() {
474                writer.add_file(entry.path(), relative_path)?;
475            } else {
476                // java-langs expects directories to have their own entries
477                writer.add_directory(entry.path(), relative_path)?;
478            }
479        }
480        writer.finish()?;
481    }
482    Ok(())
483}

491#[cfg(unix)]
492fn set_permissions(path: &Path) -> Result<(), LangsError> {
493    use nix::sys::stat;
494    use std::os::fd::AsFd;
495
496    log::info!("setting permissions in {}", path.display());
497
498    let chmod: ModeBits = 0o775; // octal, read and execute permissions for all users
499    for entry in WalkDir::new(path) {
500        let entry = entry?;
501        let file = file_util::open_file(entry.path())?;
502        stat::fchmod(
503            file.as_fd(),
504            stat::Mode::from_bits(chmod).ok_or(LangsError::NixFlag(chmod))?,
505        )
506        .map_err(|e| LangsError::NixPermissionChange(path.to_path_buf(), e))?;
507    }
508
509    Ok(())
510}

707                    .join(".tmcproject.yml")
708                    .to_string_lossy(),
709            )
710            .unwrap();

712        file.read_to_string(&mut buf).unwrap();
713        assert_eq!(buf, "some: 'yaml'");

887        };

897
