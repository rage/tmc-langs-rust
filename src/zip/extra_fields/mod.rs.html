19/// marker trait to denote the place where this extra field has been stored
20pub trait ExtraFieldVersion {}
21
22/// use this to mark extra fields specified in a local header
23
24#[derive(Debug, Clone)]
25pub struct LocalHeaderVersion;
26
27/// use this to mark extra fields specified in the central header
28
29#[derive(Debug, Clone)]
30pub struct CentralHeaderVersion;
31
32impl ExtraFieldVersion for LocalHeaderVersion {}
33impl ExtraFieldVersion for CentralHeaderVersion {}

35/// contains one extra field
36#[derive(Debug, Clone)]
37pub enum ExtraField {
38    /// NTFS extra field
39    Ntfs(Ntfs),
40
41    /// extended timestamp, as described in <https://libzip.org/specifications/extrafld.txt>
42    ExtendedTimestamp(ExtendedTimestamp),
43}

45/// Extra field used in this crate
46#[repr(u16)]
47#[derive(Debug, Copy, Clone, PartialEq, Eq)]
48pub(crate) enum UsedExtraField {
49    /// ZIP64 extended information extra field
50    Zip64ExtendedInfo = 0x0001,
51    /// NTFS
52    Ntfs = 0x000a,
53    /// extended timestamp
54    /// from <https://libzip.org/specifications/extrafld.txt>
55    ExtendedTimestamp = 0x5455,
56    /// Info-ZIP Unicode Comment Extra Field
57    UnicodeComment = 0x6375,
58    /// Info-ZIP Unicode Path Extra Field
59    UnicodePath = 0x7075,
60    /// AE-x encryption structure
61    AeXEncryption = 0x9901,
62    /// Data Stream Alignment (Apache Commons-Compress)
63    DataStreamAlignment = 0xa11e,
64}
65
66impl UsedExtraField {
67    pub const fn to_le_bytes(self) -> [u8; 2] {
68        let field_u16 = self.as_u16();
69        field_u16.to_le_bytes()
70    }
71
72    pub const fn as_u16(self) -> u16 {
73        self as u16
74    }
75}
76
77impl From<UsedExtraField> for u16 {
78    fn from(value: UsedExtraField) -> Self {
79        value.as_u16()
80    }
81}
82
83impl Display for UsedExtraField {
84    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
85        write!(f, "0x{:04X}", *self as u16)
86    }
87}

89macro_rules! extra_field_match {
90    ($x:expr, $( $variant:path ),+ $(,)?) => {
91        match $x {
92            $(
93                v if v == $variant as u16 => Ok($variant),
94            )+
95            _ => Err(()),
96        }
97    };
98}
99
100impl TryFrom<u16> for UsedExtraField {
101    type Error = ();
102
103    fn try_from(value: u16) -> Result<Self, Self::Error> {
104        extra_field_match!(
105            value,
106            UsedExtraField::Zip64ExtendedInfo,
107            UsedExtraField::Ntfs,
108            UsedExtraField::ExtendedTimestamp,
109            UsedExtraField::UnicodeComment,
110            UsedExtraField::UnicodePath,
111            UsedExtraField::DataStreamAlignment,
112            UsedExtraField::AeXEncryption,
113        )
114    }
115}
