1//! OAuth2 Device Authorization Grant (RFC 8628) and refresh-token flows for the
2//! Courses MOOC backend.
3//!
4//! Hand-rolled on `reqwest` instead of the `oauth2` crate's device-flow helper,
5//! which blocks until approval: it cannot hand the caller the verification URL
6//! before polling, nor expose per-poll `slow_down` / `authorization_pending`
7//! for a cancellable loop. Only the crate's [`api::Token`] type is reused.
8//!
9//! Endpoints (`POST .../device_authorization`, `POST .../token`) are mounted at
10//! `/api/v0/main-frontend/oauth` in the backend.
11

27const FORM_CONTENT_TYPE: HeaderValue =
28    HeaderValue::from_static("application/x-www-form-urlencoded");
29
30/// Overall timeout for auth-flow HTTP requests.
31///
32/// The refresh path holds the refresh mutex and the credentials file lock across
33/// the call (`tmc-langs` `mooc_credentials::refresh_locked`), so a hung connection
34/// would block every CLI process. Public so that caller can assert its lock-wait
35/// budget exceeds this.
36pub const AUTH_REQUEST_TIMEOUT: Duration = Duration::from_secs(30);
37
38/// Blocking HTTP client for the auth flows, with [`AUTH_REQUEST_TIMEOUT`].
39/// Panics if the client cannot be built, like `reqwest::blocking::Client::new()`.
40fn auth_client() -> Client {
41    Client::builder()
42        .user_agent(crate::user_agent())
43        .timeout(AUTH_REQUEST_TIMEOUT)
44        .build()
45        .expect("failed to build auth HTTP client")
46}

48/// Encodes `application/x-www-form-urlencoded` bodies; `reqwest` is built
49/// without `RequestBuilder::form`.
50fn urlencode(pairs: &[(&str, &str)]) -> String {
51    let mut serializer = url::form_urlencoded::Serializer::new(String::new());
52    for (k, v) in pairs {
53        serializer.append_pair(k, v);
54    }
55    serializer.finish()
56}

68/// RFC 8628 §3.5 default polling interval (seconds) when the authorization

72/// The authorization server's response to a device authorization request
73/// (RFC 8628 §3.2).
74#[derive(Debug, Clone, Deserialize)]
75pub struct DeviceAuthorizationResponse {
76    /// The device verification code, polled on the token endpoint.
77    pub device_code: String,
78    /// The end-user code shown to the user to enter on the verification page.
79    pub user_code: String,
80    /// The URL the user visits to enter the `user_code`.
81    pub verification_uri: String,
82    /// A URL that already includes the `user_code`, if the server provides one.
83    #[serde(default)]
84    pub verification_uri_complete: Option<String>,
85    /// Lifetime in seconds of the `device_code` and `user_code`.
86    pub expires_in: u32,
87    /// Minimum interval in seconds between token-endpoint polls.
88    #[serde(default = "default_interval")]
89    pub interval: u32,
90}

92fn default_interval() -> u32 {
93    DEFAULT_POLL_INTERVAL_SECS

96/// The result of a single token-endpoint poll during the device flow.
97#[derive(Debug)]
98pub enum DeviceTokenPoll {
99    /// `authorization_pending`: the user has not yet approved; keep polling.
100    Pending,
101    /// `slow_down`: poll less frequently (interval must be increased by 5s).
102    SlowDown,
103    /// The grant was approved; the token pair was issued.
104    Authorized(Box<api::Token>),
105}
106
107/// Requests a device + user code from the authorization server (RFC 8628 §3.1).
108pub fn device_authorization(
109    root_url: &Url,
110    client_id: &str,
111) -> MoocClientResult<DeviceAuthorizationResponse> {
112    let url = oauth_url(root_url, "device_authorization")?;
113    let client = auth_client();
114    let body = urlencode(&[("client_id", client_id), ("scope", DEVICE_SCOPE)]);
115    let response = client
116        .post(url.clone())
117        .header(CLIENT_VERSION_HEADER, env!("CARGO_PKG_VERSION"))
118        .header(CONTENT_TYPE, FORM_CONTENT_TYPE)
119        .body(body)
120        .send()
121        .map_err(|e| {
122            Box::new(MoocClientError::ConnectionError(
123                Method::POST,
124                url.clone(),
125                e.without_url(),
126            ))
127        })?;
128
129    let status = response.status();
130    if status.is_success() {
131        let bytes = read_bytes(response, &url)?;
132        let parsed = serde_json::from_slice(&bytes).map_err(|e| {
133            Box::new(MoocClientError::DeserializingResponse {
134                url: url.clone(),
135                error: Box::new(e),
136            })
137        })?;
138        Ok(parsed)
139    } else {
140        Err(http_error(response, url))
141    }
142}

144/// Polls the token endpoint once with the device-code grant (RFC 8628 §3.4).
145///
146/// `authorization_pending` and `slow_down` map to [`DeviceTokenPoll`] variants;
147/// `expired_token` and `access_denied` map to [`MoocClientError::DeviceCodeExpired`]
148/// and [`MoocClientError::DeviceAccessDenied`]. Other failures are plain errors.
149pub fn poll_device_token(
150    root_url: &Url,
151    client_id: &str,
152    device_code: &str,
153) -> MoocClientResult<DeviceTokenPoll> {
154    let url = oauth_url(root_url, "token")?;
155    let client = auth_client();
156    let body = urlencode(&[
157        ("grant_type", DEVICE_GRANT_TYPE),
158        ("device_code", device_code),
159        ("client_id", client_id),
160    ]);
161    let response = client
162        .post(url.clone())
163        .header(CLIENT_VERSION_HEADER, env!("CARGO_PKG_VERSION"))
164        .header(CONTENT_TYPE, FORM_CONTENT_TYPE)
165        .body(body)
166        .send()
167        .map_err(|e| {
168            Box::new(MoocClientError::ConnectionError(
169                Method::POST,
170                url.clone(),
171                e.without_url(),
172            ))
173        })?;
174
175    let status = response.status();
176    if status.is_success() {
177        let token = parse_token(response, &url)?;
178        return Ok(DeviceTokenPoll::Authorized(Box::new(token)));
179    }
180
181    // RFC 8628 §3.5: every non-success state is a 400 with an OAuth error code.
182    let obsolete_client = status == StatusCode::UPGRADE_REQUIRED;
183    let meta = ResponseMeta::of(response.headers());
184    let body = read_text(response, &url)?;
185    match parse_oauth_error_code(&body).as_deref() {
186        Some("authorization_pending") => Ok(DeviceTokenPoll::Pending),
187        Some("slow_down") => Ok(DeviceTokenPoll::SlowDown),
188        Some("expired_token") => Err(Box::new(MoocClientError::DeviceCodeExpired)),
189        Some("access_denied") => Err(Box::new(MoocClientError::DeviceAccessDenied)),
190        _ => Err(Box::new(MoocClientError::HttpError {
191            url,
192            status,
193            error: body,
194            obsolete_client,
195            message_key: None,
196            meta,
197        })),
198    }
199}

201/// How long a logout waits for the revocation; offline it must still log out promptly.
202const REVOKE_TIMEOUT: Duration = Duration::from_secs(5);
203
204/// Revokes a refresh token at the authorization server (RFC 7009), ending this device's login
205/// there; the server answers success for an already invalid token too.
206pub fn revoke_refresh_token(
207    root_url: &Url,
208    client_id: &str,
209    refresh_token: &str,
210) -> MoocClientResult<()> {
211    let url = oauth_url(root_url, "revoke")?;
212    let body = urlencode(&[
213        ("token", refresh_token),
214        ("token_type_hint", "refresh_token"),
215        ("client_id", client_id),
216    ]);
217    let response = auth_client()
218        .post(url.clone())
219        .timeout(REVOKE_TIMEOUT)
220        .header(CLIENT_VERSION_HEADER, env!("CARGO_PKG_VERSION"))
221        .header(CONTENT_TYPE, FORM_CONTENT_TYPE)
222        .body(body)
223        .send()
224        .map_err(|e| {
225            Box::new(MoocClientError::ConnectionError(
226                Method::POST,
227                url.clone(),
228                e.without_url(),
229            ))
230        })?;
231    if response.status().is_success() {
232        return Ok(());
233    }
234    Err(http_error(response, url))
235}

237/// Redeems a refresh token for a fresh token pair (OAuth2 refresh grant).
238///
239/// The backend rotates the refresh token, returning a new one; if a response
240/// omits it, the old refresh token is carried over so the stored credentials
241/// remain refreshable.
242pub fn refresh_token(
243    root_url: &Url,
244    client_id: &str,
245    refresh_token: &str,
246) -> MoocClientResult<api::Token> {
247    let url = oauth_url(root_url, "token")?;
248    let client = auth_client();
249    let body = urlencode(&[
250        ("grant_type", "refresh_token"),
251        ("refresh_token", refresh_token),
252        ("client_id", client_id),
253    ]);
254    let response = client
255        .post(url.clone())
256        .header(CLIENT_VERSION_HEADER, env!("CARGO_PKG_VERSION"))
257        .header(CONTENT_TYPE, FORM_CONTENT_TYPE)
258        .body(body)
259        .send()
260        .map_err(|e| {
261            Box::new(MoocClientError::ConnectionError(
262                Method::POST,
263                url.clone(),
264                e.without_url(),
265            ))
266        })?;
267
268    if response.status().is_success() {
269        let mut token = parse_token(response, &url)?;
270        if token.refresh_token().is_none() {
271            token.set_refresh_token(Some(RefreshToken::new(refresh_token.to_string())));
272        }
273        return Ok(token);
274    }
275
276    // RFC 6749 §5.2: a 400 with an OAuth `error` code (`invalid_grant`) means the
277    // refresh token is permanently dead. Anything else is transient and must
278    // never lead the caller to delete credentials.
279    let status = response.status();
280    let obsolete_client = status == StatusCode::UPGRADE_REQUIRED;
281    let meta = ResponseMeta::of(response.headers());
282    let body = read_text(response, &url)?;
283    if status == StatusCode::BAD_REQUEST {
284        if let Some(error) = parse_oauth_error_code(&body) {
285            return Err(Box::new(MoocClientError::RefreshTokenRejected { error }));
286        }
287    }
288    Err(Box::new(MoocClientError::HttpError {
289        url,
290        status,
291        error: body,
292        obsolete_client,
293        message_key: None,
294        meta,
295    }))
296}

298/// Builds a URL under the backend's main-frontend OAuth mount.
299fn oauth_url(root_url: &Url, endpoint: &str) -> MoocClientResult<Url> {
300    root_url
301        .join("/api/v0/main-frontend/oauth/")
302        .and_then(|u| u.join(endpoint))
303        .map_err(|e| Box::new(MoocClientError::UrlParse(endpoint.to_string(), e)))
304}

306fn parse_token(response: reqwest::blocking::Response, url: &Url) -> MoocClientResult<api::Token> {
307    let bytes = read_bytes(response, url)?;
308    serde_json::from_slice(&bytes).map_err(|e| {
309        Box::new(MoocClientError::DeserializingResponse {
310            url: url.clone(),
311            error: Box::new(e),
312        })
313    })
314}
315
316fn read_bytes(response: reqwest::blocking::Response, url: &Url) -> MoocClientResult<bytes::Bytes> {
317    response.bytes().map_err(|e| {
318        Box::new(MoocClientError::ReadingResponseBody {
319            method: Method::POST,
320            url: url.clone(),
321            error: Box::new(e),
322        })
323    })
324}
325
326fn read_text(response: reqwest::blocking::Response, url: &Url) -> MoocClientResult<String> {
327    response.text().map_err(|e| {
328        Box::new(MoocClientError::ReadingResponseBody {
329            method: Method::POST,
330            url: url.clone(),
331            error: Box::new(e),
332        })
333    })
334}

336/// Maps a non-success response to an [`MoocClientError::HttpError`], flagging
337/// `426 Upgrade Required` as an obsolete client (as the resource API does).
338fn http_error(response: reqwest::blocking::Response, url: Url) -> Box<MoocClientError> {
339    let status = response.status();
340    let obsolete_client = status == StatusCode::UPGRADE_REQUIRED;
341    let meta = ResponseMeta::of(response.headers());
342    let error = response.text().unwrap_or_default();
343    Box::new(MoocClientError::HttpError {
344        url,
345        status,
346        error,
347        obsolete_client,
348        message_key: None,
349        meta,
350    })
351}

353/// The single field of an RFC 6749 / RFC 8628 error response the poll loop
354/// needs. Parsed leniently so a non-conforming body yields `None`.
355#[derive(Deserialize)]
356struct OAuthErrorBody {
357    #[serde(default)]
358    error: Option<String>,
359}
360
361fn parse_oauth_error_code(body: &str) -> Option<String> {
362    serde_json::from_str::<OAuthErrorBody>(body)
363        .ok()
364        .and_then(|b| b.error)
365}
