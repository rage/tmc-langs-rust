30/// The chapter an exercise belongs to, for grouping a course's exercises the way its material
31/// does.
32#[derive(Debug, Serialize, Deserialize, JsonSchema)]
33#[cfg_attr(feature = "ts-rs", derive(TS))]
34pub struct ExerciseChapter {
35    pub id: Uuid,
36    pub name: String,
37    /// The chapter's position in the course, from 1.
38    pub chapter_number: i32,
39    /// The user's status in the course's chapter locking; submissions are refused unless it is
40    /// `Unlocked`. `None` when the course does not lock chapters, or the host does not report it.
41    #[serde(default)]
42    pub locking_status: Option<ChapterLockingStatus>,
43}

45impl From<api::ExerciseChapter> for ExerciseChapter {

47        Self {
48            id: value.id,
49            name: value.name,

51            locking_status: value

54        }
55    }
56}

58/// Where the user is in a course's chapter locking, which opens chapters one at a time as the
59/// previous ones are completed.
60#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
61#[cfg_attr(feature = "ts-rs", derive(TS))]
62pub enum ChapterLockingStatus {
63    /// The chapter's exercises can be submitted to.
64    Unlocked,
65    /// The user has completed the chapter and locked it; its exercises take no more submissions.
66    CompletedAndLocked,
67    /// The previous chapters must be completed first.
68    NotUnlockedYet,
69}
70
71impl ChapterLockingStatus {
72    /// `None` for a status newer than this client.
73    fn from_api(value: api::ChapterLockingStatus) -> Option<Self> {
74        match value {
75            api::ChapterLockingStatus::Unlocked => Some(Self::Unlocked),
76            api::ChapterLockingStatus::CompletedAndLocked => Some(Self::CompletedAndLocked),
77            api::ChapterLockingStatus::NotUnlockedYet => Some(Self::NotUnlockedYet),
78            api::ChapterLockingStatus::Unrecognized => None,
79        }
80    }
81}

83impl TmcExerciseSlide {
84    /// Stub archive download URL for this slide's first editor task, if any.
85    /// `None` for browser exercises, which expose no downloadable archive.
86    pub fn editor_stub_download_url(&self) -> Option<&str> {
87        self.tasks
88            .iter()
89            .filter_map(|task| task.public_spec.as_ref())
90            .find_map(|spec| spec.editor_stub_download_url())
91    }
92
93    /// Task id of this slide's first editor task (the submit target), if any.
94    pub fn editor_task_id(&self) -> Option<Uuid> {
95        self.editor_task().map(|task| task.task_id)
96    }
97
98    /// Checksum of this slide's first editor task, if any; compared with the
99    /// stored one to detect an out-of-date local exercise.
100    pub fn editor_checksum(&self) -> Option<&str> {
101        self.editor_task().and_then(|task| task.checksum.as_deref())
102    }
103
104    /// This slide's first editor task, the one a native client works on. Browser tasks do not count.
105    fn editor_task(&self) -> Option<&TmcExerciseTask> {
106        self.tasks.iter().find(|task| {
107            task.public_spec
108                .as_ref()
109                .and_then(|spec| spec.editor_stub_download_url())
110                .is_some()
111        })
112    }
113}

115impl TryFrom<api::ExerciseSlide> for TmcExerciseSlide {
116    type Error = JsonError;
117    fn try_from(value: api::ExerciseSlide) -> Result<Self, Self::Error> {
118        let slide = Self {
119            slide_id: value.slide_id,
120            exercise_id: value.exercise_id,
121            course_id: value.course_id,
122            exercise_name: value.exercise_name,
123            exercise_order_number: value.exercise_order_number,
124            deadline: value.deadline,
125            tasks: value
126                .tasks
127                .into_iter()
128                .map(TryFrom::try_from)
129                .collect::<Result<_, _>>()?,
130            page_url: value.page_url,
131            chapter: value.chapter.map(Into::into),
132        };
133        Ok(slide)
134    }
135}

138#[cfg_attr(feature = "ts-rs", derive(TS))]
139pub struct TmcExerciseTask {
140    pub task_id: Uuid,
141    pub order_number: i32,
142    pub assignment: serde_json::Value,
143    pub public_spec: Option<PublicSpec>,
144    pub model_solution_spec: Option<ModelSolutionSpec>,
145    pub checksum: Option<String>,
146}
147
148impl TryFrom<api::ExerciseTask> for TmcExerciseTask {
149    type Error = JsonError;
150    fn try_from(value: api::ExerciseTask) -> Result<Self, Self::Error> {
151        let public_spec: Option<PublicSpec> = value
152            .public_spec
153            .map(deserialize::json_from_value)
154            .transpose()?;
155        let task = Self {
156            task_id: value.task_id,
157            order_number: value.order_number,
158            assignment: value.assignment,
159            checksum: public_spec.as_ref().map(|ps| ps.checksum.clone()),
160            public_spec,
161            model_solution_spec: value
162                .model_solution_spec
163                .map(deserialize::json_from_value)
164                .transpose()?,
165        };
166        Ok(task)
167    }
168}

183    archive_name: String,

191
192impl PublicSpec {
193    pub fn exercise_type(&self) -> &ExerciseType {
194        &self.exercise_type
195    }
196
197    pub fn stub_download_url(&self) -> &str {
198        &self.stub_download_url
199    }
200
201    /// Stub archive download URL; `None` for browser exercises, which have no project archive.
202    pub fn editor_stub_download_url(&self) -> Option<&str> {
203        match self.exercise_type {
204            ExerciseType::Editor => Some(&self.stub_download_url),
205            ExerciseType::Browser => None,
206        }
207    }

347    /// Guards the deserialize side, which the serialize-only bindings drift gate
348    /// can't catch: parses a mixed editor+browser slide, then round-trips it.
349    ///
350    /// The conversion has a hand-written mirror in
351    /// `tmc-vscode/src/test/moocMockReconciliation.test.ts` (`toCliStdoutSlide`),
352    /// which is what reconciles the backend's wire shape with the CLI's stdout
353    /// schema. Neither repo's CI can see the other, so a change to the `TryFrom`
354    /// impls above has to land there in the same change, or that file goes on
355    /// reconciling a conversion that no longer exists.
356    #[test]
357    fn slide_with_mixed_tasks_round_trips() {
358        let editor_task_id = Uuid::new_v4();
359        let browser_task_id = Uuid::new_v4();
360        let slide_json = serde_json::json!({
361            "slide_id": Uuid::new_v4(),
362            "exercise_id": Uuid::new_v4(),
363            "course_id": Uuid::new_v4(),
364            "exercise_name": "mixed",
365            "exercise_order_number": 3,
366            "deadline": null,
367            "tasks": [
368                {
369                    "task_id": editor_task_id,
370                    "order_number": 0,
371                    "assignment": [{"type": "paragraph"}],
372                    "public_spec": {
373                        "type": "editor",
374                        "archive_name": "stub.tar.zst",
375                        "stub_download_url": "http://example.com/e.tar.zst",
376                        "student_file_paths": ["src/main.py"],
377                        "checksum": "editorsum"
378                    },
379                    "model_solution_spec": { "type": "editor", "solution_download_url": "http://example.com/sol" },
380                    "exercise_service_slug": "tmc"
381                },
382                {
383                    "task_id": browser_task_id,
384                    "order_number": 1,
385                    "assignment": [],
386                    "public_spec": {
387                        "type": "browser",
388                        "archive_name": "stub.tar.zst",
389                        "stub_download_url": "http://example.com/b.tar.zst",
390                        "student_file_paths": [],
391                        "checksum": "browsersum",
392                        "browser_test": { "runtime": "python", "script": "print(1)" }
393                    },
394                    "model_solution_spec": null,
395                    "exercise_service_slug": "tmc"
396                }
397            ]
398        });
399
400        let api_slide: api::ExerciseSlide = serde_json::from_value(slide_json).unwrap();
401        let slide: TmcExerciseSlide = api_slide.try_into().unwrap();
402        assert_eq!(slide.tasks.len(), 2);
403        assert_eq!(slide.tasks[0].task_id, editor_task_id);
404        // checksum is derived from the public spec
405        assert_eq!(slide.tasks[0].checksum.as_deref(), Some("editorsum"));
406        assert_eq!(
407            slide.tasks[0].public_spec.as_ref().unwrap().exercise_type(),
408            &ExerciseType::Editor
409        );
410        assert_eq!(
411            slide.editor_stub_download_url(),
412            Some("http://example.com/e.tar.zst")
413        );
414        assert_eq!(
415            slide.tasks[0]
416                .model_solution_spec
417                .as_ref()
418                .map(|spec| spec.solution_download_url()),
419            Some("http://example.com/sol")
420        );
421
422        let serialized = serde_json::to_string(&slide).unwrap();
423        let reparsed: TmcExerciseSlide = serde_json::from_str(&serialized).unwrap();
424        assert_eq!(reparsed.tasks.len(), 2);
425        assert_eq!(reparsed.tasks[1].task_id, browser_task_id);
426        assert_eq!(
427            reparsed.tasks[1]
428                .public_spec
429                .as_ref()
430                .unwrap()
431                .exercise_type(),
432            &ExerciseType::Browser
433        );
434    }
