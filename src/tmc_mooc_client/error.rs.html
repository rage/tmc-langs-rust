11/// Replaces every query-string value, keeping the keys and the rest of the URL.
12///
13/// File URLs carry a signed download capability in the query, so URLs must be redacted
14/// before reaching logs or errors. Positional rather than keyed on the parameter name,
15/// which the backend owns and could change, failing open. Takes `&str` so it also covers
16/// [`MoocClientError::UrlParse`], whose input did not parse as a URL.
17pub(crate) fn redact_query(url: &str) -> String {
18    const REDACTED: &str = "<redacted>";
19    let Some((base, query)) = url.split_once('?') else {
20        return url.to_string();
21    };
22    let mut redacted = String::with_capacity(base.len() + query.len());
23    redacted.push_str(base);
24    redacted.push('?');
25    for (index, pair) in query.split('&').enumerate() {
26        if index > 0 {
27            redacted.push('&');
28        }
29        match pair.split_once('=') {
30            Some((key, _)) => {
31                redacted.push_str(key);
32                redacted.push('=');
33                redacted.push_str(REDACTED);
34            }
35            // A valueless parameter is itself the value.
36            None => redacted.push_str(REDACTED),
37        }
38    }
39    redacted
40}

42/// What a backend's error body says to a person: an `ApiErrorResponse` `message` or an OAuth
43/// `error_description`/`error`, else the body itself when it is fit to quote.
44pub(crate) fn response_detail(body: &str) -> Option<String> {
45    if let Ok(serde_json::Value::Object(fields)) = serde_json::from_str(body) {
46        let detail = ["message", "error_description", "error"]
47            .into_iter()
48            .filter_map(|key| fields.get(key)?.as_str())
49            .map(str::trim)
50            .find(|text| !text.is_empty());
51        if let Some(detail) = detail {
52            return Some(detail.to_string());
53        }
54    }
55    quotable_body(body)
56}

58/// What an error response says besides its status and body.
59#[derive(Debug, Clone, Default, PartialEq, Eq)]
60pub struct ResponseMeta {
61    /// The `request-id` courses.mooc.fi stamps on every response and logs the request under.
62    pub request_id: Option<String>,
63    /// From `Retry-After`, in seconds; only the delta-seconds form is read.
64    pub retry_after_secs: Option<u64>,
65}
66
67impl ResponseMeta {
68    pub(crate) fn of(headers: &reqwest::header::HeaderMap) -> Self {
69        let header = |name: &str| headers.get(name).and_then(|value| value.to_str().ok());
70        Self {
71            request_id: header("request-id").map(str::to_string),
72            retry_after_secs: header("retry-after").and_then(|value| value.trim().parse().ok()),
73        }
74    }
75}

77/// Why a request got no response at all.
78#[derive(Debug, Clone, Copy, PartialEq, Eq)]
79pub enum ConnectionFailure {
80    /// The server did not answer in time.
81    Timeout,
82    /// TLS failed, typically an untrusted certificate (an intercepting proxy or security
83    /// product) or a wrong system clock.
84    Certificate,
85    /// A proxy refused the connection or asked for credentials.
86    Proxy,
87    /// Anything else: no network, DNS failure, connection refused.
88    Unreachable,
89}
90
91impl ConnectionFailure {
92    pub fn of(error: &reqwest::Error) -> Self {
93        if error.is_timeout() {
94            return Self::Timeout;
95        }
96        let mut causes = Vec::new();
97        let mut source: Option<&(dyn Error + 'static)> = error.source();
98        while let Some(cause) = source {
99            causes.push(cause.to_string().to_lowercase());
100            source = cause.source();
101        }
102        Self::from_causes(&causes)
103    }
104
105    // Split out because a `reqwest::Error` cannot be built in a test.
106    fn from_causes(causes: &[String]) -> Self {
107        let mentions = |needles: &[&str]| {
108            causes
109                .iter()
110                .any(|cause| needles.iter().any(|needle| cause.contains(needle)))
111        };
112        if mentions(&["certificate", "tls", "ssl", "handshake"]) {
113            Self::Certificate
114        } else if mentions(&["proxy", "407"]) {
115            Self::Proxy
116        } else if mentions(&["timed out", "timeout"]) {
117            Self::Timeout
118        } else {
119            Self::Unreachable
120        }
121    }
122}

124#[derive(Debug, Error)]
125pub enum MoocClientError {
126    #[error(
127        "{}",
128        describe_http_error(status, &redact_query(url.as_str()), response_detail(error).as_deref(), *obsolete_client)
129    )]
130    HttpError {
131        url: Url,
132        status: StatusCode,
133        /// The raw response body.
134        error: String,
135        obsolete_client: bool,
136        /// `message_key` of the backend's `ApiErrorResponse` body, if it had that shape;
137        /// lets the CLI map controlled errors (e.g. `not_enrolled`) to a typed `Kind`.
138        message_key: Option<String>,
139        meta: ResponseMeta,
140    },
141    /// A file download answered 404: the exercise's or submission's files are gone from the
142    /// backend's file store.
143    #[error("The exercise's files are missing on courses.mooc.fi ({})", redact_query(url.as_str()))]
144    FileNotFound { url: Url },
145    #[error("Connection error trying to {} {}", .0, redact_query(.1.as_str()))]
146    ConnectionError(Method, Url, #[source] reqwest::Error),
147    #[error("Failed to parse as URL: {}", redact_query(.0))]
148    UrlParse(String, #[source] url::ParseError),
149    #[error(
150        "Refusing to use an insecure URL scheme for {}: courses.mooc.fi must be reached over \
151         https so the bearer token is never sent in plaintext",
152        redact_query(url.as_str())
153    )]
154    InsecureScheme { url: Url },
155    /// A 401 from the resource API: the token is missing, expired or revoked.
156    #[error("Authentication required{}", detail.as_ref().map(|detail| format!(": {detail}")).unwrap_or_default())]
157    NotAuthenticated {
158        detail: Option<String>,
159        meta: ResponseMeta,
160    },
161    #[error("The device authorization request expired before it was approved")]
162    DeviceCodeExpired,
163    #[error("The device authorization request was denied")]
164    DeviceAccessDenied,
165    /// The token endpoint rejected a refresh grant with an OAuth error (a 400 with an
166    /// `error` code, e.g. `invalid_grant`): the refresh token is revoked or expired.
167    /// Permanent, so stored credentials should be discarded, unlike the transient
168    /// [`Self::HttpError`] / [`Self::ConnectionError`], which must not delete them.
169    #[error("The authorization server rejected the refresh token: {error}")]
170    RefreshTokenRejected { error: String },
171    #[error("Failed to attach file to submission form: {error}")]
172    AttachFileToForm { error: Box<dyn Error + Send + Sync> },
173    #[error(
174        "Failed to read {method} response body from {}: {error}.",
175        redact_query(url.as_str())
176    )]
177    ReadingResponseBody {
178        method: Method,
179        url: Url,
180        error: Box<dyn Error + Send + Sync>,
181    },
182    #[error("Failed to deserialize response body from {}: {error}.", redact_query(url.as_str()))]
183    DeserializingResponse {
184        url: Url,
185        error: Box<dyn Error + Send + Sync>,
186    },
187    #[error(transparent)]
188    JsonError(#[from] tmc_langs_util::JsonError),
189    #[error(
190        "Exercise {exercise_id} has no downloadable editor task; browser exercises \
191         have no project archive to download"
192    )]
193    NoDownloadableExerciseTask { exercise_id: Uuid },
194    #[error(
195        "Exercise {exercise_id} has no submittable editor task; browser exercises \
196         are solved in-browser and cannot be submitted from a native client"
197    )]
198    NoSubmittableExerciseTask { exercise_id: Uuid },
199    #[error(
200        "Exercise {exercise_id} has more than one editor task; answer it in the course material"
201    )]
202    SeveralEditorTasks { exercise_id: Uuid },
203}

205impl MoocClientError {
206    pub fn request_id(&self) -> Option<&str> {
207        match self {
208            Self::HttpError { meta, .. } | Self::NotAuthenticated { meta, .. } => {
209                meta.request_id.as_deref()
210            }
211            _ => None,
212        }
213    }
214
215    pub fn retry_after_secs(&self) -> Option<u64> {
216        match self {
217            Self::HttpError { meta, .. } => meta.retry_after_secs,
218            _ => None,
219        }
220    }
221
222    pub fn connection_failure(&self) -> Option<ConnectionFailure> {
223        match self {
224            Self::ConnectionError(_, _, error) => Some(ConnectionFailure::of(error)),
225            _ => None,
226        }
227    }
228}

234    #[test]
235    fn connection_failures_are_told_apart_by_their_cause() {
236        let of = |cause: &str| ConnectionFailure::from_causes(&[cause.to_lowercase()]);
237        assert_eq!(
238            of("invalid peer certificate: UnknownIssuer"),
239            ConnectionFailure::Certificate
240        );
241        assert_eq!(
242            of("unsuccessful tunnel: 407 Proxy Authentication Required"),
243            ConnectionFailure::Proxy
244        );
245        assert_eq!(of("operation timed out"), ConnectionFailure::Timeout);
246        assert_eq!(
247            of("dns error: failed to lookup address"),
248            ConnectionFailure::Unreachable
249        );
250    }
251
252    #[test]
253    fn redact_query_leaves_a_url_without_a_query_alone() {
254        assert_eq!(
255            redact_query("https://courses.mooc.fi/api/v0/files/claimed/an-id"),
256            "https://courses.mooc.fi/api/v0/files/claimed/an-id"
257        );
258    }
259
260    #[test]
261    fn redact_query_replaces_every_value_and_keeps_every_key() {
262        // Positional, so a renamed claim parameter is still caught.
263        assert_eq!(
264            redact_query("https://host/f?download-claim=secret&renamed-later=secret&bare"),
265            "https://host/f?download-claim=<redacted>&renamed-later=<redacted>&<redacted>"
