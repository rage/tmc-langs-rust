1//! User-facing wording of a backend's error response, shared by both backend clients.
2
3use std::fmt::Display;
4
5/// Longest response body quoted in an error message, in characters.
6const MAX_QUOTED_BODY_CHARS: usize = 300;
7
8/// The sentence a failed request reads as: `"{status} from {url}"`, then `": {detail}"` when
9/// the backend said something, then a note when the client is too old for the backend.
10///
11/// `url` must already be safe to show (see the mooc client's query redaction).
12pub fn describe_http_error(
13    status: impl Display,
14    url: &str,
15    detail: Option<&str>,
16    obsolete_client: bool,
17) -> String {
18    let mut message = format!("{status} from {url}");
19    if let Some(detail) = detail {
20        message.push_str(": ");
21        message.push_str(detail);
22    }
23    if obsolete_client {
24        if !message.ends_with(['.', '!', '?']) {
25            message.push('.');
26        }
27        message.push_str(" This client version is no longer supported; update it.");
28    }
29    message
30}
31
32/// A response body fit to quote in an error message: trimmed and shortened, or `None` when it
33/// is empty or an HTML page (a proxy's or web server's error page says nothing useful).
34pub fn quotable_body(body: &str) -> Option<String> {
35    let body = body.trim();
36    if body.is_empty() || body.starts_with('<') {
37        return None;
38    }
39    let mut chars = body.chars();
40    let mut quoted: String = chars.by_ref().take(MAX_QUOTED_BODY_CHARS).collect();
41    if chars.next().is_some() {
42        quoted.push('…');
43    }
44    Some(quoted)
45}

47#[cfg(test)]
48#[allow(clippy::unwrap_used)]
49mod test {
50    use super::*;
51
52    #[test]
53    fn leaves_out_an_absent_detail() {
54        assert_eq!(
55            describe_http_error("404 Not Found", "https://host/x", None, false),
56            "404 Not Found from https://host/x"
57        );
58    }
59
60    #[test]
61    fn appends_the_detail() {
62        assert_eq!(
63            describe_http_error(
64                "422 Unprocessable Entity",
65                "https://host/x",
66                Some("Exercise deadline passed."),
67                false
68            ),
69            "422 Unprocessable Entity from https://host/x: Exercise deadline passed."
70        );
71    }
72
73    #[test]
74    fn mentions_an_obsolete_client_only_when_it_is_one() {
75        assert_eq!(
76            describe_http_error(
77                "426 Upgrade Required",
78                "https://host/x",
79                Some("Too old."),
80                true
81            ),
82            "426 Upgrade Required from https://host/x: Too old. \
83             This client version is no longer supported; update it."
84        );
85        assert!(!describe_http_error("400 Bad Request", "u", None, false).contains("client"));
86    }
87
88    #[test]
89    fn quotable_body_drops_empty_and_html_bodies() {
90        assert_eq!(quotable_body("  \n"), None);
91        assert_eq!(quotable_body("<html><body>404</body></html>"), None);
92        assert_eq!(
93            quotable_body(" plain text \n"),
94            Some("plain text".to_string())
95        );
96    }
97
98    #[test]
99    fn quotable_body_shortens_a_long_body() {
100        let quoted = quotable_body(&"x".repeat(1000)).unwrap();
101        assert_eq!(quoted.chars().count(), MAX_QUOTED_BODY_CHARS + 1);
102        assert!(quoted.ends_with('…'));
103    }
104}
