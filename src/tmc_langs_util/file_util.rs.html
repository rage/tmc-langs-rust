29#[derive(Debug, Clone, Copy)]
30pub enum LockOptions {
31    /// Shared read lock
32    Read,
33    /// Shared read lock, create file if it doesn't exist instead of erroring (including intermediate directories)
34    ReadCreate,
35    /// Exclusive write lock
36    Write,
37    /// Exclusive write lock, create file if it doesn't exist instead of erroring (including intermediate directories)
38    WriteCreate,
39    /// Exclusive write lock, create file if it doesn't exist, truncate if it does
40    WriteTruncate,
41}
42
43impl LockOptions {
44    // no truncate: O_TRUNC would wipe the file before the lock is held, letting two
45    // racing writers each truncate. `Lock::lock` truncates once it holds the lock.
46    fn into_open_options(self) -> OpenOptions {
47        let mut opts = OpenOptions::new();
48        match self {
49            Self::Read => opts.read(true),
50            // create requires write
51            Self::ReadCreate => opts.read(true).write(true).create(true),
52            Self::Write => opts.write(true),
53            Self::WriteCreate => opts.write(true).create(true),
54            Self::WriteTruncate => opts.write(true).create(true),
55        };
56        opts
57    }
58
59    fn requests_truncate(self) -> bool {
60        matches!(self, Self::WriteTruncate)
61    }
62}

64/// Truncates `file` if `options` requests it. The caller must already hold the lock: truncating at
65/// open time would let two racing writers each wipe the file before either holds it.
66///
67/// `report_path` must name the file actually truncated — for `Lock::dir` the central lock file, not
68/// the locked directory.
69pub(crate) fn truncate_locked_file(
70    options: LockOptions,
71    file: &File,
72    report_path: &Path,
73) -> Result<(), FileError> {
74    if options.requests_truncate() {
75        file.set_len(0)
76            .map_err(|e| FileError::FileWrite(report_path.to_path_buf(), e))?;
77    }
78    Ok(())
79}

81/// Set when the OS reports locking as unsupported and the caller proceeds
82/// unlocked. See [`report_locking_unavailable`].
83static LOCKING_UNAVAILABLE: AtomicBool = AtomicBool::new(false);
84
85/// Whether any lock in this process has fallen back to running *unlocked* because
86/// the OS refused it.
87///
88/// For callers that need the lock for correctness (the mooc credentials refresh:
89/// unserialized refreshes redeem one refresh token twice and the backend revokes
90/// the session). Sticky and process-wide, since the cause is the filesystem.
91pub fn locking_unavailable() -> bool {
92    LOCKING_UNAVAILABLE.load(Ordering::Relaxed)
93}
94
95/// Records that the caller is proceeding *without* a lock, warning once per
96/// process. Returns whether this was the first report (and so emitted the warning).
97///
98/// Failing open is deliberate, since some filesystems cannot lock at all. The usual
99/// cause is a home or config directory on NFS with no reachable `rpc.lockd`, where
100/// `fcntl` fails with `ENOLCK`.
101pub fn report_locking_unavailable(path: &Path, err: &dyn Display) -> bool {
102    if LOCKING_UNAVAILABLE.swap(true, Ordering::Relaxed) {
103        // Debug level so a run where every lock fails doesn't flood the log.
104        log::debug!("Failed to lock {} again: {err}", path.display());
105        return false;
106    }
107    log::warn!(
108        "Failed to lock {}: {err}. Continuing WITHOUT the lock, so operations \
109         that rely on it are no longer serialized between processes. This \
110         usually means the filesystem does not support file locking -- most \
111         often a home or config directory on NFS with no lock daemon \
112         (rpc.lockd/NLM) reachable. Concurrent commands can then corrupt each \
113         other's writes, and two of them refreshing the login token at once \
114         will invalidate it.",
115        path.display()
116    );
117    true
118}

120/// Sleep between acquisition attempts in [`with_file_lock_timeout`].
121const LOCK_POLL_INTERVAL: Duration = Duration::from_millis(50);
122
123/// Runs `f` while holding a lock on the file at `path`, giving up with
124/// [`FileError::LockTimeout`] if the lock cannot be taken within `timeout`.
125///
126/// Unlike [`Lock::lock`], which blocks forever behind a wedged holder, this polls
127/// a non-blocking acquisition so the caller gets a retryable error instead of a hang.
128///
129/// `f` runs with the lock held and its value is returned. A closure is taken
130/// because the guard borrows the internal [`Lock`] created per loop iteration.
131pub fn with_file_lock_timeout<T>(
132    path: impl AsRef<Path>,
133    options: LockOptions,
134    timeout: Duration,
135    f: impl FnOnce(&mut Guard<'_>) -> T,
136) -> Result<T, FileError> {
137    let path = path.as_ref();
138    let deadline = Instant::now() + timeout;
139    loop {
140        let mut lock = Lock::file(path, options)?;
141        if let Some(mut guard) = lock.try_lock()? {
142            return Ok(f(&mut guard));
143        }
144        // The deadline is checked after an attempt, so a zero timeout is a plain try-lock.
145        if Instant::now() >= deadline {
146            log::warn!(
147                "Gave up after {timeout:?} waiting for a lock on {}",
148                path.display()
149            );
150            return Err(FileError::LockTimeout {
151                path: path.to_path_buf(),
152                timeout,
153            });
154        }
155        std::thread::sleep(LOCK_POLL_INTERVAL);
156    }
157}

301/// Copies the file or directory at source into the target path.
302/// If the source is a file and the target is not a directory, the source file is copied to the target path.
303/// If the source is a file and the target is a directory, the source file is copied into the target directory.
304/// If the source is a directory and the target is not a file, the source directory and all files in it are copied recursively into the target directory. For example, with source=dir1 and target=dir2, dir1/file would be copied to dir2/dir1/file.
305/// If the source is a directory and the target is a file, an error is returned.
306pub fn copy<P: AsRef<Path>, Q: AsRef<Path>>(source: P, target: Q) -> Result<(), FileError> {
307    let source = source.as_ref();
308    let target = target.as_ref();
309
310    if source.is_file() {
311        if target.is_dir() {
312            log::trace!(
313                "copying into dir {} -> {}",
314                source.display(),
315                target.display()
316            );
317            let file_name = if let Some(file_name) = source.file_name() {
318                file_name
319            } else {
320                return Err(FileError::NoFileName(source.to_path_buf()));
321            };
322            let path_in_target = target.join(file_name);
323            std::fs::copy(source, path_in_target).map_err(|e| FileError::FileCopy {
324                from: source.to_path_buf(),
325                to: target.to_path_buf(),
326                source: e,
327            })?;
328        } else {
329            log::trace!("copying file {} -> {}", source.display(), target.display());
330            if let Some(parent) = target.parent() {
331                if !parent.exists() {
332                    create_dir_all(parent)?;
333                }
334            }
335            std::fs::copy(source, target).map_err(|e| FileError::FileCopy {
336                from: source.to_path_buf(),
337                to: target.to_path_buf(),
338                source: e,
339            })?;
340        }
341    } else {
342        log::trace!(
343            "recursively copying {} -> {}",
344            source.display(),
345            target.display()
346        );
347        if target.is_file() {
348            return Err(FileError::UnexpectedFile(target.to_path_buf()));
349        } else {
350            let prefix = source.parent().unwrap_or_else(|| Path::new(""));
351            for entry in WalkDir::new(source) {
352                let entry = entry?;
353                let entry_path = entry.path();
354                let stripped = entry_path
355                    .strip_prefix(prefix)
356                    .expect("prefix is derived from the source which entry_path is in");
357
358                let target = target.join(stripped);
359                if entry_path.is_dir() {
360                    create_dir_all(target)?;
361                } else {
362                    if let Some(parent) = target.parent() {
363                        create_dir_all(parent)?;
364                    }
365                    std::fs::copy(entry_path, &target).map_err(|e| FileError::FileCopy {
366                        from: entry_path.to_path_buf(),
367                        to: target.clone(),
368                        source: e,
369                    })?;
370                }
371            }
372        }
373    }
374    Ok(())
375}
