15pub struct Input {
16    pub crate_: Path,
17    pub types: Vec<ForeignType>,
18}
19
20impl Parse for Input {
21    fn parse(input: ParseStream) -> parse::Result<Input> {
22        let crate_ = input.parse()?;
23        let mut types = vec![];
24        while !input.is_empty() {
25            types.push(input.parse()?);
26        }
27
28        Ok(Input { crate_, types })
29    }

32pub struct ForeignType {
33    pub attrs: Vec<Attribute>,
34    pub visibility: Visibility,
35    pub name: Ident,
36    pub generics: Generics,
37    pub oibits: Punctuated<Ident, Token![+]>,
38    pub phantom_data: Option<Type>,
39    pub ctype: Type,
40    pub drop: Expr,
41    pub clone: Option<Expr>,
42}
43
44impl Parse for ForeignType {
45    fn parse(input: ParseStream) -> parse::Result<ForeignType> {
46        let attrs = input.call(Attribute::parse_outer)?;
47        let visibility = input.parse()?;
48        input.parse::<Token![unsafe]>()?;
49        input.parse::<Token![type]>()?;
50        let name = input.parse()?;
51        let generics = input.parse()?;
52        let oibits = input.call(parse_oibits)?;
53        let inner;
54        braced!(inner in input);
55        let ctype = inner.call(parse_type::<kw::CType>)?;
56        let phantom_data = inner.call(parse_phantom_data)?;
57        let drop = inner.call(parse_fn::<kw::drop>)?;
58        let clone = inner.call(parse_clone)?;
59
60        Ok(ForeignType {
61            attrs,
62            visibility,
63            name,
64            generics,
65            oibits,
66            ctype,
67            phantom_data,
68            drop,
69            clone,
70        })
71    }
72}

74fn parse_oibit(input: ParseStream) -> parse::Result<Ident> {
75    let lookahead = input.lookahead1();
76    if lookahead.peek(kw::Sync) || lookahead.peek(kw::Send) {
77        input.parse()
78    } else {
79        Err(lookahead.error())
80    }
81}
82
83fn parse_oibits(input: ParseStream) -> parse::Result<Punctuated<Ident, Token![+]>> {
84    let mut out = Punctuated::new();
85
86    if input.parse::<Option<Token![:]>>()?.is_some() {
87        loop {
88            out.push_value(input.call(parse_oibit)?);
89            if input.peek(token::Brace) {
90                break;
91            }
92            out.push_punct(input.parse()?);
93            if input.peek(token::Brace) {
94                break;
95            }
96        }
97    }
98
99    Ok(out)
100}

102fn parse_type<T>(input: ParseStream) -> parse::Result<Type>
103where
104    T: Parse,
105{
106    input.parse::<Token![type]>()?;
107    input.parse::<T>()?;
108    input.parse::<Token![=]>()?;
109    let type_ = input.parse()?;
110    input.parse::<Token![;]>()?;
111    Ok(type_)
112}

114fn parse_phantom_data(input: ParseStream) -> parse::Result<Option<Type>> {
115    if input.peek(Token![type]) && input.peek2(kw::PhantomData) {
116        input.call(parse_type::<kw::PhantomData>).map(Some)
117    } else {
118        Ok(None)
119    }
120}

122fn parse_fn<T>(input: ParseStream) -> parse::Result<Expr>
123where
124    T: Parse,
125{
126    input.parse::<Token![fn]>()?;
127    input.parse::<T>()?;
128    input.parse::<Token![=]>()?;
129    let path = input.parse()?;
130    input.parse::<Token![;]>()?;
131    Ok(path)
132}
133
134fn parse_clone(input: ParseStream) -> parse::Result<Option<Expr>> {
135    if input.peek(Token![fn]) && input.peek2(kw::clone) {
136        input.call(parse_fn::<kw::clone>).map(Some)
137    } else {
138        Ok(None)
139    }
