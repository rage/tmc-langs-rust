5/// Represents a position inside some textual document.
6#[derive(Copy, Clone, PartialEq, Eq)]
7pub struct TextPosition {
8    #[doc(hidden)]
9    pub row: u64,
10
11    #[doc(hidden)]
12    pub column: u64,
13}
14
15impl TextPosition {
16    /// Creates a new position initialized to the beginning of the document
17    #[inline]
18    #[must_use]
19    pub const fn new() -> Self {
20        Self { row: 0, column: 0 }
21    }
22
23    /// Advances the position in a line
24    #[inline]
25    pub fn advance(&mut self, count: u8) {
26        self.column += u64::from(count);
27    }
28
29    #[doc(hidden)]
30    #[deprecated]
31    pub fn advance_to_tab(&mut self, width: u8) {
32        let width = u64::from(width);
33        self.column += width - self.column % width;
34    }
35
36    /// Advances the position to the beginning of the next line
37    #[inline]
38    pub fn new_line(&mut self) {
39        self.column = 0;
40        self.row += 1;
41    }
42
43    /// Row, counting from 0. Add 1 to display as users expect!
44    #[must_use]
45    pub fn row(&self) -> u64 {
46        self.row
47    }
48
49    /// Column, counting from 0. Add 1 to display as users expect!
50    #[must_use]
51    pub fn column(&self) -> u64 {
52        self.column
53    }
54}
55
56impl fmt::Debug for TextPosition {
57    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
58        fmt::Display::fmt(self, f)
59    }
60}
61
62impl fmt::Display for TextPosition {
63    #[inline]
64    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
65        write!(f, "{}:{}", self.row + 1, self.column + 1)
66    }
67}

69/// Get the position in the document corresponding to the object
70///
71/// This trait is implemented by parsers, lexers and errors.
72pub trait Position {
73    /// Returns the current position or a position corresponding to the object.
74    fn position(&self) -> TextPosition;
75}

77impl Position for TextPosition {
78    #[inline]
79    fn position(&self) -> TextPosition {
80        *self
81    }

84/// XML version enumeration.
85#[derive(Copy, Clone, PartialEq, Eq, PartialOrd, Ord)]
86pub enum XmlVersion {
87    /// XML version 1.0, or any 1.x version other than 1.1
88    ///
89    /// All future versions are disallowed since XML 1.1, so any version beyond 1.1 is an error tolerated only in XML 1.0.
90    /// <https://www.w3.org/TR/REC-xml/#sec-prolog-dtd>
91    Version10,
92
93    /// XML version 1.1.
94    Version11,
95}
96
97impl XmlVersion {
98    /// Convenience helper which returns a string representation of the given version.
99    ///
100    /// ```
101    /// # use xml::common::XmlVersion;
102    /// assert_eq!(XmlVersion::Version10.as_str(), "1.0");
103    /// assert_eq!(XmlVersion::Version11.as_str(), "1.1");
104    /// ```
105    #[must_use]
106    pub fn as_str(self) -> &'static str {
107        match self {
108            Self::Version10 => "1.0",
109            Self::Version11 => "1.1",
110        }
111    }
112}
113
114impl fmt::Display for XmlVersion {
115    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
116        self.as_str().fmt(f)
117    }
118}
119
120impl fmt::Debug for XmlVersion {
121    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
122        fmt::Display::fmt(self, f)
123    }
124}

126/// Checks whether the given character is a white space character (`S`)
127/// as is defined by XML 1.1 specification, [section 2.3][1].
128///
129/// [1]: http://www.w3.org/TR/2006/REC-xml11-20060816/#sec-common-syn
130#[must_use]
131#[inline]
132pub const fn is_whitespace_char(c: char) -> bool {
133    matches!(c, '\x20' | '\x0a' | '\x09' | '\x0d')
134}
135
136/// Matches the `PubIdChar` production.
137pub(crate) fn is_pubid_char(c: char) -> bool {
138    matches!(c, '\x20' | '\x0D' | '\x0A' | 'a'..='z' | 'A'..='Z' | '0'..='9' |
139        '-' | '\'' | '(' | ')' | '+' | ',' | '.' | '/' | ':' | '=' | '?' | ';' |
140        '!' | '*' | '#' | '@' | '$' | '_' | '%')
141}
142
143/// Checks whether the given string is compound only by white space
144/// characters (`S`) using the previous `is_whitespace_char` to check
145/// all characters of this string
146pub fn is_whitespace_str(s: &str) -> bool {
147    s.chars().all(is_whitespace_char)
148}
149
150/// Is it a valid character in XML 1.0
151#[must_use]
152pub const fn is_xml10_char(c: char) -> bool {
153    matches!(c, '\u{09}' | '\u{0A}' | '\u{0D}' | '\u{20}'..='\u{D7FF}' | '\u{E000}'..='\u{FFFD}' | '\u{10000}'..)
154}
155
156/// Is it a valid character in XML 1.1
157#[must_use]
158pub const fn is_xml11_char(c: char) -> bool {
159    matches!(c, '\u{01}'..='\u{D7FF}' | '\u{E000}'..='\u{FFFD}' | '\u{10000}'..)
160}
161
162/// Is it a valid character in XML 1.1 but not part of the restricted character set
163#[must_use]
164pub const fn is_xml11_char_not_restricted(c: char) -> bool {
165    is_xml11_char(c) &&
166        !matches!(c, '\u{01}'..='\u{08}' | '\u{0B}'..='\u{0C}' | '\u{0E}'..='\u{1F}' | '\u{7F}'..='\u{84}' | '\u{86}'..='\u{9F}')
167}

169/// Checks whether the given character is a name start character (`NameStartChar`)
170/// as is defined by XML 1.1 specification, [section 2.3][1].
171///
172/// [1]: http://www.w3.org/TR/2006/REC-xml11-20060816/#sec-common-syn
173#[must_use]
174pub const fn is_name_start_char(c: char) -> bool {
175    matches!(c,
176        ':' | 'A'..='Z' | '_' | 'a'..='z' |
177        '\u{C0}'..='\u{D6}' | '\u{D8}'..='\u{F6}' | '\u{F8}'..='\u{2FF}' |
178        '\u{370}'..='\u{37D}' | '\u{37F}'..='\u{1FFF}' |
179        '\u{200C}'..='\u{200D}' | '\u{2070}'..='\u{218F}' |
180        '\u{2C00}'..='\u{2FEF}' | '\u{3001}'..='\u{D7FF}' |
181        '\u{F900}'..='\u{FDCF}' | '\u{FDF0}'..='\u{FFFD}' |
182        '\u{10000}'..='\u{EFFFF}'
183    )
184}
185
186/// Checks whether the given character is a name character (`NameChar`)
187/// as is defined by XML 1.1 specification, [section 2.3][1].
188///
189/// [1]: http://www.w3.org/TR/2006/REC-xml11-20060816/#sec-common-syn
190#[must_use]
191pub const fn is_name_char(c: char) -> bool {
192    if is_name_start_char(c) {
193        return true;
194    }
195    matches!(c,
196        '-' | '.' | '0'..='9' | '\u{B7}' |
197        '\u{300}'..='\u{36F}' | '\u{203F}'..='\u{2040}'
198    )
