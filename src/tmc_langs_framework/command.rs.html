20/// How long a build or dependency-resolution step (`mvn compile`, `ant compile-test`,
21/// `make test`, `dotnet build`) may run. Below the client's own 10-minute process timeout so
22/// the user gets a build-specific error instead of a generic one.
23pub const BUILD_TIMEOUT: Duration = Duration::from_secs(8 * 60);
24
25/// How long to wait for a command's output pipes to reach EOF once the command has exited or
26/// been killed. A descendant that outlived it may hold the pipes open indefinitely.
27const OUTPUT_DRAIN_GRACE: Duration = Duration::from_secs(5);
28
29/// How long to wait for a killed command to be reaped.
30const KILL_REAP_TIMEOUT: Duration = Duration::from_secs(5);

32/// Wrapper around subprocess::Exec
33#[must_use]
34pub struct TmcCommand {
35    exec: Exec,
36    stdin: Option<String>,
37    program: OsString,
38    /// First argument that could not be passed to a batch file; reported when run.
39    argument_error: Option<io::Error>,
40}

42impl TmcCommand {
43    /// Creates a new command. A bare program name is resolved to an absolute path first; see
44    /// [`tool_lookup::resolve_tool`].
45    pub fn new(cmd: impl Into<OsString>) -> Self {
46        let program = cmd.into();
47        let exec = match tool_lookup::resolve_tool(&program) {
48            Some(ResolvedTool { path, child_path }) => {
49                let exec = Exec::cmd(path);
50                match child_path {
51                    Some(child_path) => exec.env("PATH", child_path),
52                    None => exec,
53                }
54            }
55            None => Exec::cmd(&program),
56        };
57        // Stops cmd.exe (running batch files) from searching the current directory for programs.
58        let exec = exec
59            .env("LANG", "en_US.UTF-8")
60            .env("NoDefaultCurrentDirectoryInExePath", "1");
61        Self {
62            exec,
63            stdin: None,
64            program,
65            argument_error: None,
66        }
67    }
68
69    /// Creates a new command with piped stdout/stderr.
70    pub fn piped(cmd: impl Into<OsString>) -> Self {
71        let command = Self::new(cmd);
72        Self {
73            exec: command
74                .exec
75                .stdout(Redirection::Pipe)
76                .stderr(Redirection::Pipe),
77            ..command
78        }
79    }
80
81    /// Creates a piped command for a batch file such as `mvn.cmd`, passing `args` so that
82    /// `cmd.exe` takes them literally; plain arguments elsewhere.
83    pub fn piped_batch_file<A: AsRef<OsStr>>(
84        script: impl Into<OsString>,
85        args: impl IntoIterator<Item = A>,
86    ) -> Self {
87        let script = script.into();
88        let args: Vec<A> = args.into_iter().collect();
89        #[cfg(windows)]
90        {
91            use subprocess::ExecExt;
92            let mut command = Self::piped(
93                std::env::var_os("COMSPEC").unwrap_or_else(|| OsString::from("cmd.exe")),
94            );
95            let mut line = vec![&script as &OsStr];
96            line.extend(args.iter().map(AsRef::as_ref));
97            match batch_command_line(line) {
98                Ok(line) => {
99                    command.exec = command
100                        .exec
101                        .args(["/d", "/s", "/c"])
102                        .raw_arg(format!("\"{line}\""));
103                }
104                Err(err) => command.argument_error = Some(err),
105            }
106            Self {
107                program: script,
108                ..command
109            }
110        }
111        #[cfg(not(windows))]
112        {
113            Self::piped(script).with(|e| e.args(args.iter().map(AsRef::as_ref)))
114        }
115    }

117    /// Allows modification of the internal command without providing access to it.
118    pub fn with(self, f: impl FnOnce(Exec) -> Exec) -> Self {
119        Self {
120            exec: f(self.exec),
121            ..self
122        }
123    }

127        Self {
128            exec: self.exec.stdin(Redirection::Pipe),

131        }

134    // executes the given command and collects its output
135    fn execute(self, timeout: Option<Duration>, checked: bool) -> Result<Output, TmcError> {
136        let cmd = self.exec.to_cmdline_lossy();
137        log::info!("executing {cmd}");
138
139        let Self {
140            exec,
141            stdin,
142            program,
143            argument_error,
144        } = self;
145        if let Some(err) = argument_error {
146            return Err(CommandError::FailedToRun(cmd, err).into());
147        }
148
149        // Only a command that may be killed needs a group of its own: it keeps a terminal-less
150        // child from being stopped by a background read of the controlling terminal.
151        #[cfg(unix)]
152        let exec = if timeout.is_some() {
153            subprocess::ExecExt::setpgid(exec)
154        } else {
155            exec
156        };
157
158        // starts executing the command
159        let mut popen = exec
160            .start()
161            .map_err(|e| popen_to_tmc_err(cmd.clone(), &program, e))?;
162        let tree = ProcessTree::attach(popen.pid());
163        let stdin_handle = spawn_writer(popen.stdin.take(), stdin);
164        let stdout_reader = spawn_reader(popen.stdout.take());
165        let stderr_reader = spawn_reader(popen.stderr.take());
166
167        let exit_status = if let Some(timeout) = timeout {
168            // timeout set
169            let exit_status = popen
170                .wait_timeout(timeout)
171                .map_err(|e| popen_to_tmc_err(cmd.clone(), &program, e))?;
172
173            match exit_status {
174                Some(exit_status) => exit_status,
175                None => {
176                    // None means that we timed out
177                    kill_process_tree(&popen, &tree)
178                        .map_err(|e| CommandError::Terminate(cmd.clone(), e))?;
179                    let _ = popen.wait_timeout(KILL_REAP_TIMEOUT);
180                    let (stdout, stderr) = collect_output(stdout_reader, stderr_reader);
181                    return Err(TmcError::Command(CommandError::TimeOut {
182                        command: cmd,
183                        timeout,
184                        stdout: String::from_utf8_lossy(&stdout).into_owned(),
185                        stderr: String::from_utf8_lossy(&stderr).into_owned(),
186                    }));
187                }
188            }
189        } else {
190            // no timeout, block until done
191            popen
192                .wait()
193                .map_err(|e| popen_to_tmc_err(cmd.clone(), &program, e))?
194        };
195
196        log::info!("finished executing {cmd}");
197        stdin_handle
198            .join()
199            .expect("the thread should not be able to panic");
200        let (stdout, stderr) = collect_output(stdout_reader, stderr_reader);
201
202        // on success, log stdout trace and stderr debug
203        // on failure if checked, log warn
204        // on failure if not checked, log debug
205        if !exit_status.success() {
206            // if checked is set, error with failed exit status
207            if checked {
208                log::warn!("stdout: {}", String::from_utf8_lossy(&stdout).into_owned());
209                log::warn!("stderr: {}", String::from_utf8_lossy(&stderr).into_owned());
210                return Err(CommandError::Failed {
211                    command: cmd,
212                    status: exit_status,
213                    stdout: String::from_utf8_lossy(&stdout).into_owned(),
214                    stderr: String::from_utf8_lossy(&stderr).into_owned(),
215                }
216                .into());
217            } else {
218                log::debug!("stdout: {}", String::from_utf8_lossy(&stdout).into_owned());
219                log::debug!("stderr: {}", String::from_utf8_lossy(&stderr).into_owned());
220            }
221        } else {
222            log::trace!("stdout: {}", String::from_utf8_lossy(&stdout).into_owned());
223            log::debug!("stderr: {}", String::from_utf8_lossy(&stderr).into_owned());
224        }
225        Ok(Output {
226            status: exit_status.into(),
227            stdout,
228            stderr,
229        })
230    }

258// it's assumed the thread will never panic
259fn spawn_writer(file: Option<File>, data: Option<String>) -> JoinHandle<()> {
260    std::thread::spawn(move || {
261        if let Some(mut file) = file {
262            if let Some(data) = data {
263                log::debug!("writing data");
264                if let Err(err) = file.write_all(data.as_bytes()) {
265                    log::error!("failed to write data in writer thread: {err}");
266                }
267            }
268        }
269    })
270}

272/// Output read from a pipe on a background thread, so that waiting for it can be bounded.
273struct PipeReader {
274    buffer: Arc<Mutex<Vec<u8>>>,
275    finished: mpsc::Receiver<()>,
276}
277
278// it's assumed the thread will never panic
279fn spawn_reader(file: Option<File>) -> PipeReader {
280    let buffer = Arc::new(Mutex::new(vec![]));
281    let (finished_tx, finished) = mpsc::channel();
282    let thread_buffer = Arc::clone(&buffer);
283    std::thread::spawn(move || {
284        if let Some(mut file) = file {
285            let mut chunk = [0u8; 8192];
286            loop {
287                match file.read(&mut chunk) {
288                    Ok(0) => break,
289                    Ok(n) => thread_buffer
290                        .lock()
291                        .unwrap_or_else(|poisoned| poisoned.into_inner())
292                        .extend_from_slice(&chunk[..n]),
293                    Err(err) if err.kind() == io::ErrorKind::Interrupted => {}
294                    Err(err) => {
295                        log::error!("failed to read data in reader thread: {err}");
296                        break;
297                    }
298                }
299            }
300        }
301        let _ = finished_tx.send(());
302    });
303    PipeReader { buffer, finished }
304}
305
306impl PipeReader {
307    /// Everything read so far once the pipe closes, or at `deadline` if something still holds
308    /// it open.
309    fn collect(self, deadline: Instant) -> Vec<u8> {
310        let remaining = deadline.saturating_duration_since(Instant::now());
311        if self.finished.recv_timeout(remaining).is_err() {
312            log::warn!("output pipe still open after the command ended");
313        }
314        std::mem::take(
315            &mut *self
316                .buffer
317                .lock()
318                .unwrap_or_else(|poisoned| poisoned.into_inner()),
319        )
320    }
321}
322
323/// The output of an ended command, waiting at most [`OUTPUT_DRAIN_GRACE`] for both pipes.
324fn collect_output(stdout: PipeReader, stderr: PipeReader) -> (Vec<u8>, Vec<u8>) {
325    let deadline = Instant::now() + OUTPUT_DRAIN_GRACE;
326    (stdout.collect(deadline), stderr.collect(deadline))
327}

329/// Kills the command and everything it started.
330fn kill_process_tree(popen: &subprocess::Job, tree: &ProcessTree) -> io::Result<()> {
331    tree.kill();
332    #[cfg(unix)]
333    {
334        use subprocess::unix::JobExt;
335        const SIGKILL: i32 = 9;
336        if popen.send_signal_group(SIGKILL).is_ok() {
337            return Ok(());
338        }
339    }
340    popen.kill()
341}

343/// `ERROR_ACCESS_DENIED`
344const WINDOWS_ACCESS_DENIED: i32 = 5;
345/// `ERROR_VIRUS_INFECTED` and `ERROR_VIRUS_DELETED`
346const WINDOWS_VIRUS_ERRORS: [i32; 2] = [225, 226];
347
348/// Whether a failure to start `program` looks like antivirus software blocked or removed it:
349/// the OS said so, or on Windows an executable named by path is gone or denied right after being
350/// built. Elsewhere a missing path is a missing tool or shebang interpreter.
351fn is_blocked_executable(program: &OsStr, err: &io::Error) -> bool {
352    if !cfg!(windows) {
353        return false;
354    }
355    if err
356        .raw_os_error()
357        .is_some_and(|code| WINDOWS_VIRUS_ERRORS.contains(&code))
358    {
359        return true;
360    }
361    let is_path = Path::new(program).components().count() > 1;
362    match err.kind() {
363        io::ErrorKind::NotFound => is_path,
364        _ => err.raw_os_error() == Some(WINDOWS_ACCESS_DENIED),
365    }
366}
367
368/// Whether a tool's output carries Windows' `ERROR_VIRUS_INFECTED` message, which `make` and
369/// shells relay when they cannot start a program that antivirus software blocked.
370pub fn output_shows_blocked_executable(output: &str) -> bool {
371    output
372        .to_lowercase()
373        .contains("contains a virus or potentially unwanted software")
374}

376// convenience function to convert an error while checking for command not found error
377fn popen_to_tmc_err(cmd: String, program: &OsStr, err: io::Error) -> TmcError {
378    if is_blocked_executable(program, &err) {
379        return TmcError::Command(CommandError::ExecutableBlocked {
380            command: cmd,
381            source: err,
382        });
383    }
384    if let std::io::ErrorKind::NotFound = err.kind() {
385        let tool = Path::new(program)
386            .file_name()
387            .unwrap_or(program)
388            .to_string_lossy()
389            .into_owned();
390        let hint = tool_lookup::missing_tool_hint(&tool);
391        TmcError::Command(CommandError::NotFound {
392            cmd,
393            tool,
394            hint,
395            source: err,
396        })
397    } else {
398        TmcError::Command(CommandError::FailedToRun(cmd, err))
399    }
400}

402/// `args` as the single command line `cmd.exe /d /s /c "<line>"` should receive, each argument
403/// quoted so that `cmd.exe` metacharacters (`& | ^ < > ( )`) stay literal.
404pub fn batch_command_line<'a>(args: impl IntoIterator<Item = &'a OsStr>) -> io::Result<String> {
405    let mut line = String::new();
406    for (i, arg) in args.into_iter().enumerate() {
407        if i > 0 {
408            line.push(' ');
409        }
410        line.push_str(&quote_batch_arg(arg)?);
411    }
412    Ok(line)
413}

415/// Always quoted. `"` is doubled and `%` is broken up (`%%cd:~,%` expands to `%`) the way Rust's
416/// standard library does for `.bat`/`.cmd` files; line breaks and NULs cannot be quoted.
417fn quote_batch_arg(arg: &OsStr) -> io::Result<String> {
418    let arg = arg.to_str().ok_or_else(|| {
419        io::Error::new(io::ErrorKind::InvalidInput, "argument is not valid UTF-8")
420    })?;
421    if arg.contains(['\0', '\r', '\n']) {
422        return Err(io::Error::new(
423            io::ErrorKind::InvalidInput,
424            "argument contains a NUL or line break",
425        ));
426    }
427    let mut quoted = String::with_capacity(arg.len() + 2);
428    quoted.push('"');
429    let mut backslashes = 0usize;
430    for c in arg.chars() {
431        match c {
432            '\\' => {
433                backslashes += 1;
434                quoted.push('\\');
435                continue;
436            }
437            '"' => {
438                // MSVCRT: backslashes before a quote are doubled and the quote is escaped;
439                // cmd.exe additionally needs the quote doubled.
440                quoted.extend(std::iter::repeat_n('\\', backslashes));
441                quoted.push_str("\"\"");
442            }
443            '%' => quoted.push_str("%%cd:~,%"),
444            c => quoted.push(c),
445        }
446        backslashes = 0;
447    }
448    quoted.extend(std::iter::repeat_n('\\', backslashes));
449    quoted.push('"');
450    Ok(quoted)
451}

460#[derive(Debug)]
461pub struct ExitStatus {
462    /// `None` if killed by a signal or undetermined — both treated as failure.
463    code: Option<i32>,
464}
465
466impl ExitStatus {
467    /// For a process that exited normally with `code`.
468    pub fn new(code: i32) -> Self {
469        Self { code: Some(code) }
470    }
471
472    pub fn success(&self) -> bool {
473        self.code == Some(0)
474    }
475}
476
477impl From<subprocess::ExitStatus> for ExitStatus {
478    fn from(value: subprocess::ExitStatus) -> Self {
479        // Not `signal()`: it is `None` on a normal exit, so every success would read as failure.
480        Self {
481            code: value.code().map(|c| c as i32),
482        }
483    }
