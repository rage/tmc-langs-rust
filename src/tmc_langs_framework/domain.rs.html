11/// A description of an exercise's test case.
12#[derive(Debug, Deserialize, Serialize, JsonSchema, PartialEq, Eq, Hash)]
13#[cfg_attr(feature = "ts-rs", derive(ts_rs::TS))]
14pub struct TestDesc {
15    /// The full name of the test.
16    ///
17    /// If the language organises tests into suites or classes, it is customary
18    /// to name the test as "class_name.method_name".
19    pub name: String,
20    /// The list of point names that passing this test may give.
21    ///
22    /// To obtain a point X, the user must pass all exercises that require point X.
23    pub points: Vec<String>,
24}
25
26impl TestDesc {
27    pub fn new(name: String, points: Vec<String>) -> Self {
28        Self { name, points }
29    }
30}

62/// How long an exercise's tests may run when its `.tmcproject.yml` sets no `tests_timeout_ms`.
63/// Covers the whole test command, so a build the plugin does as part of it (C#, Maven's test
64/// goal) counts too; Java and make projects are built before the clock starts.
65pub const DEFAULT_TESTS_TIMEOUT: Duration = Duration::from_secs(60);
66
67/// Name of the failing test a timed-out run reports; see [`RunResult::timed_out`].
68pub const TIMEOUT_TEST_NAME: &str = "Timeout test";
69
70/// The result of running an exercise's test suite against a submission.
71#[derive(Debug, Deserialize, Serialize, JsonSchema)]
72#[serde(rename_all = "camelCase")]
73#[cfg_attr(feature = "ts-rs", derive(ts_rs::TS))]
74pub struct RunResult {
75    /// The overall status of a test run.
76    pub status: RunStatus,
77    /// Whether each test passed and which points were awarded.
78    pub test_results: Vec<TestResult>,
79    /// Logs from the test run.
80    /// The key may be an arbitrary string identifying the type of log.
81    pub logs: HashMap<String, String>,
82    /// Set when the tests were stopped for running longer than this many milliseconds: the
83    /// exercise's `tests_timeout_ms`, else 60000. The status is then `TESTS_FAILED`, with one
84    /// failing test named "Timeout test" whose message says so; an infinite loop is the usual
85    /// cause.
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    #[cfg_attr(feature = "ts-rs", ts(optional))]
88    pub timed_out_after_ms: Option<u64>,
89}
90
91impl RunResult {
92    pub fn new(
93        status: RunStatus,
94        test_results: Vec<TestResult>,
95        logs: HashMap<String, String>,
96    ) -> Self {
97        Self {
98            status,
99            test_results,
100            logs,
101            timed_out_after_ms: None,
102        }
103    }
104
105    /// The result of a test run stopped after `timeout`, keeping the output it had produced.
106    pub fn timed_out(timeout: Duration, stdout: String, stderr: String) -> Self {
107        let limit = if timeout.subsec_millis() == 0 {
108            format!("{} s", timeout.as_secs())
109        } else {
110            format!("{} ms", timeout.as_millis())
111        };
112        let mut logs = HashMap::new();
113        logs.insert("stdout".to_string(), stdout);
114        logs.insert("stderr".to_string(), stderr);
115        Self {
116            status: RunStatus::TestsFailed,
117            test_results: vec![TestResult {
118                name: TIMEOUT_TEST_NAME.to_string(),
119                successful: false,
120                points: vec![],
121                message: format!(
122                    "The tests did not finish within {limit} and were stopped. \
123                     Make sure your code has no infinite loop."
124                ),
125                exception: vec![],
126            }],
127            logs,
128            timed_out_after_ms: Some(u64::try_from(timeout.as_millis()).unwrap_or(u64::MAX)),
129        }
130    }
131}

133/// The overall status of a test run.
134#[derive(Debug, Deserialize, Serialize, PartialEq, Eq, JsonSchema)]
135#[cfg_attr(feature = "ts-rs", derive(ts_rs::TS))]
136#[serde(rename_all = "SCREAMING_SNAKE_CASE")]
137pub enum RunStatus {
138    /// The submission and tests compiled and all tests passed.
139    Passed,
140    /// The submission and tests compiled but some tests failed.
141    TestsFailed,
142    /// The submission or tests did not compile.
143    // TODO: "The compiler error should be given in {@code logs[SpecialLogs.COMPILER_OUTPUT]}."
144    CompileFailed,
145    /// The submission compiled but testrun was interrupted.
146    TestrunInterrupted,
147    /// For when no other status seems suitable, or the language plugin has
148    /// suffered an internal error.
149    // TODO: "Details should be given in {@code logs[SpecialLogs.GENERIC_ERROR_MESSAGE]}.""
150    GenericError,
151}

153/// Represents configuration based on which submission may be packaged.
154#[derive(Debug, Deserialize, Serialize, JsonSchema)]
155#[cfg_attr(feature = "ts-rs", derive(ts_rs::TS))]
156pub struct ExercisePackagingConfiguration {
157    /// Student folders or files which are copied from submission.
158    pub student_file_paths: HashSet<PathBuf>,
159    /// Exercise folders or files which are copied from exercise template or clone.
160    pub exercise_file_paths: HashSet<PathBuf>,
161    /// How long the exercise's tests may run: its `.tmcproject.yml`'s `tests_timeout_ms`, else
162    /// [`DEFAULT_TESTS_TIMEOUT`]. Absent only from CLIs older than this field.
163    #[serde(default, skip_serializing_if = "Option::is_none")]
164    #[cfg_attr(feature = "ts-rs", ts(optional))]
165    pub tests_timeout_ms: Option<u64>,
166    /// The language plugin that recognised the exercise: `python3`, `apache-maven`,
167    /// `apache-ant`, `csharp`, `make`, `r` or `No-Tests`. Absent only from CLIs older than this
168    /// field.
169    #[serde(default, skip_serializing_if = "Option::is_none")]
170    #[cfg_attr(feature = "ts-rs", ts(optional))]
171    pub plugin: Option<String>,
172}
