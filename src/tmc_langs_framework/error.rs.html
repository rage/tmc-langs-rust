108/// Most characters of each output stream [`describe_output`] keeps, from the end, where a
109/// failing program says what went wrong.
110const MAX_OUTPUT_CHARS: usize = 4000;
111
112/// A command's output for an error message: each non-empty stream on its own lines, its tail
113/// when long, or nothing when both are empty. Plugins' error messages are the only place a
114/// client sees why a tool failed.
115pub fn describe_output(stdout: &str, stderr: &str) -> String {
116    let mut description = String::new();
117    for (name, output) in [("stdout", stdout), ("stderr", stderr)] {
118        let output = output.trim();
119        if output.is_empty() {
120            continue;
121        }
122        let char_count = output.chars().count();
123        description.push_str(&format!("\n{name}:\n"));
124        if char_count > MAX_OUTPUT_CHARS {
125            description.push('…');
126            description.extend(output.chars().skip(char_count - MAX_OUTPUT_CHARS));
127        } else {
128            description.push_str(output);
129        }
130    }
131    description
132}

134#[cfg(test)]
135mod test {
136    use super::*;
137
138    #[test]
139    fn describe_output_leaves_out_empty_streams() {
140        assert_eq!(describe_output(" \n", ""), "");
141        assert_eq!(
142            describe_output("", "Error occurred during initialization of VM\n"),
143            "\nstderr:\nError occurred during initialization of VM"
144        );
145    }
146
147    #[test]
148    fn describe_output_keeps_the_tail_of_a_long_stream() {
149        let stdout = format!("{}the cause", "x".repeat(MAX_OUTPUT_CHARS * 2));
150        let description = describe_output(&stdout, "");
151        assert!(description.ends_with("the cause"));
152        assert_eq!(
153            description.chars().count(),
154            "\nstdout:\n…".chars().count() + MAX_OUTPUT_CHARS
155        );
156    }
