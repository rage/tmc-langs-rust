1//! Locating the external tools (java, python, make, ...) that plugins run.
2//!
3//! Tools are resolved to absolute paths before spawning: `CreateProcess` given a bare name
4//! searches the current directory before `PATH`, and the CLI inherits an environment fixed when
5//! VS Code started, so tools installed since are missing from it on Windows.
6
7use std::{
8    ffi::{OsStr, OsString},
9    path::{Path, PathBuf},
10};
11
12/// Resolved location of a tool and the directories its own helpers may live in.
13#[derive(Debug, PartialEq, Eq)]
14pub struct ResolvedTool {
15    pub path: PathBuf,
16    /// `Some` when the tool was found outside the inherited `PATH`: the child's `PATH` should
17    /// be this value so the tool finds its sibling programs (a MinGW `make` needs `gcc`).
18    pub child_path: Option<OsString>,
19}
20
21/// Resolves a bare tool name to the file that would run, or `None` if it is not found or is
22/// not a bare name.
23pub fn resolve_tool(program: &OsStr) -> Option<ResolvedTool> {
24    if !is_bare_name(program) {
25        return None;
26    }
27    let tool = normalized_tool_name(&program.to_string_lossy());
28    let tool = tool.as_str();
29    let names = candidate_names(tool, cfg!(windows));
30    let extensions = executable_extensions(tool, cfg!(windows), std::env::var_os("PATHEXT"));
31
32    let path_var = std::env::var_os("PATH").unwrap_or_default();
33    let path_dirs: Vec<PathBuf> = std::env::split_paths(&path_var).collect();
34    if let Some(path) = find_in_dirs(&names, &path_dirs, &extensions) {
35        return Some(ResolvedTool {
36            path,
37            child_path: None,
38        });
39    }
40
41    if !cfg!(windows) {
42        return None;
43    }
44    let registry_dirs = windows_registry_path_dirs();
45    let known_dirs = known_install_dirs(tool, &|name| std::env::var_os(name));
46    let fallback_dirs: Vec<PathBuf> = registry_dirs.iter().cloned().chain(known_dirs).collect();
47    let path = find_in_dirs(&names, &fallback_dirs, &extensions)?;
48    let found_dir = path.parent().map(Path::to_path_buf);
49    let extra_dirs: Vec<PathBuf> = registry_dirs.into_iter().chain(found_dir).collect();
50    Some(ResolvedTool {
51        path,
52        child_path: Some(merge_path_dirs(&path_var, &extra_dirs)),
53    })
54}

56/// `tool` lowercased without an `.exe` suffix, as Windows matches names.
57fn normalized_tool_name(tool: &str) -> String {
58    let lower = tool.to_lowercase();
59    lower.strip_suffix(".exe").unwrap_or(&lower).to_string()
60}
61
62fn is_bare_name(program: &OsStr) -> bool {
63    !program.is_empty() && !program.to_string_lossy().contains(['/', '\\'])
64}
65
66/// Names a tool may be installed under; the first existing one wins.
67fn candidate_names(tool: &str, is_windows: bool) -> Vec<String> {
68    let mut names = vec![tool.to_string()];
69    if is_windows {
70        match tool {
71            "make" => names.extend(["mingw32-make".to_string(), "gmake".to_string()]),
72            "python3" => names.push("python".to_string()),
73            _ => {}
74        }
75    }
76    names
77}
78
79/// Suffixes tried after `tool`'s candidate names. On Windows the empty one is tried, last, only
80/// when `tool` already carries an extension: an extensionless file there is a shell shim that
81/// cannot be started (Rtools puts a `cmd` script on `PATH` ahead of `cmd.exe`).
82fn executable_extensions(tool: &str, is_windows: bool, pathext: Option<OsString>) -> Vec<String> {
83    if !is_windows {
84        return vec![String::new()];
85    }
86    let pathext = pathext
87        .map(|v| v.to_string_lossy().into_owned())
88        .filter(|v| !v.is_empty())
89        .unwrap_or_else(|| ".COM;.EXE;.BAT;.CMD".to_string());
90    let has_extension = Path::new(tool).extension().is_some();
91    pathext
92        .split(';')
93        .filter(|e| !e.is_empty())
94        .map(|e| e.to_lowercase())
95        .chain(has_extension.then(String::new))
96        .collect()
97}

99/// A regular file the shell could run; on Unix one without an execute bit is skipped like it does.
100fn is_runnable_file(path: &Path) -> bool {
101    #[cfg(unix)]
102    {
103        use std::os::unix::fs::PermissionsExt;
104        path.metadata()
105            .is_ok_and(|meta| meta.is_file() && meta.permissions().mode() & 0o111 != 0)
106    }
107    #[cfg(not(unix))]
108    {
109        path.is_file()
110    }
111}
112
113fn find_in_dirs(names: &[String], dirs: &[PathBuf], extensions: &[String]) -> Option<PathBuf> {
114    for dir in dirs {
115        if dir.as_os_str().is_empty() || !dir.is_absolute() {
116            // a relative PATH entry is the current-directory search by another name
117            continue;
118        }
119        for name in names {
120            for extension in extensions {
121                let candidate = dir.join(format!("{name}{extension}"));
122                if is_runnable_file(&candidate) && !is_store_alias_stub(&candidate) {
123                    return Some(candidate);
124                }
125            }
126        }
127    }
128    None
129}

131/// The `python.exe` in `WindowsApps` is a Microsoft Store launcher stub that opens the Store
132/// instead of running Python.
133fn is_store_alias_stub(path: &Path) -> bool {
134    let is_python = path
135        .file_stem()
136        .is_some_and(|stem| stem.to_string_lossy().to_lowercase().starts_with("python"));
137    is_python
138        && path
139            .components()
140            .any(|c| c.as_os_str().eq_ignore_ascii_case("WindowsApps"))
141}

143/// Per-tool install locations that installers use without adding to `PATH`.
144fn known_install_dirs(tool: &str, env: &dyn Fn(&str) -> Option<OsString>) -> Vec<PathBuf> {
145    let program_files: Vec<PathBuf> = ["ProgramFiles", "ProgramW6432", "ProgramFiles(x86)"]
146        .iter()
147        .filter_map(|name| env(name))
148        .map(PathBuf::from)
149        .collect();
150    let mut dirs = vec![];
151    match tool {
152        "rscript" => {
153            if let Some(r_home) = env("R_HOME") {
154                dirs.push(PathBuf::from(r_home).join("bin"));
155            }
156            for pf in &program_files {
157                dirs.extend(
158                    newest_first_subdirs(&pf.join("R"))
159                        .into_iter()
160                        .map(|d| d.join("bin")),
161                );
162            }
163        }
164        "make" => {
165            for root in [
166                "C:\\msys64",
167                "C:\\MinGW",
168                "C:\\Strawberry\\c",
169                "C:\\ProgramData\\chocolatey",
170            ] {
171                dirs.push(PathBuf::from(root).join("usr").join("bin"));
172                dirs.push(PathBuf::from(root).join("mingw64").join("bin"));
173                dirs.push(PathBuf::from(root).join("bin"));
174            }
175        }
176        "dotnet" => dirs.extend(program_files.iter().map(|pf| pf.join("dotnet"))),
177        "python3" | "python" | "py" => {
178            if let Some(local) = env("LocalAppData") {
179                let programs = PathBuf::from(local).join("Programs").join("Python");
180                for dir in newest_first_subdirs(&programs) {
181                    dirs.push(dir.join("Scripts"));
182                    dirs.push(dir);
183                }
184            }
185            for pf in &program_files {
186                dirs.extend(newest_first_subdirs(&pf.join("Python")));
187            }
188            if let Some(profile) = env("USERPROFILE") {
189                for conda in ["anaconda3", "miniconda3"] {
190                    dirs.push(PathBuf::from(&profile).join(conda));
191                }
192            }
193            for conda in ["C:\\ProgramData\\anaconda3", "C:\\ProgramData\\miniconda3"] {
194                dirs.push(PathBuf::from(conda));
195            }
196        }
197        _ => {}
198    }
199    dirs
200}

202/// Subdirectories of `dir` in descending name order (so `R-4.4.1` before `R-4.3.0`).
203fn newest_first_subdirs(dir: &Path) -> Vec<PathBuf> {
204    let Ok(entries) = std::fs::read_dir(dir) else {
205        return vec![];
206    };
207    let mut dirs: Vec<PathBuf> = entries
208        .filter_map(Result::ok)
209        .map(|e| e.path())
210        .filter(|p| p.is_dir())
211        .collect();
212    dirs.sort_by(|a, b| b.file_name().cmp(&a.file_name()));
213    dirs
214}

216/// `current` with each of `extra` appended unless already present (compared ignoring case,
217/// as Windows does).
218fn merge_path_dirs(current: &OsStr, extra: &[PathBuf]) -> OsString {
219    let mut dirs: Vec<PathBuf> = std::env::split_paths(current).collect();
220    for dir in extra {
221        let is_present = dirs
222            .iter()
223            .any(|existing| existing.as_os_str().eq_ignore_ascii_case(dir.as_os_str()));
224        if !is_present {
225            dirs.push(dir.clone());
226        }
227    }
228    std::env::join_paths(dirs).unwrap_or_else(|_| current.to_os_string())
229}

231/// `PATH` directories as an installer last saved them (machine, then user), which a process
232/// started before the installation does not see.
233fn windows_registry_path_dirs() -> Vec<PathBuf> {
234    #[cfg(windows)]
235    {
236        windows_registry::path_dirs()
237    }
238    #[cfg(not(windows))]
239    {
240        vec![]
241    }

244/// `path` made absolute, for use as a toolchain's working directory or class-path entry.
245///
246/// Not canonicalized: resolving a mapped drive (`H:\x`) yields its UNC target, which `cmd.exe`
247/// cannot use as a working directory. A `\\?\UNC\` form is rewritten to `\\server\share`, which
248/// the JVM, Python and CreateProcess accept.
249pub fn absolute_tool_path(path: &Path) -> std::io::Result<PathBuf> {
250    let absolute = std::path::absolute(path)?;
251    Ok(simplify_verbatim_path(&absolute))
252}
253
254fn simplify_verbatim_path(path: &Path) -> PathBuf {
255    const VERBATIM_UNC: &str = "\\\\?\\UNC\\";
256    const VERBATIM_DISK: &str = "\\\\?\\";
257    let Some(text) = path.to_str() else {
258        return path.to_path_buf();
259    };
260    if let Some(rest) = text.strip_prefix(VERBATIM_UNC) {
261        return PathBuf::from(format!("\\\\{rest}"));
262    }
263    match text.strip_prefix(VERBATIM_DISK) {
264        Some(rest) if rest.as_bytes().get(1..3) == Some(b":\\") => PathBuf::from(rest),
265        _ => path.to_path_buf(),
266    }
267}

269/// What to tell a user whose `tool` was not found.
270pub fn missing_tool_hint(tool: &str) -> String {
271    let install = match normalized_tool_name(tool).as_str() {
272        "java" | "javac" | "mvn" | "mvn.cmd" | "ant" | "ant.bat" => {
273            "Install a JDK (Java 11 or newer) and make sure JAVA_HOME points to it or java is on PATH."
274        }
275        "python3" | "python" | "py" => {
276            "Install Python 3 or set TMC_LANGS_PYTHON_EXEC to the full path of a Python 3 interpreter."
277        }
278        "make" | "mingw32-make" | "gmake" => {
279            "Install a C toolchain with make (on Windows, MSYS2 or MinGW) and add its bin folder to PATH."
280        }
281        "rscript" => {
282            "Install R and add its bin folder (for example C:\\Program Files\\R\\R-x.y.z\\bin) to PATH."
283        }
284        "dotnet" => "Install the .NET SDK.",
285        "git" => "Install Git.",
286        _ => "Install it and make sure it is on PATH.",
287    };
288    format!(
289        "{install} If you installed it after starting VS Code, close all VS Code windows and start it again."
290    )
291}

293#[cfg(windows)]
294mod windows_registry {
295    use std::{ffi::OsString, os::windows::ffi::OsStringExt, path::PathBuf};
296    use windows_sys::Win32::System::{
297        Environment::ExpandEnvironmentStringsW,
298        Registry::{
299            HKEY, HKEY_CURRENT_USER, HKEY_LOCAL_MACHINE, KEY_READ, REG_EXPAND_SZ, REG_SZ,
300            RegCloseKey, RegOpenKeyExW, RegQueryValueExW,
301        },
302    };
303
304    fn wide(s: &str) -> Vec<u16> {
305        s.encode_utf16().chain(std::iter::once(0)).collect()
306    }
307
308    fn read_path_value(root: HKEY, subkey: &str) -> Option<OsString> {
309        let subkey = wide(subkey);
310        let value_name = wide("Path");
311        let mut key: HKEY = std::ptr::null_mut();
312        // SAFETY: pointers refer to live, NUL-terminated buffers for the call duration.
313        unsafe {
314            if RegOpenKeyExW(root, subkey.as_ptr(), 0, KEY_READ, &mut key) != 0 {
315                return None;
316            }
317            let mut kind = 0u32;
318            let mut size = 0u32;
319            let mut result = None;
320            if RegQueryValueExW(
321                key,
322                value_name.as_ptr(),
323                std::ptr::null(),
324                &mut kind,
325                std::ptr::null_mut(),
326                &mut size,
327            ) == 0
328                && (kind == REG_SZ || kind == REG_EXPAND_SZ)
329                && size > 0
330            {
331                let mut buffer = vec![0u16; (size as usize).div_ceil(2) + 1];
332                let mut size_in_out = size;
333                if RegQueryValueExW(
334                    key,
335                    value_name.as_ptr(),
336                    std::ptr::null(),
337                    &mut kind,
338                    buffer.as_mut_ptr().cast(),
339                    &mut size_in_out,
340                ) == 0
341                {
342                    let len = buffer.iter().position(|&c| c == 0).unwrap_or(buffer.len());
343                    buffer.truncate(len);
344                    buffer.push(0);
345                    result = Some(if kind == REG_EXPAND_SZ {
346                        expand(&buffer)
347                    } else {
348                        OsString::from_wide(&buffer[..buffer.len() - 1])
349                    });
350                }
351            }
352            RegCloseKey(key);
353            result
354        }
355    }
356
357    /// `source` is NUL-terminated.
358    fn expand(source: &[u16]) -> OsString {
359        // SAFETY: `source` is NUL-terminated; the buffer length passed is its real length.
360        unsafe {
361            let needed = ExpandEnvironmentStringsW(source.as_ptr(), std::ptr::null_mut(), 0);
362            if needed == 0 {
363                return OsString::from_wide(&source[..source.len() - 1]);
364            }
365            let mut buffer = vec![0u16; needed as usize];
366            let written = ExpandEnvironmentStringsW(source.as_ptr(), buffer.as_mut_ptr(), needed);
367            if written == 0 || written > needed {
368                return OsString::from_wide(&source[..source.len() - 1]);
369            }
370            OsString::from_wide(&buffer[..(written as usize).saturating_sub(1)])
371        }
372    }
373
374    pub fn path_dirs() -> Vec<PathBuf> {
375        let machine = read_path_value(
376            HKEY_LOCAL_MACHINE,
377            "SYSTEM\\CurrentControlSet\\Control\\Session Manager\\Environment",
378        );
379        let user = read_path_value(HKEY_CURRENT_USER, "Environment");
380        machine
381            .into_iter()
382            .chain(user)
383            .flat_map(|value| std::env::split_paths(&value).collect::<Vec<_>>())
384            .collect()
385    }
386}
