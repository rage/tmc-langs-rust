4use crate::weak_slice::WeakSliceMut;
5
6pub(crate) struct SymBuf<'a> {
7    buf: WeakSliceMut<'a, u8>,
8    filled: usize,
9}
10
11impl<'a> SymBuf<'a> {
12    #[inline]
13    pub fn iter(&self) -> impl Iterator<Item = (u16, u8)> + '_ {
14        self.buf.as_slice()[..self.filled]
15            .chunks_exact(3)
16            .map(|chunk| match *chunk {
17                [dist_low, dist_high, lc] => (u16::from_le_bytes([dist_low, dist_high]), lc),
18                _ => unreachable!("chunks are exactly 3 elements wide"),
19            })
20    }
21
22    #[inline]
23    pub fn should_flush_block(&self) -> bool {
24        self.filled == self.buf.len() - 3
25    }
26
27    /// Returns true if there are no bytes in this ReadBuf
28    #[inline]
29    pub fn is_empty(&self) -> bool {
30        self.filled == 0
31    }
32
33    /// Clears the buffer, resetting the filled region to empty.
34    ///
35    /// The number of initialized bytes is not changed, and the contents of the buffer are not modified.
36    #[inline]
37    pub fn clear(&mut self) {
38        self.buf.as_mut_slice().fill(0);
39        self.filled = 0;
40    }
41
42    #[inline(always)]
43    pub fn push_lit(&mut self, byte: u8) {
44        // NOTE: we rely on the buffer being zeroed here!
45        self.buf.as_mut_slice()[self.filled + 2] = byte;
46
47        self.filled += 3;
48    }
49
50    #[inline(always)]
51    pub fn push_dist(&mut self, dist: u16, len: u8) {
52        let buf = &mut self.buf.as_mut_slice()[self.filled..][..3];
53        let [dist1, dist2] = dist.to_le_bytes();
54
55        buf[0] = dist1;
56        buf[1] = dist2;
57        buf[2] = len;
58
59        self.filled += 3;
60    }
61
62    pub(crate) unsafe fn from_raw_parts(ptr: *mut u8, lit_bufsize: usize) -> Self {
63        Self {
64            buf: unsafe { WeakSliceMut::from_raw_parts_mut(ptr, lit_bufsize * 3) },
65            filled: 0,
66        }
67    }
68
69    pub(crate) unsafe fn clone_to(&self, ptr: *mut u8) -> Self {
70        unsafe { ptr.copy_from_nonoverlapping(self.buf.as_ptr(), self.buf.len()) };
71        Self {
72            buf: unsafe { WeakSliceMut::from_raw_parts_mut(ptr, self.buf.len()) },
73            filled: self.filled,
74        }
75    }
76}
