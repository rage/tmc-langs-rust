25/// Initialize the stream in an inflate state
26pub fn back_init(stream: &mut z_stream, config: InflateConfig, window: Window) -> ReturnCode {
27    assert_eq!(1 << config.window_bits, window.buffer_size());
28
29    stream.msg = core::ptr::null_mut();
30
31    // for safety we must really make sure that alloc and free are consistent
32    // this is a (slight) deviation from stock zlib. In this crate we pick the rust
33    // allocator as the default, but `libz-rs-sys` configures the C allocator
34    #[cfg(feature = "rust-allocator")]
35    if stream.zalloc.is_none() || stream.zfree.is_none() {
36        stream.configure_default_rust_allocator()
37    }
38
39    #[cfg(feature = "c-allocator")]
40    if stream.zalloc.is_none() || stream.zfree.is_none() {
41        stream.configure_default_c_allocator()
42    }
43
44    if stream.zalloc.is_none() || stream.zfree.is_none() {
45        return ReturnCode::StreamError;
46    }
47
48    let mut state = State::new(&[], Writer::new(&mut []));
49
50    // TODO this can change depending on the used/supported SIMD instructions
51    state.chunksize = 32;
52
53    let alloc = Allocator {
54        zalloc: stream.zalloc.unwrap(),
55        zfree: stream.zfree.unwrap(),
56        opaque: stream.opaque,
57        _marker: PhantomData,
58    };
59    let allocs = InflateAllocOffsets::new();
60
61    let Some(allocation_start) = alloc.allocate_slice_raw::<u8>(allocs.total_size) else {
62        return ReturnCode::MemError;
63    };
64
65    let address = allocation_start.as_ptr() as usize;
66    let align_offset = address.next_multiple_of(64) - address;
67    let buf = unsafe { allocation_start.as_ptr().add(align_offset) };
68
69    // NOTE: the window part of the allocation is ignored in this case.
70    state.window = window;
71
72    let state_allocation = unsafe { buf.add(allocs.state_pos).cast::<State>() };
73    unsafe { state_allocation.write(state) };
74    stream.state = state_allocation.cast::<internal_state>();
75
76    // SAFETY: we've correctly initialized the stream to be an InflateStream
77    let Some(stream) = (unsafe { InflateStream::from_stream_mut(stream) }) else {
78        return ReturnCode::StreamError;
79    };
80
81    stream.state.allocation_start = allocation_start.as_ptr();
82    stream.state.total_allocation_size = allocs.total_size;
83
84    stream.state.wbits = config.window_bits as u8;
85    stream.state.flags.update(Flags::SANE, true);
86
87    ReturnCode::Ok
88}

90pub unsafe fn back(
91    strm: &mut InflateStream,
92    in_: in_func,
93    in_desc: *mut c_void,
94    out: out_func,
95    out_desc: *mut c_void,
96) -> ReturnCode {
97    let mut ret;
98
99    /* Reset the state */
100    strm.msg = core::ptr::null_mut();
101    strm.state.mode = Mode::Type;
102    strm.state.flags.update(Flags::IS_LAST_BLOCK, false);
103    strm.state.window.clear();
104
105    let mut next = strm.next_in.cast_const();
106    let mut have = if !next.is_null() { strm.avail_in } else { 0 };
107    let mut hold = 0;
108    let mut bits = 0u8;
109    let mut put = strm.state.window.as_ptr().cast_mut();
110    let mut left = strm.state.window.buffer_size();
111
112    let state = &mut strm.state;
113
114    'inf_leave: loop {
115        macro_rules! initbits {
116            () => {
117                hold = 0;
118                bits = 0;
119            };
120        }
121
122        macro_rules! bytebits {
123            () => {
124                hold >>= bits & 7;
125                bits -= bits & 7;
126            };
127        }
128
129        macro_rules! dropbits {
130            ($n:expr) => {
131                hold >>= $n;
132                bits -= $n;
133            };
134        }
135
136        macro_rules! bits {
137            ($n:expr) => {
138                hold & ((1 << $n) - 1)
139            };
140        }
141
142        macro_rules! needbits {
143            ($n:expr) => {
144                while usize::from(bits) < $n {
145                    pullbyte!();
146                }
147            };
148        }
149
150        macro_rules! pull {
151            () => {
152                if have == 0 {
153                    have = unsafe { in_(in_desc, &mut next) };
154                    if have == 0 {
155                        #[allow(unused_assignments)]
156                        {
157                            next = core::ptr::null();
158                        }
159                        ret = ReturnCode::BufError;
160                        break 'inf_leave;
161                    }
162                }
163            };
164        }
165
166        macro_rules! pullbyte {
167            () => {
168                pull!();
169                have -= 1;
170                hold += (unsafe { *next as u64 }) << bits;
171                next = unsafe { next.add(1) };
172                bits += 8;
173            };
174        }
175
176        macro_rules! room {
177            () => {
178                if left == 0 {
179                    left = state.window.buffer_size();
180                    let window = state.window.as_slice();
181                    put = window.as_ptr().cast_mut();
182
183                    unsafe { state.window.set_have(left) };
184
185                    if unsafe { out(out_desc, put, left as u32) } != 0 {
186                        ret = ReturnCode::BufError;
187                        break 'inf_leave;
188                    }
189                }
190            };
191        }
192
193        match state.mode {
194            Mode::Type => {
195                if state.flags.contains(Flags::IS_LAST_BLOCK) {
196                    bytebits!();
197                    state.mode = Mode::Done;
198                    continue;
199                }
200
201                needbits!(3);
202
203                let last = bits!(1) != 0;
204                state.flags.update(Flags::IS_LAST_BLOCK, last);
205                dropbits!(1);
206
207                match bits!(2) {
208                    0b00 => {
209                        tracev!("inflate:     stored block (last = {last})");
210
211                        dropbits!(2);
212                        state.mode = Mode::Stored;
213                        continue;
214                    }
215                    0b01 => {
216                        tracev!("inflate:     fixed codes block (last = {last})");
217
218                        state.len_table = Table {
219                            codes: Codes::Fixed,
220                            bits: 9,
221                        };
222
223                        state.dist_table = Table {
224                            codes: Codes::Fixed,
225                            bits: 5,
226                        };
227
228                        dropbits!(2);
229                        state.mode = Mode::Len;
230                        continue;
231                    }
232                    0b10 => {
233                        tracev!("inflate:     dynamic codes block (last = {last})");
234
235                        dropbits!(2);
236                        state.mode = Mode::Table;
237                        continue;
238                    }
239                    0b11 => {
240                        tracev!("inflate:     invalid block type");
241
242                        dropbits!(2);
243                        state.mode = Mode::Bad;
244                        state.bad("invalid block type\0");
245                        continue;
246                    }
247                    _ => {
248                        // LLVM will optimize this branch away
249                        unreachable!("BitReader::bits(2) only yields a value of two bits, so this match is already exhaustive")
250                    }
251                }
252            }
253            Mode::Stored => {
254                bytebits!();
255                needbits!(32);
256
257                if hold as u16 != !((hold >> 16) as u16) {
258                    state.mode = Mode::Bad;
259                    state.bad("invalid stored block lengths\0");
260                    continue;
261                }
262
263                state.length = hold as usize & 0xFFFF;
264                tracev!("inflate:     stored length {}", state.length);
265
266                initbits!();
267
268                /* copy stored block from input to output */
269                while state.length != 0 {
270                    let mut copy = state.length;
271
272                    pull!();
273                    room!();
274
275                    copy = Ord::min(copy, have as usize);
276                    copy = Ord::min(copy, left);
277
278                    unsafe { core::ptr::copy(next, put, copy) };
279
280                    have -= copy as u32;
281                    next = unsafe { next.add(copy) };
282
283                    left -= copy;
284                    put = unsafe { put.add(copy) };
285
286                    state.length -= copy;
287                }
288
289                state.mode = Mode::Type;
290                continue;
291            }
292            Mode::Table => {
293                needbits!(14);
294                state.nlen = bits!(5) as usize + 257;
295                dropbits!(5);
296                state.ndist = bits!(5) as usize + 1;
297                dropbits!(5);
298                state.ncode = bits!(4) as usize + 4;
299                dropbits!(4);
300
301                // TODO pkzit_bug_workaround
302                if state.nlen > 286 || state.ndist > 30 {
303                    state.mode = Mode::Bad;
304                    state.bad("too many length or distance symbols\0");
305                    continue;
306                }
307
308                tracev!("inflate:       table sizes ok");
309                state.have = 0;
310
311                // permutation of code lengths ;
312                const ORDER: [u8; 19] = [
313                    16, 17, 18, 0, 8, 7, 9, 6, 10, 5, 11, 4, 12, 3, 13, 2, 14, 1, 15,
314                ];
315
316                while state.have < state.ncode {
317                    needbits!(3);
318                    state.lens[usize::from(ORDER[state.have])] = bits!(3) as u16;
319                    state.have += 1;
320                    dropbits!(3);
321                }
322
323                while state.have < 19 {
324                    state.lens[usize::from(ORDER[state.have])] = 0;
325                    state.have += 1;
326                }
327
328                let InflateTable::Success { root, used } = inflate_table(
329                    CodeType::Codes,
330                    &state.lens[..19],
331                    &mut state.codes_codes,
332                    7,
333                    &mut state.work,
334                ) else {
335                    state.mode = Mode::Bad;
336                    state.bad("invalid code lengths set\0");
337                    continue;
338                };
339
340                state.next = used;
341                state.len_table.codes = Codes::Codes;
342                state.len_table.bits = root;
343
344                tracev!("inflate:       table sizes ok");
345                state.have = 0;
346
347                while state.have < state.nlen + state.ndist {
348                    let here = loop {
349                        let here = state.len_table_get(bits!(state.len_table.bits) as usize);
350                        if here.bits <= bits {
351                            break here;
352                        }
353
354                        pullbyte!();
355                    };
356
357                    let here_bits = here.bits;
358
359                    match here.val {
360                        0..=15 => {
361                            dropbits!(here_bits);
362                            state.lens[state.have] = here.val;
363                            state.have += 1;
364                        }
365                        16 => {
366                            needbits!(usize::from(here_bits) + 2);
367                            dropbits!(here_bits);
368                            if state.have == 0 {
369                                state.mode = Mode::Bad;
370                                state.bad("invalid bit length repeat\0");
371                                continue 'inf_leave;
372                            }
373
374                            let len = state.lens[state.have - 1];
375                            let copy = 3 + bits!(2) as usize;
376                            dropbits!(2);
377
378                            if state.have + copy > state.nlen + state.ndist {
379                                state.mode = Mode::Bad;
380                                state.bad("invalid bit length repeat\0");
381                                continue 'inf_leave;
382                            }
383
384                            state.lens[state.have..][..copy].fill(len);
385                            state.have += copy;
386                        }
387                        17 => {
388                            needbits!(usize::from(here_bits) + 3);
389                            dropbits!(here_bits);
390                            let copy = 3 + bits!(3) as usize;
391                            dropbits!(3);
392
393                            if state.have + copy > state.nlen + state.ndist {
394                                state.mode = Mode::Bad;
395                                state.bad("invalid bit length repeat\0");
396                                continue 'inf_leave;
397                            }
398
399                            state.lens[state.have..][..copy].fill(0);
400                            state.have += copy;
401                        }
402                        18.. => {
403                            needbits!(usize::from(here_bits) + 7);
404                            dropbits!(here_bits);
405                            let copy = 11 + bits!(7) as usize;
406                            dropbits!(7);
407
408                            if state.have + copy > state.nlen + state.ndist {
409                                state.mode = Mode::Bad;
410                                state.bad("invalid bit length repeat\0");
411                                continue 'inf_leave;
412                            }
413
414                            state.lens[state.have..][..copy].fill(0);
415                            state.have += copy;
416                        }
417                    }
418                }
419
420                // check for end-of-block code (better have one)
421                if state.lens[256] == 0 {
422                    state.mode = Mode::Bad;
423                    state.bad("invalid code -- missing end-of-block\0");
424                    continue 'inf_leave;
425                }
426
427                // build code tables
428
429                let InflateTable::Success { root, used } = inflate_table(
430                    CodeType::Lens,
431                    &state.lens[..state.nlen],
432                    &mut state.len_codes,
433                    10,
434                    &mut state.work,
435                ) else {
436                    state.mode = Mode::Bad;
437                    state.bad("invalid literal/lengths set\0");
438                    continue 'inf_leave;
439                };
440
441                state.len_table.codes = Codes::Len;
442                state.len_table.bits = root;
443                state.next = used;
444
445                let InflateTable::Success { root, used } = inflate_table(
446                    CodeType::Dists,
447                    &state.lens[state.nlen..][..state.ndist],
448                    &mut state.dist_codes,
449                    9,
450                    &mut state.work,
451                ) else {
452                    state.mode = Mode::Bad;
453                    state.bad("invalid distances set\0");
454                    continue 'inf_leave;
455                };
456
457                state.dist_table.bits = root;
458                state.dist_table.codes = Codes::Dist;
459                state.next += used;
460
461                state.mode = Mode::Len;
462            }
463            Mode::Len => {
464                if (have as usize) >= INFLATE_FAST_MIN_HAVE && left >= INFLATE_FAST_MIN_LEFT {
465                    let mut bit_reader = BitReader::new(&[]);
466                    unsafe { bit_reader.update_slice(next, have as usize) };
467                    bit_reader.prime(bits, hold);
468
469                    state.bit_reader = bit_reader;
470                    state.writer = unsafe {
471                        Writer::new_uninit_raw(
472                            put.wrapping_sub(state.window.buffer_size() - left),
473                            state.window.buffer_size() - left,
474                            state.window.buffer_size(),
475                        )
476                    };
477
478                    if state.window.have() < state.window.buffer_size() {
479                        unsafe { state.window.set_have(state.window.buffer_size() - left) };
480                    }
481
482                    unsafe { inflate_fast_back(state) };
483
484                    hold = state.bit_reader.hold();
485                    bits = state.bit_reader.bits_in_buffer();
486
487                    next = state.bit_reader.as_ptr();
488                    have = state.bit_reader.bytes_remaining() as u32;
489
490                    put = state.writer.next_out().cast();
491                    left = state.writer.remaining();
492
493                    continue 'inf_leave;
494                }
495
496                let len_table = match state.len_table.codes {
497                    Codes::Fixed => &crate::inflate::inffixed_tbl::LENFIX[..],
498                    Codes::Codes => &state.codes_codes,
499                    Codes::Len => &state.len_codes,
500                    Codes::Dist => &state.dist_codes,
501                };
502
503                // get a literal, length, or end-of-block code
504                let mut here;
505                loop {
506                    here = len_table[bits!(state.len_table.bits) as usize];
507
508                    if here.bits <= bits {
509                        break;
510                    }
511
512                    pullbyte!();
513                }
514
515                if here.op != 0 && here.op & 0xf0 == 0 {
516                    let last = here;
517                    loop {
518                        let tmp = bits!((last.bits + last.op) as usize) as u16;
519                        here = len_table[(last.val + (tmp >> last.bits)) as usize];
520                        if last.bits + here.bits <= bits {
521                            break;
522                        }
523
524                        pullbyte!();
525                    }
526
527                    dropbits!(last.bits);
528                }
529
530                dropbits!(here.bits);
531                state.length = here.val as usize;
532
533                if here.op == 0 {
534                    if here.val >= 0x20 && here.val < 0x7f {
535                        tracev!("inflate:         literal '{}'", here.val as u8 as char);
536                    } else {
537                        tracev!("inflate:         literal {:#04x}", here.val);
538                    }
539                    room!();
540
541                    unsafe {
542                        *put = state.length as u8;
543                        put = put.add(1)
544                    }
545                    left -= 1;
546
547                    state.mode = Mode::Len;
548                    continue;
549                } else if here.op & 32 != 0 {
550                    // end of block
551
552                    tracev!("inflate:         end of block");
553
554                    state.mode = Mode::Type;
555                    continue;
556                } else if here.op & 64 != 0 {
557                    state.mode = Mode::Bad;
558                    state.bad("invalid literal/length code\0");
559                    continue;
560                } else {
561                    // length code
562                    state.extra = (here.op & MAX_BITS) as usize;
563                }
564
565                // get extra bits, if any
566                if state.extra != 0 {
567                    needbits!(state.extra);
568                    state.length += bits!(state.extra) as usize;
569                    dropbits!(state.extra as u8);
570                }
571                tracev!("inflate:         length {}", state.length);
572
573                // get distance code
574                let mut here;
575                loop {
576                    here = state.dist_table_get(bits!(state.dist_table.bits) as usize);
577                    if here.bits <= bits {
578                        break;
579                    }
580
581                    pullbyte!();
582                }
583
584                if here.op & 0xf0 == 0 {
585                    let last = here;
586
587                    loop {
588                        here = state.dist_table_get(
589                            last.val as usize
590                                + ((bits!((last.bits + last.op) as usize) as usize) >> last.bits),
591                        );
592
593                        if last.bits + here.bits <= bits {
594                            break;
595                        }
596
597                        pullbyte!();
598                    }
599
600                    dropbits!(last.bits);
601                }
602
603                dropbits!(here.bits);
604
605                if here.op & 64 != 0 {
606                    state.mode = Mode::Bad;
607                    state.bad("invalid distance code\0");
608                    continue 'inf_leave;
609                }
610
611                state.offset = here.val as usize;
612
613                state.extra = (here.op & MAX_BITS) as usize;
614
615                let extra = state.extra;
616
617                if extra > 0 {
618                    needbits!(extra);
619                    state.offset += bits!(extra) as usize;
620                    dropbits!(extra as u8);
621                }
622
623                if INFLATE_STRICT
624                    && state.offset
625                        > state.window.buffer_size()
626                            - (if state.window.have() < state.window.buffer_size() {
627                                left
628                            } else {
629                                0
630                            })
631                {
632                    state.mode = Mode::Bad;
633                    state.bad("invalid distance too far back\0");
634                    continue 'inf_leave;
635                }
636
637                tracev!("inflate:         distance {}", state.offset);
638
639                loop {
640                    room!();
641                    let mut copy = state.window.buffer_size() - state.offset;
642                    let mut from;
643
644                    if copy < left {
645                        from = put.wrapping_add(copy);
646                        copy = left - copy;
647                    } else {
648                        from = put.wrapping_sub(state.offset);
649                        copy = left;
650                    }
651
652                    copy = Ord::min(copy, state.length);
653                    state.length -= copy;
654                    left -= copy;
655
656                    for _ in 0..copy {
657                        unsafe {
658                            *put = *from;
659                            put = put.add(1);
660                            from = from.add(1);
661                        }
662                    }
663
664                    if state.length == 0 {
665                        break;
666                    }
667                }
668
669                continue 'inf_leave;
670            }
671            Mode::Done => {
672                ret = ReturnCode::StreamEnd;
673                break 'inf_leave;
674            }
675            Mode::Bad => {
676                ret = ReturnCode::DataError;
677                break 'inf_leave;
678            }
679
680            Mode::Head
681            | Mode::Flags
682            | Mode::Time
683            | Mode::Os
684            | Mode::ExLen
685            | Mode::Extra
686            | Mode::Name
687            | Mode::Comment
688            | Mode::HCrc
689            | Mode::Sync
690            | Mode::Mem
691            | Mode::Length
692            | Mode::TypeDo
693            | Mode::CopyBlock
694            | Mode::Check
695            | Mode::Len_
696            | Mode::Lit
697            | Mode::LenExt
698            | Mode::Dist
699            | Mode::DistExt
700            | Mode::Match
701            | Mode::LenLens
702            | Mode::CodeLens
703            | Mode::DictId
704            | Mode::Dict => {
705                // All other states should be unreachable, and return StreamError.
706                ret = ReturnCode::StreamError;
707                break 'inf_leave;
708            }
709        }
710    }
711
712    if left < state.window.buffer_size()
713        && unsafe {
714            out(
715                out_desc,
716                state.window.as_ptr().cast_mut(),
717                state.window.buffer_size() as u32 - left as u32,
718            )
719        } != 0
720        && ret == ReturnCode::StreamEnd
721    {
722        ret = ReturnCode::BufError;
723    }
724
725    strm.next_in = next.cast_mut();
726    strm.avail_in = have;
727
728    ret
729}

731#[inline(always)]
732unsafe fn inflate_fast_back(state: &mut State) {
733    let mut bit_reader = BitReader::new(&[]);
734    core::mem::swap(&mut bit_reader, &mut state.bit_reader);
735    debug_assert!(bit_reader.bytes_remaining() >= 15);
736
737    let mut writer = Writer::new(&mut []);
738    core::mem::swap(&mut writer, &mut state.writer);
739
740    let lcode = state.len_table_ref();
741    let dcode = state.dist_table_ref();
742
743    // IDEA: use const generics for the bits here?
744    let lmask = (1u64 << state.len_table.bits) - 1;
745    let dmask = (1u64 << state.dist_table.bits) - 1;
746
747    // TODO verify if this is relevant for us
748    let extra_safe = false;
749
750    let window_size = state.window.buffer_size();
751
752    let mut bad = None;
753
754    if bit_reader.bits_in_buffer() < 10 {
755        debug_assert!(bit_reader.bytes_remaining() >= 15);
756        // Safety: Caller ensured that bit_reader has >= 15 bytes available; refill only needs 8.
757        unsafe { bit_reader.refill() };
758    }
759    // We had at least 15 bytes in the slice, plus whatever was in the buffer. After filling the
760    // buffer from the slice, we now have at least 8 bytes remaining in the slice, plus a full buffer.
761    debug_assert!(
762        bit_reader.bytes_remaining() >= 8 && bit_reader.bytes_remaining_including_buffer() >= 15
763    );
764
765    'outer: loop {
766        // This condition is ensured above for the first iteration of the `outer` loop. For
767        // subsequent iterations, the loop continuation condition is
768        // `bit_reader.bytes_remaining_including_buffer() > 15`. And because the buffer
769        // contributes at most 7 bytes to the result of bit_reader.bytes_remaining_including_buffer(),
770        // that means that the slice contains at least 8 bytes.
771        debug_assert!(
772            bit_reader.bytes_remaining() >= 8
773                && bit_reader.bytes_remaining_including_buffer() >= 15
774        );
775
776        let mut here = {
777            let bits = bit_reader.bits_in_buffer();
778            let hold = bit_reader.hold();
779
780            // Safety: As described in the comments for the debug_assert at the start of
781            // the `outer` loop, it is guaranteed that `bit_reader.bytes_remaining() >= 8` here,
782            // which satisfies the safety precondition for `refill`. And, because the total
783            // number of bytes in `bit_reader`'s buffer plus its slice is at least 15, and
784            // `refill` moves at most 7 bytes from the slice to the buffer, the slice will still
785            // contain at least 8 bytes after this `refill` call.
786            unsafe { bit_reader.refill() };
787            // After the refill, there will be at least 8 bytes left in the bit_reader's slice.
788            debug_assert!(bit_reader.bytes_remaining() >= 8);
789
790            // in most cases, the read can be interleaved with the logic
791            // based on benchmarks this matters in practice. wild.
792            if bits as usize >= state.len_table.bits {
793                lcode[(hold & lmask) as usize]
794            } else {
795                lcode[(bit_reader.hold() & lmask) as usize]
796            }
797        };
798
799        if here.op == 0 {
800            writer.push(here.val as u8);
801            bit_reader.drop_bits(here.bits);
802            here = lcode[(bit_reader.hold() & lmask) as usize];
803
804            if here.op == 0 {
805                writer.push(here.val as u8);
806                bit_reader.drop_bits(here.bits);
807                here = lcode[(bit_reader.hold() & lmask) as usize];
808            }
809        }
810
811        'dolen: loop {
812            bit_reader.drop_bits(here.bits);
813            let op = here.op;
814
815            if op == 0 {
816                writer.push(here.val as u8);
817            } else if op & 16 != 0 {
818                let op = op & MAX_BITS;
819                let mut len = here.val + bit_reader.bits(op as usize) as u16;
820                bit_reader.drop_bits(op);
821
822                here = dcode[(bit_reader.hold() & dmask) as usize];
823
824                // we have two fast-path loads: 10+10 + 15+5 = 40,
825                // but we may need to refill here in the worst case
826                if bit_reader.bits_in_buffer() < MAX_BITS + MAX_DIST_EXTRA_BITS {
827                    debug_assert!(bit_reader.bytes_remaining() >= 8);
828                    // Safety: On the first iteration of the `dolen` loop, we can rely on the
829                    // invariant documented for the previous `refill` call above: after that
830                    // operation, `bit_reader.bytes_remining >= 8`, which satisfies the safety
831                    // precondition for this call. For subsequent iterations, this invariant
832                    // remains true because nothing else within the `dolen` loop consumes data
833                    // from the slice.
834                    unsafe { bit_reader.refill() };
835                }
836
837                'dodist: loop {
838                    bit_reader.drop_bits(here.bits);
839                    let op = here.op;
840
841                    if op & 16 != 0 {
842                        let op = op & MAX_BITS;
843                        let dist = here.val + bit_reader.bits(op as usize) as u16;
844
845                        if INFLATE_STRICT && dist as usize > state.dmax {
846                            bad = Some("invalid distance too far back\0");
847                            state.mode = Mode::Bad;
848                            break 'outer;
849                        }
850
851                        bit_reader.drop_bits(op);
852
853                        // max distance in output
854                        let written = writer.len();
855
856                        if dist as usize > written {
857                            // copy fropm the window
858                            if (dist as usize - written) > state.window.have() {
859                                if state.flags.contains(Flags::SANE) {
860                                    bad = Some("invalid distance too far back\0");
861                                    state.mode = Mode::Bad;
862                                    break 'outer;
863                                }
864
865                                panic!("INFLATE_ALLOW_INVALID_DISTANCE_TOOFAR_ARRR")
866                            }
867
868                            let mut op = dist as usize - written;
869                            let mut from;
870
871                            let window_next = state.window.next();
872
873                            if window_next == 0 {
874                                // This case is hit when the window has just wrapped around
875                                // by logic in `Window::extend`. It is special-cased because
876                                // apparently this is quite common.
877                                //
878                                // the match is at the end of the window, even though the next
879                                // position has now wrapped around.
880                                from = window_size - op;
881                            } else if window_next >= op {
882                                // the standard case: a contiguous copy from the window, no wrapping
883                                from = window_next - op;
884                            } else {
885                                // This case is hit when the window has recently wrapped around
886                                // by logic in `Window::extend`.
887                                //
888                                // The match is (partially) at the end of the window
889                                op -= window_next;
890                                from = window_size - op;
891
892                                if op < len as usize {
893                                    // This case is hit when part of the match is at the end of the
894                                    // window, and part of it has wrapped around to the start. Copy
895                                    // the end section here, the start section will be copied below.
896                                    len -= op as u16;
897                                    writer.extend_from_window_back(&state.window, from..from + op);
898                                    from = 0;
899                                    op = window_next;
900                                }
901                            }
902
903                            let copy = Ord::min(op, len as usize);
904                            writer.extend_from_window_back(&state.window, from..from + copy);
905
906                            if op < len as usize {
907                                // here we need some bytes from the output itself
908                                writer.copy_match_back(dist as usize, len as usize - op);
909                            }
910                        } else if extra_safe {
911                            todo!()
912                        } else {
913                            writer.copy_match_back(dist as usize, len as usize)
914                        }
915                    } else if (op & 64) == 0 {
916                        // 2nd level distance code
917                        here = dcode[(here.val + bit_reader.bits(op as usize) as u16) as usize];
918                        continue 'dodist;
919                    } else {
920                        bad = Some("invalid distance code\0");
921                        state.mode = Mode::Bad;
922                        break 'outer;
923                    }
924
925                    break 'dodist;
926                }
927            } else if (op & 64) == 0 {
928                // 2nd level length code
929                here = lcode[(here.val + bit_reader.bits(op as usize) as u16) as usize];
930                continue 'dolen;
931            } else if op & 32 != 0 {
932                // end of block
933                state.mode = Mode::Type;
934                break 'outer;
935            } else {
936                bad = Some("invalid literal/length code\0");
937                state.mode = Mode::Bad;
938                break 'outer;
939            }
940
941            break 'dolen;
942        }
943
944        // For normal `inflate`, include the bits in the bit_reader buffer in the count of available bytes.
945        let remaining = bit_reader.bytes_remaining();
946        if remaining >= INFLATE_FAST_MIN_HAVE && writer.remaining() >= INFLATE_FAST_MIN_LEFT {
947            continue;
948        }
949
950        break 'outer;
951    }
952
953    // return unused bytes (on entry, bits < 8, so in won't go too far back)
954    bit_reader.return_unused_bytes();
955
956    state.bit_reader = bit_reader;
957    state.writer = writer;
958
959    if let Some(error_message) = bad {
960        debug_assert!(matches!(state.mode, Mode::Bad));
961        state.bad(error_message);
962    }
963}
964
965pub fn back_end<'a>(strm: &'a mut InflateStream<'a>) {
966    // With infback the window is user-supplied, so we mustn't try to free it.
967    let _ = core::mem::replace(&mut strm.state.window, Window::empty());
968    crate::inflate::end(strm);
969}
